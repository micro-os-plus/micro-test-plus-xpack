195 template <class Lhs_T, class Rhs_T>
196 struct eq_ : type_traits::op
197 {
209 constexpr eq_ (const Lhs_T& lhs = {}, const Rhs_T& rhs = {})
210 : lhs_{ lhs }, rhs_{ rhs }, value_{ [&]
211 {
212 // This lambda is called in the constructor to
213 // evaluate the comparison.
214 using std::operator==;
215 using std::operator<;
216
217#if defined(__GNUC__)
218#pragma GCC diagnostic push
219#pragma GCC diagnostic ignored "-Wfloat-equal"
220#pragma GCC diagnostic ignored "-Wconversion"
221#pragma GCC diagnostic ignored "-Wdouble-promotion"
222#pragma GCC diagnostic ignored "-Wsign-compare"
223#if defined(__clang__)
224#pragma clang diagnostic ignored "-Wimplicit-int-float-conversion"
225#pragma clang diagnostic ignored "-Wpedantic"
226#endif
227#endif
228 if constexpr (type_traits::has_value_v<Lhs_T>
229 and type_traits::has_value_v<Rhs_T>)
230 {
231 // If both types have values (like numeric constants),
232 // compare them directly.
233 return Lhs_T::value == Rhs_T::value;
234 }
235 else if constexpr (type_traits::has_epsilon_v<Lhs_T>
236 and type_traits::has_epsilon_v<Rhs_T>)
237 {
238 // If both values have precision, compare them using
239 // the smallest precision.
240 return math::abs (get (lhs) - get (rhs))
241 < math::min_value (lhs.epsilon, rhs.epsilon);
242 }
243 else if constexpr (type_traits::has_epsilon_v<Lhs_T>)
244 {
245 // If only the left operand has precision, use it.
246 return math::abs (get (lhs) - get (rhs)) < lhs.epsilon;
247 }
248 else if constexpr (type_traits::has_epsilon_v<Rhs_T>)
249 {
250 // If only the right operand has precision, use it.
251 return math::abs (get (lhs) - get (rhs)) < rhs.epsilon;
252 }
253 else
254 {
255 // Call the generic getters, which might
256 // either call the type get() or return the value.
257 return get (lhs) == get (rhs);
258 }
259#if defined(__GNUC__)
260#pragma GCC diagnostic pop
261#endif
262 }() }
263 {
264 }
265
277 [[nodiscard]] constexpr
278 operator bool () const
279 {
280 return value_;
281 }
282
295 [[nodiscard]] constexpr auto
296 lhs (void) const
297 {
298 return get (lhs_);
299 }
300
313 [[nodiscard]] constexpr auto
314 rhs (void) const
315 {
316 return get (rhs_);
317 }
318
319 private:
323 const Lhs_T lhs_{};
324
328 const Rhs_T rhs_{};
329
333 const bool value_{};
334 };

371 {
383 constexpr ne_ (const Lhs_T& lhs = {}, const Rhs_T& rhs = {})
384 : lhs_{ lhs }, rhs_{ rhs }, value_{ [&]
385 {
386 using std::operator==;

388 using std::operator>;
389
390#if defined(__GNUC__)
391#pragma GCC diagnostic push

393#pragma GCC diagnostic ignored "-Wconversion"
394#pragma GCC diagnostic ignored "-Wdouble-promotion"
395#pragma GCC diagnostic ignored "-Wsign-compare"

403 {
404 return Lhs_T::value != Rhs_T::value;
405 }
406 else if constexpr (type_traits::has_epsilon_v<Lhs_T>
407 and type_traits::has_epsilon_v<Rhs_T>)

410 >= math::min_value (lhs_.epsilon, rhs_.epsilon);

415 }
416 else if constexpr (type_traits::has_epsilon_v<Rhs_T>)
417 {
418 return math::abs (get (lhs_) - get (rhs_)) >= rhs_.epsilon;
419 }
420 else
421 {
422 return get (lhs_) != get (rhs_);
423 }
424#if defined(__GNUC__)
