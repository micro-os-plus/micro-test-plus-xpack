239 template <typename Self_T, typename Callable_T, typename... Args_T>
240 auto
241 bind_callable (Callable_T&& callable, Args_T&&... arguments)
242 {
243 if constexpr (sizeof...(arguments) == 0)
244 {
245 return std::forward<Callable_T> (callable);
246 }
247 else
248 {
249 return bound_callable<Self_T, std::decay_t<Callable_T>,
250 std::decay_t<Args_T>...>{
251 std::forward<Callable_T> (callable),
252 std::forward<Args_T> (arguments)...
253 };
254 }
255 }

278 template <typename Self_T, typename Callable_T, typename... Args_T>
279 inline void
280 bound_callable<Self_T, Callable_T, Args_T...>::operator() (Self_T& self)
281 {
282 invoke (self, std::index_sequence_for<Args_T...>{});
283 }

293 template <typename Self_T, typename Callable_T, typename... Args_T>
294 template <std::size_t... Is>
295 inline void
296 bound_callable<Self_T, Callable_T, Args_T...>::invoke (
297 Self_T& self, std::index_sequence<Is...>)
298 {
299 if constexpr (std::is_member_pointer_v<Callable_T>)
300 {
301 std::invoke (callable_, self, std::get<Is> (arguments_)...);
302 }
303 else
304 {
305 callable_ (self, std::get<Is> (arguments_)...);
306 }
307 }

540 inline void
541 test_case::run (void)
542 {
543#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
544 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
545#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
546
547 detail::reporter& reporter = this->test_reporter ();
548
549 // For now, test cases do not record the time.
550 // this->stopwatch_.start ();
551 reporter.begin_test_case (*this);
552
553#if defined(__cpp_exceptions)
554 try
555 {
556 // Invoke the callable, passing the self reference followed by the
557 // variadic arguments.
558 callable_ (*this);
559 }
560 catch (const detail::failed_requirement&)
561 {
562 // A failed require() stopped the test case; already reported.
563 }
564 catch (const std::exception& e)
565 {
566 if (is_aborted ())
567 {
568 // Thrown by the abort handler; the report is complete.
569 throw;
570 }
571 fail_unexpected_exception (e.what ());
572 }
573 catch (...)
574 {
575 if (is_aborted ())
576 {
577 // Thrown by the abort handler; the report is complete.
578 throw;
579 }
580 fail_unexpected_exception (nullptr);
581 }
582#else
583 // Invoke the callable, passing the self reference followed by the variadic
584 // arguments.
585 callable_ (*this);
586#endif // defined(__cpp_exceptions)
587
588 // this->stopwatch_.stop ();
589 reporter.end_test_case (*this);
590 }

671 template <typename Callable_T, typename... Args_T>
672 void
673 suite::test (const test_case_name& name, Callable_T&& callable,
674 Args_T&&... arguments)
675 {
676#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
677 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
678#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
679
680 // Called from inside a running test case; run nested in it.
681 if (class test_case* running = running_test_case (); running != nullptr)
682 {
683 running->test (name, std::forward<Callable_T> (callable),
684 std::forward<Args_T> (arguments)...);
685 return;
686 }
687
688 create_test_case (name, *this, 1, std::forward<Callable_T> (callable),
689 std::forward<Args_T> (arguments)...);
690 }

707 template <class Self_T, class Values_T, class Callable_T>
708 void
709 test_each_value (Self_T& self, const test_case_name& name,
710 Values_T&& values, Callable_T& callable)
711 {
712 std::size_t position = 0;
713
714 const auto run_one
715 = [&self, &name, &callable, &position] (const auto& value)
716 {
717 ++position;
718
719 std::string text{ name.name () };
720 text += " (";
721 if constexpr (extension::printable<
722 std::remove_cvref_t<decltype (value)>>)
723 {
724 plain_text::append (text, value);
725 }
726 else
727 {
728 text += '#';
729 plain_text::append (text, position);
730 }
731 text += ')';
732
733 // The test case runs within this call, so the value and the
734 // callable are captured by reference, not copied.
735 self.test (test_case_name{ text, name.location () },
736 [&callable, &value] (auto& test_self)
737 {
738 if constexpr (std::is_member_pointer_v<
739 std::remove_cvref_t<Callable_T>>)
740 {
741 std::invoke (callable, test_self, value);
742 }
743 else
744 {
745 callable (test_self, value);
746 }
747 });
748 };
749
750 if constexpr (test_values_range<Values_T>)
751 {
752 for (const auto& value : values)
753 {
754 run_one (value);
755 }
756 }
757 else
758 {
759 std::apply ([&run_one] (const auto&... elements)
760 { (run_one (elements), ...); }, values);
761 }
762 }
