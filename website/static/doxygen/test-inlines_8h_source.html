294 template <typename Callable_T, typename... Args_T>
295 inline void
296 runnable_base::create_test_case (const located_name& name,
297 suite& parent_suite, size_t nesting_depth,
298 Callable_T&& callable,
299 Args_T&&... arguments)
300 {
301 size_t own_index = next_test_point_number ();
302
303 if constexpr (sizeof...(arguments) == 0
304 && !std::is_function_v<
305 std::remove_reference_t<Callable_T>>)
306 {
307 // The test case runs and ends within this call, so it refers
308 // to the callable of the caller, which is not copied, and is
309 // called in place.
310 class test_case child_test_case{
311 name, runner_, parent_suite,
312 own_index, nesting_depth, callable_ref<class test_case>{ callable }
313 };
314
315 after_test_case_create (child_test_case, parent_suite);
316 }
317 else
318 {
319 auto body = bind_callable<class test_case> (
320 std::forward<Callable_T> (callable),
321 std::forward<Args_T> (arguments)...);
322 class test_case child_test_case{
323 name, runner_, parent_suite,
324 own_index, nesting_depth, callable_ref<class test_case>{ body }
325 };
326
327 after_test_case_create (child_test_case, parent_suite);
328 }
329 }

346 template <typename Self_T, typename Callable_T, typename... Args_T>
347 auto
348 bind_callable (Callable_T&& callable, Args_T&&... arguments)
349 {
350 if constexpr (sizeof...(arguments) == 0)
351 {
352 return std::forward<Callable_T> (callable);
353 }
354 else
355 {
356 return bound_callable<Self_T, std::decay_t<Callable_T>,
357 std::decay_t<Args_T>...>{
358 std::forward<Callable_T> (callable),
359 std::forward<Args_T> (arguments)...
360 };
361 }
362 }

372 template <typename C, typename... A>
373 requires (sizeof...(A) > 0)
374 constexpr bound_callable<Self_T, Callable_T, Args_T...>::bound_callable (
375 C&& callable, A&&... arguments)
376 : callable_ (std::forward<C> (callable)),
377 arguments_ (std::forward<A> (arguments)...)

385 template <typename Self_T, typename Callable_T, typename... Args_T>
386 inline void
387 bound_callable<Self_T, Callable_T, Args_T...>::operator() (Self_T& self)
388 {
389 invoke (self, std::index_sequence_for<Args_T...>{});
390 }

400 template <typename Self_T, typename Callable_T, typename... Args_T>
401 template <std::size_t... Is>
402 inline void
403 bound_callable<Self_T, Callable_T, Args_T...>::invoke (
404 Self_T& self, std::index_sequence<Is...>)
405 {
406 if constexpr (std::is_member_pointer_v<Callable_T>)
407 {
408 std::invoke (callable_, self, std::get<Is> (arguments_)...);
409 }
410 else
411 {
412 callable_ (self, std::get<Is> (arguments_)...);
413 }
414 }

424 template <typename Callable_T, typename... Args_T>
425 constexpr void
426 reject_test_body (void) noexcept
427 {
428 if constexpr (std::is_invocable_v<std::decay_t<Callable_T>&, suite&,
429 std::decay_t<Args_T>&...>)
430 {
431 static_assert (dependent_false_v<Callable_T, Args_T...>,
432 "the body of a test case takes a test_case& as its "
433 "first parameter, not a suite&");
434 }
435 else
436 {
437 static_assert (dependent_false_v<Callable_T, Args_T...>,
438 "the body of a test case takes a test_case& as its "
439 "first parameter, followed by the additional "
440 "arguments passed to test ()");
441 }
442 }
443
450 template <typename Callable_T, typename... Args_T>
451 constexpr void
452 reject_suite_body (void) noexcept
453 {
454 if constexpr (not std::is_copy_constructible_v<std::decay_t<Callable_T>>
455 or not(std::is_copy_constructible_v<std::decay_t<Args_T>>
456 and ...))
457 {
458 static_assert (dependent_false_v<Callable_T, Args_T...>,
459 "the body of a suite and its additional arguments "
460 "must be copyable, since the suite runs later");
461 }
462 else if constexpr (std::is_invocable_v<std::decay_t<Callable_T>&,
463 test_case&,
464 std::decay_t<Args_T>&...>)
465 {
466 static_assert (dependent_false_v<Callable_T, Args_T...>,
467 "the body of a suite takes a suite& as its first "
468 "parameter, not a test_case&");
469 }
470 else
471 {
472 static_assert (dependent_false_v<Callable_T, Args_T...>,
473 "the body of a suite takes a suite& as its first "
474 "parameter, followed by the additional arguments");
475 }
476 }

486 callable_ref<Self_T>>)
487 callable_ref<Self_T>::callable_ref (Callable_T& callable) noexcept
488 : callable_{ const_cast<void*> (
489 static_cast<const volatile void*> (std::addressof (callable))) },

499 inline void
500 callable_ref<Self_T>::operator() (Self_T& self) const
501 {
502 invoke_ (callable_, self);
503 }

516 template <typename Self_T>
517 template <typename Callable_T>
518 void
519 callable_ref<Self_T>::invoke_callable (void* callable, Self_T& self)
520 {
521 if constexpr (std::is_member_pointer_v<Callable_T>)
522 {
523 std::invoke (*static_cast<Callable_T*> (callable), self);
524 }
525 else
526 {
527 (*static_cast<Callable_T*> (callable)) (self);

818 inline void
819 test_case::run (void)
820 {
821#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
822 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
823#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
824
825 detail::reporter& reporter = this->test_reporter ();
826
827 // For now, test cases do not record the time.
828 // this->stopwatch_.start ();
829 reporter.begin_test_case (*this);
830
831#if defined(__cpp_exceptions)
832 try
833 {
834 // Invoke the callable, passing the self reference; the arguments
835 // of test() are bound to it, by bind_callable().
836 callable_ (*this);
837 }
838 catch (const detail::failed_requirement&)
839 {
840 // A failed require() stopped the test case; already reported.
841 }
842 catch (const std::exception& e)
843 {
844 if (is_aborted ())
845 {
846 // Thrown by the abort handler; the report is complete.
847 throw;
848 }
849 fail_unexpected_exception (e.what ());
850 }
851 catch (...)
852 {
853 if (is_aborted ())
854 {
855 // Thrown by the abort handler; the report is complete.
856 throw;
857 }
858 fail_unexpected_exception (nullptr);
859 }
860#else
861 // Invoke the callable, passing the self reference; the arguments of
862 // test() are bound to it, by bind_callable().
863 callable_ (*this);
864#endif // defined(__cpp_exceptions)
865
866 // this->stopwatch_.stop ();
867 reporter.end_test_case (*this);
868 }

957 template <typename Callable_T, typename... Args_T>
958 requires detail::test_body<Callable_T, Args_T...>
959 void
960 suite::test (const located_name& name, Callable_T&& callable,
961 Args_T&&... arguments)
962 {
963#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
964 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
965#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
966
967 // Called from inside a running test case of this suite; run nested in
968 // it. A running test case of another suite does not qualify; the call
969 // is then rejected below, as for any suite that is not running.
970 if (class test_case* running = running_test_case ();
971 running != nullptr && has_running_test_case_of (*this))
972 {
973 running->test (name, std::forward<Callable_T> (callable),
974 std::forward<Args_T> (arguments)...);
975 return;
976 }
977
978 if (!accepts_test_case (*this, name))
979 {
980 return;
981 }
982
983 create_test_case (name, *this, 1, std::forward<Callable_T> (callable),
984 std::forward<Args_T> (arguments)...);
985 }

1017 template <class Self_T, class Values_T, class Callable_T>
1018 void
1019 test_each_value (Self_T& self, const located_name& name, Values_T&& values,
1020 Callable_T& callable)
1021 {
1022 std::size_t position = 0;
1023
1024 const auto run_one
1025 = [&self, &name, &callable, &position] (const auto& value)
1026 {
1027 ++position;
1028
1029 std::string text{ name.name () };
1030 text += " (";
1031 if constexpr (extension::printable<
1032 std::remove_cvref_t<decltype (value)>>)
1033 {
1034 plain_text::append (text, value);
1035 }
1036 else
1037 {
1038 text += '#';
1039 plain_text::append (text, position);
1040 }
1041 text += ')';
1042
1043 // The test case runs within this call, so the value and the
1044 // callable are captured by reference, not copied.
1045 self.test (located_name{ std::move (text), name.location () },
1046 [&callable, &value] (auto& test_self)
1047 {
1048 if constexpr (std::is_member_pointer_v<
1049 std::remove_cvref_t<Callable_T>>)
1050 {
1051 std::invoke (callable, test_self, value);
1052 }
1053 else
1054 {
1055 callable (test_self, value);
1056 }
1057 });
1058 };
1059
1060 if constexpr (test_values_range<Values_T>)
1061 {
1062 for (const auto& value : values)
1063 {
1064 run_one (value);
1065 }
1066 }
1067 else
1068 {
1069 std::apply ([&run_one] (const auto&... elements)
1070 { (run_one (elements), ...); }, values);
1071 }
1072 }

1080 template <typename Values_T, typename Callable_T>
1081 requires (detail::test_values_range<Values_T>
1082 or detail::test_values_tuple<Values_T>)
1083 void
1084 test_case::test_each (const located_name& name, Values_T&& values,
