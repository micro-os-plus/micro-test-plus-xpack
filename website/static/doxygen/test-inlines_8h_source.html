189 template <typename Self_T>
190 template <typename Callable_T, typename... Args_T>
191 runnable<Self_T>::runnable (const char* name, class runner& runner,
192 size_t own_index, Callable_T&& callable,
193 Args_T&&... arguments)
194 : runnable_base{ name, runner, own_index }
195 {
196 // When there are no extra arguments the callable already has the
197 // signature void(Self_T&), so store it directly. Only use std::bind when
198 // additional arguments must be pre-bound, to avoid triggering a GCC ARM
199 // bug in
200 // __is_nothrow_invocable<_Bind<...>, Self_T&> (GCC 15.2.1).
201 if constexpr (sizeof...(arguments) == 0)
202 {
203 callable_ = std::forward<Callable_T> (callable);
204 }
205 else
206 {
207 callable_ = std::bind (std::forward<Callable_T> (callable),
208 std::placeholders::_1,
209 std::forward<Args_T> (arguments)...);
210 }
211
212#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
213 trace::printf ("%s '%s' %zu\n", __PRETTY_FUNCTION__, name, own_index_);
214#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
215 }

347 /**
348 * @details
349 * The `test` function template registers and executes a test case
350 * within the µTest++ framework. It accepts a descriptive name, a callable
351 * object (such as a lambda or function pointer), and an optional list of
352 * arguments to be passed to the callable. The test case is reported using
353 * the provided name, and its execution is managed by the framework's test
354 * runner.
355 *
356 * Each test case typically involves evaluating a logical expression, such as
357 * comparing a computed result to an expected value. For C++ projects, it is
358 * also possible to verify whether evaluating an expression throws
359 * exceptions. Each test either succeeds or fails, and for expectations, the
360 * test runner maintains counts of successful and failed checks.
361 *
362 * This function template enables flexible and expressive test case
363 * definitions, supporting both parameterised and non-parameterised tests. It
364 * is typically invoked at global scope or within test suite definitions to
365 * ensure automatic registration and execution.
366 *
367 * A test case is characterised by a name, a function that performs the
368 * checks, and optionally, arguments to be passed to that function. The
369 * implementation of `test` invokes the provided function with the given
370 * arguments and reports the results to the test runner.
371 *
372 * @par Example
373 *
374 * @code{.cpp}
375 * namespace mt = micro_os_plus::micro_test_plus;
376 *
377 * ts.test ("Check answer with comparator", [] (auto& t) {
378 * t.expect (mt::eq (compute_answer (), 42)) << "answer is 42";
379 * });
380 * @endcode
381 */
382
383 template <typename Callable_T, typename... Args_T>
384 void
385 suite::test (const char* name, Callable_T&& callable, Args_T&&... arguments)
386 {
387#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
388 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name);
389#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
390
391 size_t own_index = increment_subtest_index ();
392 auto child_subtest
393 = std::make_unique<subtest> (name, runner (), *this, own_index, 1,
394 std::forward<Callable_T> (callable),
395 std::forward<Args_T> (arguments)...);
396
397 after_subtest_create_ (std::move (child_subtest), *this);
398 }

443 template <typename Callable_T, typename... Args_T>
444 static_suite::static_suite (const char* name, static_runner& runner,
445 Callable_T&& callable, Args_T&&... arguments)
446 // The nullptr passed to the base constructor is an optimisation to save
447 // some space, since this callble is not used by the static runner.
448 : suite{ name, detail::to_runner (runner), nullptr }
449 {
450 if constexpr (sizeof...(arguments) == 0)
451 {
452 static_callable_ = std::forward<Callable_T> (callable);
453 }
454 else
455 {
456 static_callable_ = std::bind (std::forward<Callable_T> (callable),
457 std::placeholders::_1,
458 std::forward<Args_T> (arguments)...);
459 }
460
461#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)

463#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
464
465 detail::register_static_suite (runner, *this);
