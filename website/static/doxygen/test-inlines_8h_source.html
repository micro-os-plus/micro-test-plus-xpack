302 template <typename Self_T, typename Callable_T, typename... Args_T>
303 auto
304 bind_callable (Callable_T&& callable, Args_T&&... arguments)
305 {
306 if constexpr (sizeof...(arguments) == 0)
307 {
308 return std::forward<Callable_T> (callable);
309 }
310 else
311 {
312 return bound_callable<Self_T, std::decay_t<Callable_T>,
313 std::decay_t<Args_T>...>{
314 std::forward<Callable_T> (callable),
315 std::forward<Args_T> (arguments)...
316 };
317 }
318 }

341 template <typename Self_T, typename Callable_T, typename... Args_T>
342 inline void
343 bound_callable<Self_T, Callable_T, Args_T...>::operator() (Self_T& self)
344 {
345 invoke (self, std::index_sequence_for<Args_T...>{});
346 }

356 template <typename Self_T, typename Callable_T, typename... Args_T>
357 template <std::size_t... Is>
358 inline void
359 bound_callable<Self_T, Callable_T, Args_T...>::invoke (
360 Self_T& self, std::index_sequence<Is...>)
361 {
362 if constexpr (std::is_member_pointer_v<Callable_T>)
363 {
364 std::invoke (callable_, self, std::get<Is> (arguments_)...);
365 }
366 else
367 {
368 callable_ (self, std::get<Is> (arguments_)...);
369 }
370 }

415 {

671 inline void
672 test_case::run (void)
673 {
674#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
675 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
676#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
677
678 detail::reporter& reporter = this->test_reporter ();
679
680 // For now, test cases do not record the time.
681 // this->stopwatch_.start ();
682 reporter.begin_test_case (*this);
683
684#if defined(__cpp_exceptions)
685 try
686 {
687 // Invoke the callable, passing the self reference; the arguments
688 // of test() are bound to it, by bind_callable().
689 callable_ (*this);
690 }
691 catch (const detail::failed_requirement&)
692 {
693 // A failed require() stopped the test case; already reported.
694 }
695 catch (const std::exception& e)
696 {
697 if (is_aborted ())
698 {
699 // Thrown by the abort handler; the report is complete.
700 throw;
701 }
702 fail_unexpected_exception (e.what ());
703 }
704 catch (...)
705 {
706 if (is_aborted ())
707 {
708 // Thrown by the abort handler; the report is complete.
709 throw;
710 }
711 fail_unexpected_exception (nullptr);
712 }
713#else
714 // Invoke the callable, passing the self reference; the arguments of
715 // test() are bound to it, by bind_callable().
716 callable_ (*this);
717#endif // defined(__cpp_exceptions)
718
719 // this->stopwatch_.stop ();
720 reporter.end_test_case (*this);
721 }

810 template <typename Callable_T, typename... Args_T>
811 requires detail::test_body<Callable_T, Args_T...>
812 void
813 suite::test (const located_name& name, Callable_T&& callable,
814 Args_T&&... arguments)
815 {
816#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
817 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
818#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
819
820 // Called from inside a running test case; run nested in it.
821 if (class test_case* running = running_test_case (); running != nullptr)
822 {
823 running->test (name, std::forward<Callable_T> (callable),
824 std::forward<Args_T> (arguments)...);
825 return;
826 }
827
828 if (!accepts_test_case (*this, name.name ()))
829 {
830 return;
831 }
832
833 create_test_case (name, *this, 1, std::forward<Callable_T> (callable),
834 std::forward<Args_T> (arguments)...);
835 }

852 template <class Self_T, class Values_T, class Callable_T>
853 void
854 test_each_value (Self_T& self, const located_name& name, Values_T&& values,
855 Callable_T& callable)
856 {
857 std::size_t position = 0;
858
859 const auto run_one
860 = [&self, &name, &callable, &position] (const auto& value)
861 {
862 ++position;
863
864 std::string text{ name.name () };
865 text += " (";
866 if constexpr (extension::printable<
867 std::remove_cvref_t<decltype (value)>>)
868 {
869 plain_text::append (text, value);
870 }
871 else
872 {
873 text += '#';
874 plain_text::append (text, position);
875 }
876 text += ')';
877
878 // The test case runs within this call, so the value and the
879 // callable are captured by reference, not copied.
880 self.test (located_name{ std::move (text), name.location () },
881 [&callable, &value] (auto& test_self)
882 {
883 if constexpr (std::is_member_pointer_v<
884 std::remove_cvref_t<Callable_T>>)
885 {
886 std::invoke (callable, test_self, value);
887 }
888 else
889 {
890 callable (test_self, value);
891 }
892 });
893 };
894
895 if constexpr (test_values_range<Values_T>)
896 {
897 for (const auto& value : values)
898 {
899 run_one (value);
900 }
901 }
902 else
903 {
904 std::apply ([&run_one] (const auto&... elements)
905 { (run_one (elements), ...); }, values);
906 }
907 }
