248 template <typename Self_T, typename Callable_T, typename... Args_T>
249 auto
250 bind_callable (Callable_T&& callable, Args_T&&... arguments)
251 {
252 if constexpr (sizeof...(arguments) == 0)
253 {
254 return std::forward<Callable_T> (callable);
255 }
256 else
257 {
258 return bound_callable<Self_T, std::decay_t<Callable_T>,
259 std::decay_t<Args_T>...>{
260 std::forward<Callable_T> (callable),
261 std::forward<Args_T> (arguments)...
262 };
263 }
264 }

302 template <typename Self_T, typename Callable_T, typename... Args_T>
303 template <std::size_t... Is>
304 inline void
305 bound_callable<Self_T, Callable_T, Args_T...>::invoke (
306 Self_T& self, std::index_sequence<Is...>)
307 {
308 if constexpr (std::is_member_pointer_v<Callable_T>)
309 {
310 std::invoke (callable_, self, std::get<Is> (arguments_)...);
311 }
312 else
313 {
314 callable_ (self, std::get<Is> (arguments_)...);
315 }
316 }

510 inline void
511 test_case::run (void)
512 {
513#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
514 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
515#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
516
517 detail::reporter& reporter = this->test_reporter ();
518
519 // For now, test cases do not record the time.
520 // this->stopwatch_.start ();
521 reporter.begin_test_case (*this);
522
523#if defined(__cpp_exceptions)
524 try
525 {
526 // Invoke the callable, passing the self reference followed by the
527 // variadic arguments.
528 callable_ (*this);
529 }
530 catch (const detail::failed_requirement&)
531 {
532 // A failed require() stopped the test case; already reported.
533 }
534 catch (const std::exception& e)
535 {
536 if (is_aborted ())
537 {
538 // Thrown by the abort handler; the report is complete.
539 throw;
540 }
541 fail_unexpected_exception (e.what ());
542 }
543 catch (...)
544 {
545 if (is_aborted ())
546 {
547 // Thrown by the abort handler; the report is complete.
548 throw;
549 }
550 fail_unexpected_exception (nullptr);
551 }
552#else
553 // Invoke the callable, passing the self reference followed by the variadic
554 // arguments.
555 callable_ (*this);
556#endif // defined(__cpp_exceptions)
557
558 // this->stopwatch_.stop ();
559 reporter.end_test_case (*this);
560 }

641 template <typename Callable_T, typename... Args_T>
642 void
643 suite::test (const test_case_name& name, Callable_T&& callable,
644 Args_T&&... arguments)
645 {
646#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
647 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
648#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
649
650 // Called from inside a running test case; run nested in it.
651 if (class test_case* running = running_test_case (); running != nullptr)
652 {
653 running->test (name, std::forward<Callable_T> (callable),
654 std::forward<Args_T> (arguments)...);
655 return;
656 }
657
658 create_test_case (name, *this, 1, std::forward<Callable_T> (callable),
659 std::forward<Args_T> (arguments)...);
660 }

675 template <class Self_T, class Values_T, class Callable_T>
676 void
677 test_each_value (Self_T& self, const test_case_name& name,
678 Values_T&& values, Callable_T& callable)
679 {
680 std::size_t position = 0;
681
682 const auto run_one
683 = [&self, &name, &callable, &position] (const auto& value)
684 {
685 ++position;
686
687 std::string text{ name.name () };
688 text += " (";
689 if constexpr (extension::printable<
690 std::remove_cvref_t<decltype (value)>>)
691 {
692 plain_text::append (text, value);
693 }
694 else
695 {
696 text += '#';
697 plain_text::append (text, position);
698 }
699 text += ')';
700
701 self.test (test_case_name{ text, name.location () }, callable,
702 value);
703 };
704
705 if constexpr (test_values_range<Values_T>)
706 {
707 for (const auto& value : values)
708 {
709 run_one (value);
710 }
711 }
712 else
713 {
714 std::apply ([&run_one] (const auto&... elements)
715 { (run_one (elements), ...); }, values);
716 }
717 }
