276 template <typename Self_T, typename Callable_T, typename... Args_T>
277 auto
278 bind_callable (Callable_T&& callable, Args_T&&... arguments)
279 {
280 if constexpr (sizeof...(arguments) == 0)
281 {
282 return std::forward<Callable_T> (callable);
283 }
284 else
285 {
286 return bound_callable<Self_T, std::decay_t<Callable_T>,
287 std::decay_t<Args_T>...>{
288 std::forward<Callable_T> (callable),
289 std::forward<Args_T> (arguments)...
290 };
291 }
292 }

315 template <typename Self_T, typename Callable_T, typename... Args_T>
316 inline void
317 bound_callable<Self_T, Callable_T, Args_T...>::operator() (Self_T& self)
318 {
319 invoke (self, std::index_sequence_for<Args_T...>{});
320 }
321
330 template <typename Self_T, typename Callable_T, typename... Args_T>
331 template <std::size_t... Is>
332 inline void
333 bound_callable<Self_T, Callable_T, Args_T...>::invoke (
334 Self_T& self, std::index_sequence<Is...>)
335 {
336 if constexpr (std::is_member_pointer_v<Callable_T>)
337 {
338 std::invoke (callable_, self, std::get<Is> (arguments_)...);
339 }
340 else
341 {
342 callable_ (self, std::get<Is> (arguments_)...);
343 }
344 }

577 inline void
578 test_case::run (void)
579 {
580#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
581 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
582#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
583
584 detail::reporter& reporter = this->test_reporter ();
585
586 // For now, test cases do not record the time.
587 // this->stopwatch_.start ();
588 reporter.begin_test_case (*this);
589
590#if defined(__cpp_exceptions)
591 try
592 {
593 // Invoke the callable, passing the self reference followed by the
594 // variadic arguments.
595 callable_ (*this);
596 }
597 catch (const detail::failed_requirement&)
598 {
599 // A failed require() stopped the test case; already reported.
600 }
601 catch (const std::exception& e)
602 {
603 if (is_aborted ())
604 {
605 // Thrown by the abort handler; the report is complete.
606 throw;
607 }
608 fail_unexpected_exception (e.what ());
609 }
610 catch (...)
611 {
612 if (is_aborted ())
613 {
614 // Thrown by the abort handler; the report is complete.
615 throw;
616 }
617 fail_unexpected_exception (nullptr);
618 }
619#else
620 // Invoke the callable, passing the self reference followed by the variadic
621 // arguments.
622 callable_ (*this);
623#endif // defined(__cpp_exceptions)
624
625 // this->stopwatch_.stop ();
626 reporter.end_test_case (*this);
627 }

708 template <typename Callable_T, typename... Args_T>
709 void
710 suite::test (const test_case_name& name, Callable_T&& callable,
711 Args_T&&... arguments)
712 {
713#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
714 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
715#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
716
717 // Called from inside a running test case; run nested in it.
718 if (class test_case* running = running_test_case (); running != nullptr)
719 {
720 running->test (name, std::forward<Callable_T> (callable),
721 std::forward<Args_T> (arguments)...);
722 return;
723 }
724
725 create_test_case (name, *this, 1, std::forward<Callable_T> (callable),
726 std::forward<Args_T> (arguments)...);
727 }

744 template <class Self_T, class Values_T, class Callable_T>
745 void
746 test_each_value (Self_T& self, const test_case_name& name,
747 Values_T&& values, Callable_T& callable)
748 {
749 std::size_t position = 0;
750
751 const auto run_one
752 = [&self, &name, &callable, &position] (const auto& value)
753 {
754 ++position;
755
756 std::string text{ name.name () };
757 text += " (";
758 if constexpr (extension::printable<
759 std::remove_cvref_t<decltype (value)>>)
760 {
761 plain_text::append (text, value);
762 }
763 else
764 {
765 text += '#';
766 plain_text::append (text, position);
767 }
768 text += ')';
769
770 // The test case runs within this call, so the value and the
771 // callable are captured by reference, not copied.
772 self.test (test_case_name{ text, name.location () },
773 [&callable, &value] (auto& test_self)
774 {
775 if constexpr (std::is_member_pointer_v<
776 std::remove_cvref_t<Callable_T>>)
777 {
778 std::invoke (callable, test_self, value);
779 }
780 else
781 {
782 callable (test_self, value);
783 }
784 });
785 };
786
787 if constexpr (test_values_range<Values_T>)
788 {
789 for (const auto& value : values)
790 {
791 run_one (value);
792 }
793 }
794 else
795 {
796 std::apply ([&run_one] (const auto&... elements)
797 { (run_one (elements), ...); }, values);
798 }
799 }
