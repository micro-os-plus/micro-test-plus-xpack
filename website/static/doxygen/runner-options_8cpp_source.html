109 runner_options::runner_options (int argc, char* argv[])
110 {
111#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
112 trace::printf ("%s\n", __PRETTY_FUNCTION__);
113#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED

115 if (argc <= 0 || argv == nullptr)
116 {
117 return;
118 }
119
120 // Copy the strings one by one, since the vector assignment uses
121 // std::copy, which triggers -Waggregate-return in the standard library
122 // headers on some targets.
123 arguments.reserve (static_cast<size_t> (argc));
124 for (int i = 0; i < argc; ++i)
125 {
126 arguments.emplace_back (argv[i] != nullptr ? argv[i] : "");
127 }
128
129 static constexpr std::string_view reporter_option{ "--reporter" };
130 static constexpr std::string_view output_file_option{ "--output-file" };
131 static constexpr std::string_view color_option{ "--color" };
132 static constexpr std::string_view use_colour_option{ "--use-colour" };
133 static constexpr std::string_view filter_option{ "--filter" };
134
135 // Get the value of an option given as `--option=<value>` or as
136 // `--option <value>`; the latter advances the index past the value.
137 // Return false if the argument is not this option, or if its value is
138 // missing, which is reported as an error.
139 auto get_value = [&] (std::string_view option, size_t& i,
140 std::string_view& value) -> bool
141 {
142 const std::string_view argument{ arguments[i] };
143 if (argument == option)
144 {
145 if (i + 1 >= arguments.size ())
146 {
147 add_error ("", option, " option requires an argument");
148 return false;
149 }
150 value = arguments[++i];
151 return true;
152 }
153 if (argument.starts_with (option)
154 && argument.substr (option.size ()).starts_with ('='))
155 {
156 value = argument.substr (option.size () + 1);
157 return true;
158 }
159 return false;
160 };
161
162 // Whether the argument is one of the options with a value, possibly
163 // with the value missing, which was already reported.
164 auto is_known_option = [&] (std::string_view argument) -> bool
165 {
166 for (const std::string_view option :
167 { reporter_option, output_file_option, use_colour_option,
168 filter_option })
169 {
170 if (argument == option
171 || (argument.starts_with (option)
172 && argument.substr (option.size ()).starts_with ('=')))
173 {
174 return true;
175 }
176 }
177 return false;
178 };
179
180 // Set the colour mode from the value of `--color` or of
181 // `--use-colour`; both accept the values and the synonyms of the GNU
182 // tools, which include those of Catch2 and Boost UT (`yes`, `no`, and
183 // `auto`). An unknown mode is reported as an error.
184 auto set_colour_mode = [&] (std::string_view value)
185 {
186 if (value == "always" || value == "yes" || value == "force")
187 {
188 colour = colour_mode::always;
189 }
190 else if (value == "never" || value == "no" || value == "none")
191 {
192 colour = colour_mode::never;
193 }
194 else if (value == "auto" || value == "tty" || value == "if-tty")
195 {
196 colour = colour_mode::automatic;
197 }
198 else
199 {
200 add_error ("unknown colour mode '", value, "'");
201 }
202 };
203
204 for (size_t i = 1; i < arguments.size (); ++i)
205 {
206 const std::string_view argument{ arguments[i] };
207 std::string_view value{};
208
209 if (argument == "--verbose")
210 {
211 verbosity_level = verbosity::verbose;
212 }
213 else if (argument == "--quiet")
214 {
215 verbosity_level = verbosity::quiet;
216 }
217 else if (argument == "--silent")
218 {
219 verbosity_level = verbosity::silent;
220 }
221 else if (argument == "--list-tests" || argument == "--list")
222 {
223 list_tests = true;
224 }
225 else if (get_value (reporter_option, i, value))
226 {
227 if (value == "human")
228 {
229 selected_reporter = reporter_kind::human;
230 }
231 else if (value == "tap")
232 {
233 selected_reporter = reporter_kind::tap;
234 }
235 else
236 {
237 add_error ("unknown reporter '", value, "'");
238 has_invalid_reporter_ = true;
239 }
240 }
241 else if (get_value (output_file_option, i, value))
242 {
243 // An empty path disables the output file.
244 output_file_path = value;
245 }
246 else if (get_value (use_colour_option, i, value))
247 {
248 set_colour_mode (value);
249 }
250 else if (argument == color_option)
251 {
252 // As in the GNU tools, the option alone means `always`, and
253 // the next argument is never taken as its value.
254 colour = colour_mode::always;
255 }
256 else if (argument.starts_with (color_option)
257 && argument.substr (color_option.size ()).starts_with ('='))
258 {
259 set_colour_mode (argument.substr (color_option.size () + 1));
260 }
261 else if (get_value (filter_option, i, value))
262 {
263 filter_patterns.emplace_back (value);
264 }
265 else if (argument.starts_with ("--") && !is_known_option (argument))
266 {
267 // The other arguments are left to the application, but a
268 // mistyped option must not silently disable the tests.
269 add_error ("unknown option '", argument, "'");
270 }
271 }
272 }
