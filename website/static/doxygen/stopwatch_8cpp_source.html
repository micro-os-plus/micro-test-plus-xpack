115 std::chrono::microseconds
116 stopwatch::elapsed (void) const noexcept
117 {
118 if (!has_elapsed ())
119 {
120 return std::chrono::microseconds::zero ();
121 }
122
123 const std::chrono::nanoseconds delta = std::chrono::seconds{
124 stop_time_.tv_sec - start_time_.tv_sec
125 } + std::chrono::nanoseconds{ stop_time_.tv_nsec - start_time_.tv_nsec };
126
127 return std::chrono::duration_cast<std::chrono::microseconds> (delta);
128 }
