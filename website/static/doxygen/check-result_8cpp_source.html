139 check_result::~check_result () noexcept (false)
140 {
141#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
142 trace::printf ("%s\n", __PRETTY_FUNCTION__);
143#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
144
145 // A result that was not initialised, such as a default argument
146 // whose function did not run, reports nothing.
147 if (test_case_ == nullptr) [[unlikely]]
148 {
149 return;
150 }
151
152 // The index starts at 0; it must be incremented before the check is
153 // reported, so that the first check is reported as check #1.
154 test_case_->next_test_point_number ();
155
156 // The text formatted by this check, even if the formatter was used by
157 // other checks in the meantime (when this result was kept).
158 auto& expression = test_case_->test_reporter ().expression_;
159 const std::string& expression_str = expression.check_text (this);
160
161 if (value_) [[likely]]
162 {
163 if (is_written_)
164 {
165 test_case_->test_reporter ().pass (info_.message, expression_str,
166 *test_case_);
167 }
168 else
169 {
170 test_case_->test_reporter ().pass_without_text (*test_case_);
171 }
172 test_case_->totals_.increment_passed_checks ();
173 }
174 else
175 {
176 test_case_->test_reporter ().fail (info_, expression_str,
177 *test_case_);
178 test_case_->totals_.increment_failed_checks ();
179 }
180
181 expression.end_check (this);
182
183 if (info_.kind == check_kind::requirement && !value_) [[unlikely]]
184 {
185#if defined(__cpp_exceptions)
186 if (std::uncaught_exceptions () <= uncaught_exceptions_)
187 {
188 throw failed_requirement{};
189 }
190#else
191 test_case_->abort (info_.location);
192#endif // defined(__cpp_exceptions)
193 }
194 }
