134 [[nodiscard]] bool
135 is_match (std::string_view input, std::string_view pattern)
136 {
137 constexpr auto npos = std::string_view::npos;
138
139 std::size_t i = 0; // Index in the input.
140 std::size_t p = 0; // Index in the pattern.
141
142 // The position of the most recent `*` in the pattern, and the input
143 // index where its match ends.
144 std::size_t star_p = npos;
145 std::size_t star_i = 0;
146
147 while (i < input.size ())
148 {
149 if (p < pattern.size ()
150 && (pattern[p] == '?' || pattern[p] == input[i]))
151 {
152 ++i;
153 ++p;
154 }
155 else if (p < pattern.size () && pattern[p] == '*')
156 {
157 // Start by matching an empty sequence.
158 star_p = p++;
159 star_i = i;
160 }
161 else if (star_p != npos)
162 {
163 // Extend the match of the most recent `*` by one character.
164 p = star_p + 1;
165 i = ++star_i;
166 }
167 else
168 {
169 return false;
170 }
171 }
172
173 // The input is exhausted; the rest of the pattern must be only `*`.
174 while (p < pattern.size () && pattern[p] == '*')
175 {
176 ++p;
177 }
178
179 return p == pattern.size ();
180 }

201 [[nodiscard]] std::vector<std::string_view>
202 split (std::string_view input, std::string_view delimiters)
203 {
204 std::vector<std::string_view> output{};
205 std::size_t first{};
206 while (first < input.size ())
207 {
208 const auto second = input.find_first_of (delimiters, first);
209 if (first != second)
210 {
211 output.emplace_back (input.substr (first, second - first));
212 }
213 if (second == std::string_view::npos)
214 {
215 break;
216 }
217 first = second + 1;
218 }
219 return output;
220 }
