260 template <class T>
261 requires detail::formattable_range<T>
262 expression_formatter&
263 expression_formatter::operator<< (const T& t)
264 {
265 *this << "{";
266 size_t count = 0;
267 for (const auto& arg : t)
268 {
269 if (count < max_range_elements_)
270 {
271 *this << (count == 0 ? "" : ", ");
272 using element_type = std::remove_cvref_t<decltype (arg)>;
273 if constexpr (std::is_same_v<element_type, signed char>
274 or std::is_same_v<element_type, unsigned char>)
275 {
276 *this << static_cast<int> (arg);
277 }
278 else
279 {
280 append_value (arg);
281 }
282 }
283 ++count;
284 }
285 if (count > max_range_elements_)
286 {
287 *this << ", ... (";
288 detail::append_number (buffer_, count);
289 *this << " elements)";
290 }
291 *this << "}";
292 return *this;
293 }

368 template <class Rep, class Period>
369 requires std::is_arithmetic_v<Rep>
370 expression_formatter&
371 expression_formatter::operator<< (
372 const std::chrono::duration<Rep, Period>& d)
373 {
374 detail::append_number (buffer_, d.count ());
375
376 if constexpr (std::is_same_v<Period, std::nano>)
377 {
378 return *this << "ns";
379 }
380 else if constexpr (std::is_same_v<Period, std::micro>)
381 {
382 return *this << "us";
383 }
384 else if constexpr (std::is_same_v<Period, std::milli>)
385 {
386 return *this << "ms";
387 }
388 else if constexpr (std::is_same_v<Period, std::ratio<1>>)
389 {
390 return *this << "s";
391 }
392 else if constexpr (std::is_same_v<Period, std::ratio<60>>)
393 {
394 return *this << "min";
395 }
396 else if constexpr (std::is_same_v<Period, std::ratio<3600>>)
397 {
398 return *this << "h";
399 }
400 else
401 {
402 *this << "[";
403 detail::append_number (buffer_, static_cast<long long> (Period::num));
404 *this << "/";
405 detail::append_number (buffer_, static_cast<long long> (Period::den));
406 return *this << "]s";
407 }

516 template <class Lhs_T, class Rhs_T, class Compare_T>
517 expression_formatter&
518 expression_formatter::operator<< (
519 const detail::compare_op<Lhs_T, Rhs_T, Compare_T>& op)
520 {
521 *this << colour (op);
522 append_operand (op.lhs ());
523 // A tag without a symbol is rejected when the expression is built, and
524 // must not give a second error here.
525 if constexpr (detail::tag_has_symbol<Compare_T>)
526 {
527 *this << Compare_T::symbol;
528 }
529 append_operand (op.rhs ());
530 using lhs_type = std::remove_cvref_t<decltype (op.lhs ())>;
531 using rhs_type = std::remove_cvref_t<decltype (op.rhs ())>;
532 if constexpr (std::is_same_v<Compare_T, detail::equal_to>
533 and detail::formattable_range<lhs_type>
534 and detail::formattable_range<rhs_type>
535 and requires (const lhs_type& l, const rhs_type& r) {
536 {
537 *std::begin (l) == *std::begin (r)
538 } -> std::convertible_to<bool>;
539 })
540 {
541 append_first_difference (op.lhs (), op.rhs ());
542 }
543 return (*this << colour_none ());
544 }
545
553 template <class Lhs_T, class Rhs_T>
554 void
555 expression_formatter::append_first_difference (const Lhs_T& lhs,
556 const Rhs_T& rhs)
557 {
558 auto l = std::begin (lhs);
559 const auto l_end = std::end (lhs);
560 auto r = std::begin (rhs);
561 const auto r_end = std::end (rhs);
562
563 size_t index = 0;
564 while (l != l_end and r != r_end and detail::equal_to::apply (*l, *r))
565 {
566 l = std::next (l);
567 r = std::next (r);
568 ++index;
569 }
570
571 if (index < max_range_elements_ or (l == l_end and r == r_end))
572 {
573 return;
574 }
575
576 *this << " (first difference at index ";
577 detail::append_number (buffer_, index);
578 *this << ")";
579 }

809 inline void
810 expression_formatter::append_throw_reason (const detail::callable_op& op)
811 {
812 if (static_cast<bool> (op))
813 {
814 return;
815 }
816
817 if (op.outcome () == detail::throw_outcome::std_exception)
818 {
819 buffer_.append (", but a std::exception was thrown: ");
820 *this << std::string_view{ op.what () };
821 }
822 else if (op.outcome () == detail::throw_outcome::unknown_exception)
823 {
824 buffer_.append (", but an exception of unknown type was thrown");
825 }
826 else
827 {
828 buffer_.append (", but nothing was thrown");
829 }
830 }
