189 template <class T>
190 requires detail::formattable_range<T>
191 expression_formatter&
192 expression_formatter::operator<< (const T& t)
193 {
194 *this << "{";
195 auto first = true;
196 for (const auto& arg : t)
197 {
198 *this << (first ? "" : ", ") << arg;
199 first = false;
200 }
201 *this << "}";
202 return *this;
203 }

272 template <class Rep, class Period>
273 expression_formatter&
274 expression_formatter::operator<< (
275 const std::chrono::duration<Rep, Period>& d)
276 {
277 detail::append_number (buffer_, d.count ());
278
279 if constexpr (std::is_same_v<Period, std::nano>)
280 {
281 return *this << "ns";
282 }
283 else if constexpr (std::is_same_v<Period, std::micro>)
284 {
285 return *this << "us";
286 }
287 else if constexpr (std::is_same_v<Period, std::milli>)
288 {
289 return *this << "ms";
290 }
291 else if constexpr (std::is_same_v<Period, std::ratio<1>>)
292 {
293 return *this << "s";
294 }
295 else if constexpr (std::is_same_v<Period, std::ratio<60>>)
296 {
297 return *this << "min";
298 }
299 else if constexpr (std::is_same_v<Period, std::ratio<3600>>)
300 {
301 return *this << "h";
302 }
303 else
304 {
305 return *this << "[" << static_cast<long long> (Period::num) << "/"
306 << static_cast<long long> (Period::den) << "]s";
307 }
308 }

475 template <class Exception_T>

477 expression_formatter::operator<< (const detail::throws_op<Exception_T>& op)
478 {
479 if constexpr (std::is_void_v<Exception_T>)
480 {
481 return (*this << colour (op) << "throws" << colours_.none);
