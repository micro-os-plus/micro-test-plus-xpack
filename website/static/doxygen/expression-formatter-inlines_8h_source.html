174 template <typename T>
175 expression_formatter&
176 expression_formatter::operator<< (T* v)
177 {
178 if (v == nullptr)
179 {
180 buffer_.append ("0x0");
181 return *this;
182 }
183
184#if defined(__GNUC__)
185#pragma GCC diagnostic push
186#if defined(__clang__)
187#pragma clang diagnostic ignored "-Wunsafe-buffer-usage-in-libc-call"
188#endif // defined(__clang__)
189#endif // defined(__GNUC__)
190
191 char buff[20];
192 snprintf (buff, sizeof (buff), "%p", reinterpret_cast<void*> (v));
193 buffer_.append (buff);
194
195#if defined(__GNUC__)
196#pragma GCC diagnostic pop
197#endif // defined(__GNUC__)
198
199 return *this;
200 }

237 template <class T>
238 requires (type_traits::container_like<T> and not type_traits::has_npos<T>)
239 expression_formatter&
240 expression_formatter::operator<< (const T& t)
241 {
242 *this << '{';
243 auto first = true;
244 for (const auto& arg : t)
245 {
246 *this << (first ? "" : ", ") << arg;
247 first = false;
248 }
249 *this << '}';
250 return *this;
251 }
