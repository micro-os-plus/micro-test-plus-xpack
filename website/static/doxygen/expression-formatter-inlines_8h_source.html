231 template <class T>
232 requires detail::formattable_range<T>
233 expression_formatter&
234 expression_formatter::operator<< (const T& t)
235 {
236 *this << "{";
237 size_t count = 0;
238 for (const auto& arg : t)
239 {
240 if (count < max_range_elements_)
241 {
242 *this << (count == 0 ? "" : ", ");
243 using element_type = std::remove_cvref_t<decltype (arg)>;
244 if constexpr (std::is_same_v<element_type, signed char>
245 or std::is_same_v<element_type, unsigned char>)
246 {
247 *this << static_cast<int> (arg);
248 }
249 else
250 {
251 append_value (arg);
252 }
253 }
254 ++count;
255 }
256 if (count > max_range_elements_)
257 {
258 *this << ", ... (";
259 detail::append_number (buffer_, count);
260 *this << " elements)";
261 }
262 *this << "}";
263 return *this;
264 }

337 template <class Rep, class Period>
338 requires std::is_arithmetic_v<Rep>
339 expression_formatter&
340 expression_formatter::operator<< (
341 const std::chrono::duration<Rep, Period>& d)
342 {
343 detail::append_number (buffer_, d.count ());
344
345 if constexpr (std::is_same_v<Period, std::nano>)
346 {
347 return *this << "ns";
348 }
349 else if constexpr (std::is_same_v<Period, std::micro>)
350 {
351 return *this << "us";
352 }
353 else if constexpr (std::is_same_v<Period, std::milli>)
354 {
355 return *this << "ms";
356 }
357 else if constexpr (std::is_same_v<Period, std::ratio<1>>)
358 {
359 return *this << "s";
360 }
361 else if constexpr (std::is_same_v<Period, std::ratio<60>>)
362 {
363 return *this << "min";
364 }
365 else if constexpr (std::is_same_v<Period, std::ratio<3600>>)
366 {
367 return *this << "h";
368 }
369 else
370 {
371 return *this << "[" << static_cast<long long> (Period::num) << "/"
372 << static_cast<long long> (Period::den) << "]s";
373 }
374 }

450 template <class Lhs_T, class Rhs_T, class Compare_T>
451 expression_formatter&
452 expression_formatter::operator<< (
453 const detail::compare_op<Lhs_T, Rhs_T, Compare_T>& op)
454 {
455 *this << colour (op);
456 append_operand (op.lhs ());
457 *this << Compare_T::symbol;
458 append_operand (op.rhs ());
459 using lhs_type = std::remove_cvref_t<decltype (op.lhs ())>;
460 using rhs_type = std::remove_cvref_t<decltype (op.rhs ())>;
461 if constexpr (std::is_same_v<Compare_T, detail::equal_to>
462 and detail::formattable_range<lhs_type>
463 and detail::formattable_range<rhs_type>
464 and requires (const lhs_type& l, const rhs_type& r) {
465 {
466 *std::begin (l) == *std::begin (r)
467 } -> std::convertible_to<bool>;
468 })
469 {
470 append_first_difference (op.lhs (), op.rhs ());
471 }
472 return (*this << colour_none ());
473 }
474
480 template <class Lhs_T, class Rhs_T>
481 void
482 expression_formatter::append_first_difference (const Lhs_T& lhs,
483 const Rhs_T& rhs)
484 {
485 auto l = std::begin (lhs);
486 const auto l_end = std::end (lhs);
487 auto r = std::begin (rhs);
488 const auto r_end = std::end (rhs);
489
490 size_t index = 0;
491 while (l != l_end and r != r_end and *l == *r)
492 {
493 l = std::next (l);
494 r = std::next (r);
495 ++index;
496 }
497
498 if (index < max_range_elements_ or (l == l_end and r == r_end))
499 {
500 return;
501 }
502
503 *this << " (first difference at index ";
504 detail::append_number (buffer_, index);
505 *this << ")";
506 }

732 inline void
733 expression_formatter::append_throw_reason (const detail::callable_op& op)
734 {
735 if (static_cast<bool> (op))
736 {
737 return;
738 }
739
740 if (op.outcome () == detail::throw_outcome::std_exception)
741 {
742 buffer_.append (", but a std::exception was thrown: ");
743 *this << std::string_view{ op.what () };
744 }
745 else if (op.outcome () == detail::throw_outcome::unknown_exception)
746 {
747 buffer_.append (", but an exception of unknown type was thrown");
748 }
749 else
750 {
751 buffer_.append (", but nothing was thrown");
752 }
753 }
754#endif // defined(__cpp_exceptions)
