260 template <class T>
261 requires detail::formattable_range<T>
262 expression_formatter&
263 expression_formatter::operator<< (const T& t)
264 {
265 *this << "{";
266 size_t count = 0;
267 for (const auto& arg : t)
268 {
269 if (count < max_range_elements_)
270 {
271 *this << (count == 0 ? "" : ", ");
272 using element_type = std::remove_cvref_t<decltype (arg)>;
273 if constexpr (std::is_same_v<element_type, signed char>
274 or std::is_same_v<element_type, unsigned char>)
275 {
276 *this << static_cast<int> (arg);
277 }
278 else
279 {
280 append_value (arg);
281 }
282 }
283 ++count;
284 }
285 if (count > max_range_elements_)
286 {
287 *this << ", ... (";
288 detail::append_number (buffer_, count);
289 *this << " elements)";
290 }
291 *this << "}";
292 return *this;
293 }

367 requires std::is_arithmetic_v<Rep>
368 expression_formatter&
369 expression_formatter::operator<< (
370 const std::chrono::duration<Rep, Period>& d)
371 {
372 detail::append_number (buffer_, d.count ());
373
374 if constexpr (std::is_same_v<Period, std::nano>)
375 {
376 return *this << "ns";
377 }
378 else if constexpr (std::is_same_v<Period, std::micro>)
379 {
380 return *this << "us";
381 }
382 else if constexpr (std::is_same_v<Period, std::milli>)
383 {
384 return *this << "ms";
385 }
386 else if constexpr (std::is_same_v<Period, std::ratio<1>>)
387 {
388 return *this << "s";
389 }
390 else if constexpr (std::is_same_v<Period, std::ratio<60>>)
391 {
392 return *this << "min";
393 }
394 else if constexpr (std::is_same_v<Period, std::ratio<3600>>)
395 {
396 return *this << "h";
397 }
398 else
399 {
400 return *this << "[" << static_cast<long long> (Period::num) << "/"
401 << static_cast<long long> (Period::den) << "]s";

511 template <class Lhs_T, class Rhs_T, class Compare_T>
512 expression_formatter&
513 expression_formatter::operator<< (
514 const detail::compare_op<Lhs_T, Rhs_T, Compare_T>& op)
515 {
516 *this << colour (op);
517 append_operand (op.lhs ());
518 // A tag without a symbol is rejected when the expression is built, and
519 // must not give a second error here.
520 if constexpr (detail::tag_has_symbol<Compare_T>)
521 {
522 *this << Compare_T::symbol;
523 }
524 append_operand (op.rhs ());
525 using lhs_type = std::remove_cvref_t<decltype (op.lhs ())>;
526 using rhs_type = std::remove_cvref_t<decltype (op.rhs ())>;
527 if constexpr (std::is_same_v<Compare_T, detail::equal_to>
528 and detail::formattable_range<lhs_type>
529 and detail::formattable_range<rhs_type>
530 and requires (const lhs_type& l, const rhs_type& r) {
531 {
532 *std::begin (l) == *std::begin (r)
533 } -> std::convertible_to<bool>;
534 })
535 {
536 append_first_difference (op.lhs (), op.rhs ());
537 }
538 return (*this << colour_none ());
539 }
540
548 template <class Lhs_T, class Rhs_T>
549 void
550 expression_formatter::append_first_difference (const Lhs_T& lhs,
551 const Rhs_T& rhs)
552 {
553 auto l = std::begin (lhs);
554 const auto l_end = std::end (lhs);
555 auto r = std::begin (rhs);
556 const auto r_end = std::end (rhs);
557
558 size_t index = 0;
559 while (l != l_end and r != r_end and detail::equal_to::apply (*l, *r))
560 {
561 l = std::next (l);
562 r = std::next (r);
563 ++index;
564 }
565
566 if (index < max_range_elements_ or (l == l_end and r == r_end))
567 {
568 return;
569 }
570
571 *this << " (first difference at index ";
572 detail::append_number (buffer_, index);
573 *this << ")";
574 }

804 inline void
805 expression_formatter::append_throw_reason (const detail::callable_op& op)
806 {
807 if (static_cast<bool> (op))
808 {
809 return;
810 }
811
812 if (op.outcome () == detail::throw_outcome::std_exception)
813 {
814 buffer_.append (", but a std::exception was thrown: ");
815 *this << std::string_view{ op.what () };
816 }
817 else if (op.outcome () == detail::throw_outcome::unknown_exception)
818 {
819 buffer_.append (", but an exception of unknown type was thrown");
820 }
821 else
822 {
823 buffer_.append (", but nothing was thrown");
824 }
825 }
