165 template <class T>
166 requires detail::formattable_range<T>
167 expression_formatter&
168 expression_formatter::operator<< (const T& t)
169 {
170 *this << "{";
171 auto first = true;
172 for (const auto& arg : t)
173 {
174 *this << (first ? "" : ", ") << arg;
175 first = false;
176 }
177 *this << "}";
178 return *this;
179 }

248 template <class Rep, class Period>
249 expression_formatter&
250 expression_formatter::operator<< (
251 const std::chrono::duration<Rep, Period>& d)
252 {
253 detail::append_number (buffer_, d.count ());
254
255 if constexpr (std::is_same_v<Period, std::nano>)
256 {
257 return *this << "ns";
258 }
259 else if constexpr (std::is_same_v<Period, std::micro>)
260 {
261 return *this << "us";
262 }
263 else if constexpr (std::is_same_v<Period, std::milli>)
264 {
265 return *this << "ms";
266 }
267 else if constexpr (std::is_same_v<Period, std::ratio<1>>)
268 {
269 return *this << "s";
270 }
271 else if constexpr (std::is_same_v<Period, std::ratio<60>>)
272 {
273 return *this << "min";
274 }
275 else if constexpr (std::is_same_v<Period, std::ratio<3600>>)
276 {
277 return *this << "h";
278 }
279 else
280 {
281 return *this << "[" << static_cast<long long> (Period::num) << "/"
282 << static_cast<long long> (Period::den) << "]s";
283 }
284 }
