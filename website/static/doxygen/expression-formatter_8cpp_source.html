98 void
99 append_hex (std::string& buffer, char32_t code, std::string_view escape,
100 std::size_t count)
101 {
102 constexpr std::string_view digits{ "0123456789abcdef" };
103 buffer.append (escape);
104 for (std::size_t i = count; i > 0; --i)
105 {
106 // The promoted type of `code` is `std::size_t` on some
107 // platforms, so a cast would be reported as useless.
108 const auto nibble = (code >> ((i - 1) * 4)) & 0x0Fu;
109 buffer.push_back (digits[nibble]);
110 }
111 }

129 void
130 append_quoted (std::string& buffer, std::string_view sv)
131 {
132 buffer.push_back ('"');
133 for (const char c : sv)
134 {
135 switch (c)
136 {
137 case '\0':
138 buffer.append ("\\0");
139 break;
140 case '\a':
141 buffer.append ("\\a");
142 break;
143 case '\b':
144 buffer.append ("\\b");
145 break;
146 case '\t':
147 buffer.append ("\\t");
148 break;
149 case '\n':
150 buffer.append ("\\n");
151 break;
152 case '\v':
153 buffer.append ("\\v");
154 break;
155 case '\f':
156 buffer.append ("\\f");
157 break;
158 case '\r':
159 buffer.append ("\\r");
160 break;
161 case '"':
162 buffer.append ("\\\"");
163 break;
164 case '\\':
165 buffer.append ("\\\\");
166 break;
167 default:
168 if (static_cast<unsigned char> (c) < 0x20 || c == 0x7F)
169 {
170 append_hex (buffer, static_cast<unsigned char> (c), "\\x",
171 2);
172 }
173 else
174 {
175 buffer.push_back (c);
176 }
177 break;
178 }
179 }
180 buffer.push_back ('"');
181 }

193 void
194 append_utf8 (std::string& text, char32_t code)
195 {
196 if ((code >= 0xD800 && code <= 0xDFFF) || code > 0x10FFFF)
197 {
198 code = 0xFFFD;
199 }
200
201 if (code < 0x80)
202 {
203 text.push_back (static_cast<char> (code));
204 }
205 else if (code < 0x800)
206 {
207 text.push_back (static_cast<char> (0xC0 | (code >> 6)));
208 text.push_back (static_cast<char> (0x80 | (code & 0x3F)));
209 }
210 else if (code < 0x10000)
211 {
212 text.push_back (static_cast<char> (0xE0 | (code >> 12)));
213 text.push_back (static_cast<char> (0x80 | ((code >> 6) & 0x3F)));
214 text.push_back (static_cast<char> (0x80 | (code & 0x3F)));
215 }
216 else
217 {
218 text.push_back (static_cast<char> (0xF0 | (code >> 18)));
219 text.push_back (static_cast<char> (0x80 | ((code >> 12) & 0x3F)));
220 text.push_back (static_cast<char> (0x80 | ((code >> 6) & 0x3F)));
221 text.push_back (static_cast<char> (0x80 | (code & 0x3F)));
222 }
223 }

238 template <class Char_T>
239 std::string
240 utf16_to_utf8 (std::basic_string_view<Char_T> sv)
241 {
242 std::string text;
243 text.reserve (sv.size ());
244 for (std::size_t i = 0; i < sv.size (); ++i)
245 {
246 const auto unit = static_cast<char32_t> (sv[i]) & 0xFFFFu;
247 if (unit >= 0xD800 && unit <= 0xDBFF && i + 1 < sv.size ())
248 {
249 const auto next = static_cast<char32_t> (sv[i + 1]) & 0xFFFFu;
250 if (next >= 0xDC00 && next <= 0xDFFF)
251 {
252 append_utf8 (text, 0x10000 + ((unit - 0xD800) << 10)
253 + (next - 0xDC00));
254 ++i;
255 continue;
256 }
257 }
258 append_utf8 (text, unit);
259 }
260 return text;
261 }

933 void
934 expression_formatter::append_decimal_point (std::size_t start)
935 {
936 if (plain_)
937 {
938 return;
939 }
940 if (start < buffer_.size ()
941 && (buffer_[start] == '-' || buffer_[start] == '+'))
942 {
943 ++start;
944 }
945 if (start == buffer_.size ())
946 {
947 return;
948 }
949 for (auto i = start; i < buffer_.size (); ++i)
950 {
951 if (buffer_[i] < '0' || buffer_[i] > '9')
952 {
953 return;
954 }
955 }
956 buffer_.append (".0");
957 }

999 void
1000 expression_formatter::append_character (std::string_view prefix,
1001 char32_t code, std::size_t size)
1002 {
1003 buffer_.append (prefix);
1004 buffer_.push_back ('\'');
1005 switch (code)
1006 {
1007 case U'\0':
1008 buffer_.append ("\\0");
1009 break;
1010 case U'\a':
1011 buffer_.append ("\\a");
1012 break;
1013 case U'\b':
1014 buffer_.append ("\\b");
1015 break;
1016 case U'\t':
1017 buffer_.append ("\\t");
1018 break;
1019 case U'\n':
1020 buffer_.append ("\\n");
1021 break;
1022 case U'\v':
1023 buffer_.append ("\\v");
1024 break;
1025 case U'\f':
1026 buffer_.append ("\\f");
1027 break;
1028 case U'\r':
1029 buffer_.append ("\\r");
1030 break;
1031 case U'\'':
1032 buffer_.append ("\\'");
1033 break;
1034 case U'\\':
1035 buffer_.append ("\\\\");
1036 break;
1037 default:
1038 if (code < 0x20 || code == 0x7F)
1039 {
1040 append_hex (buffer_, code, "\\x", 2);
1041 }
1042 else if (code < 0x7F)
1043 {
1044 buffer_.push_back (static_cast<char> (code));
1045 }
1046 else if (size == 1 || (code >= 0xD800 && code <= 0xDFFF)
1047 || code > 0x10FFFF)
1048 {
1049 append_hex (buffer_, code, "\\x", size * 2);
1050 }
1051 else if (code <= 0xFFFF)
1052 {
1053 append_hex (buffer_, code, "\\u", 4);
1054 }
1055 else
1056 {
1057 append_hex (buffer_, code, "\\U", 8);
1058 }
1059 break;
1060 }
1061 buffer_.push_back ('\'');
1062 }

1117 void
1118 expression_formatter::end_check (const void* owner) noexcept
1119 {
1120 if (owner == owner_)
1121 {
1122 owner_ = nullptr;
1123 return;
1124 }
1125
1126 // Use indices and `swap()` + `pop_back()` (both returning void) rather
1127 // than iterators and `erase()`, which returns an iterator object and
1128 // triggers -Waggregate-return; the order of the entries is irrelevant.
1129 for (size_t i = 0; i < pending_.size (); ++i)
1130 {
1131 if (pending_[i].first == owner)
1132 {
1133 if (i + 1 != pending_.size ())
1134 {
1135 pending_[i].swap (pending_.back ());
1136 }
1137 pending_.pop_back ();
1138 return;
1139 }
