97 void
98 append_hex (std::string& buffer, char32_t code, std::string_view escape,
99 std::size_t count)
100 {
101 constexpr std::string_view digits{ "0123456789abcdef" };
102 buffer.append (escape);
103 for (std::size_t i = count; i > 0; --i)
104 {
105 // The promoted type of `code` is `std::size_t` on some
106 // platforms, so a cast would be reported as useless.
107 const auto nibble = (code >> ((i - 1) * 4)) & 0x0Fu;
108 buffer.push_back (digits[nibble]);
109 }
110 }

128 void
129 append_quoted (std::string& buffer, std::string_view sv)
130 {
131 buffer.push_back ('"');
132 for (const char c : sv)
133 {
134 switch (c)
135 {
136 case '\0':
137 buffer.append ("\\0");
138 break;
139 case '\a':
140 buffer.append ("\\a");
141 break;
142 case '\b':
143 buffer.append ("\\b");
144 break;
145 case '\t':
146 buffer.append ("\\t");
147 break;
148 case '\n':
149 buffer.append ("\\n");
150 break;
151 case '\v':
152 buffer.append ("\\v");
153 break;
154 case '\f':
155 buffer.append ("\\f");
156 break;
157 case '\r':
158 buffer.append ("\\r");
159 break;
160 case '"':
161 buffer.append ("\\\"");
162 break;
163 case '\\':
164 buffer.append ("\\\\");
165 break;
166 default:
167 if (static_cast<unsigned char> (c) < 0x20 || c == 0x7F)
168 {
169 append_hex (buffer, static_cast<unsigned char> (c), "\\x",
170 2);
171 }
172 else
173 {
174 buffer.push_back (c);
175 }
176 break;
177 }
178 }
179 buffer.push_back ('"');
180 }

573 void
574 expression_formatter::append_decimal_point (std::size_t start)
575 {
576 if (plain_)
577 {
578 return;
579 }
580 if (start < buffer_.size ()
581 && (buffer_[start] == '-' || buffer_[start] == '+'))
582 {
583 ++start;
584 }
585 if (start == buffer_.size ())
586 {
587 return;
588 }
589 for (auto i = start; i < buffer_.size (); ++i)
590 {
591 if (buffer_[i] < '0' || buffer_[i] > '9')
592 {
593 return;
594 }
595 }
596 buffer_.append (".0");
597 }

624 void
625 expression_formatter::append_character (std::string_view prefix,
626 char32_t code, std::size_t size)
627 {
628 buffer_.append (prefix);
629 buffer_.push_back ('\'');
630 switch (code)
631 {
632 case U'\0':
633 buffer_.append ("\\0");
634 break;
635 case U'\a':
636 buffer_.append ("\\a");
637 break;
638 case U'\b':
639 buffer_.append ("\\b");
640 break;
641 case U'\t':
642 buffer_.append ("\\t");
643 break;
644 case U'\n':
645 buffer_.append ("\\n");
646 break;
647 case U'\v':
648 buffer_.append ("\\v");
649 break;
650 case U'\f':
651 buffer_.append ("\\f");
652 break;
653 case U'\r':
654 buffer_.append ("\\r");
655 break;
656 case U'\'':
657 buffer_.append ("\\'");
658 break;
659 case U'\\':
660 buffer_.append ("\\\\");
661 break;
662 default:
663 if (code < 0x20 || code == 0x7F)
664 {
665 append_hex (buffer_, code, "\\x", 2);
666 }
667 else if (code < 0x7F)
668 {
669 buffer_.push_back (static_cast<char> (code));
670 }
671 else if (size == 1 || (code >= 0xD800 && code <= 0xDFFF)
672 || code > 0x10FFFF)
673 {
674 append_hex (buffer_, code, "\\x", size * 2);
675 }
676 else if (code <= 0xFFFF)
677 {
678 append_hex (buffer_, code, "\\u", 4);
679 }
680 else
681 {
682 append_hex (buffer_, code, "\\U", 8);
683 }
684 break;
685 }
686 buffer_.push_back ('\'');
687 }

716 const std::string&
717 expression_formatter::check_text (const void* owner) const noexcept
718 {
719 if (owner == owner_)
720 {
721 return buffer_;
722 }
723
724 for (size_t i = 0; i < pending_.size (); ++i)
725 {
726 if (pending_[i].first == owner)
727 {
728 return pending_[i].second;
729 }
730 }
731
732 static const std::string empty{};
733 return empty;
734 }
735
742 void
743 expression_formatter::end_check (const void* owner) noexcept
744 {
745 if (owner == owner_)
746 {
747 owner_ = nullptr;
748 return;
749 }
750
751 // Use indices and `swap()` + `pop_back()` (both returning void) rather
752 // than iterators and `erase()`, which returns an iterator object and
753 // triggers -Waggregate-return; the order of the entries is irrelevant.
754 for (size_t i = 0; i < pending_.size (); ++i)
755 {
756 if (pending_[i].first == owner)
757 {
758 if (i + 1 != pending_.size ())
759 {
760 pending_[i].swap (pending_.back ());
761 }
762 pending_.pop_back ();
763 return;
764 }
765 }
766 }
