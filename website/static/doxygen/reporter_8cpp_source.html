112 void
113 write_without_colours (FILE* file, std::string_view text)
114 {
115 size_t start = 0;
116 size_t pos = text.find ('\033');
117 while (pos != std::string_view::npos)
118 {
119 // Parse `ESC [ <digits and semicolons> m`.
120 size_t end = pos + 1;
121 if (end < text.size () && text[end] == '[')
122 {
123 ++end;
124 while (end < text.size ()
125 && ((text[end] >= '0' && text[end] <= '9')
126 || text[end] == ';'))
127 {
128 ++end;
129 }
130 if (end < text.size () && text[end] == 'm')
131 {
132 // A colour sequence; write the preceding text and skip
133 // it.
134 const std::string_view chunk
135 = text.substr (start, pos - start);
136 fwrite (chunk.data (), 1, chunk.size (), file);
137 start = end + 1;
138 }
139 }
140 pos = text.find ('\033', pos + 1);
141 }
142
143 const std::string_view chunk = text.substr (start);
144 fwrite (chunk.data (), 1, chunk.size (), file);
145 }

250 output_route
251 reporter::route (record_kind kind) const noexcept
252 {
253 bool console;
254 if (kind == record_kind::failed_check || kind == record_kind::total
255 || kind == record_kind::abort)
256 {
257 console = (verbosity_ != verbosity::silent);
258 }
259 else if (kind == record_kind::test_case_heading
260 || kind == record_kind::passed_check)
261 {
262 console = (verbosity_ == verbosity::verbose);
263 }
264#if defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
265 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
266 else if (kind == record_kind::info)
267 {
268 // The startup code already displays this information.
269 console = false;
270 }
271#endif // defined(MICRO_OS_PLUS_STARTUP_ENABLED) && ...
272 else
273 {
274 console = (verbosity_ == verbosity::normal
275 || verbosity_ == verbosity::verbose);
276 }
277
278 return { console, output_file_ != nullptr };
279 }

427 void
428 reporter::write_line (bool to_console, const char* format, ...)
429 {
430 if (output_file_ == nullptr && !to_console)
431 {
432 return;
433 }
434
435 va_list args;
436 va_start (args, format);
437 va_list args_copy;
438 va_copy (args_copy, args);
439
440 char local_text[128];
441 const int length
442 = vsnprintf (local_text, sizeof (local_text), format, args);
443 va_end (args);
444
445 if (length < 0)
446 {
447 va_end (args_copy);
448 return;
449 }
450
451 const size_t size = static_cast<size_t> (length);
452 std::string long_text;
453 std::string_view text{ local_text, size };
454 if (size >= sizeof (local_text))
455 {
456 // The terminating null character is written over the one kept by
457 // the string after its content.
458 long_text.resize (size);
459 vsnprintf (long_text.data (), size + 1, format, args_copy);
460 text = long_text;
461 }
462 va_end (args_copy);
463
464 if (output_file_ != nullptr)
465 {
466 write_without_colours (output_file_, text);
467 }
468
469 if (to_console)
470 {
471 fwrite (text.data (), 1, text.size (), stdout);
472 }
473 }

489 void
490 reporter::write_info (void)
491 {
492 const bool to_console = route (record_kind::info).console;
493
494 if (!arguments_.empty ())
495 {
496 std::string line;
497 line.reserve (256);
498 line.append (comment_prefix ());
499 line.append ("Running: ");
500
501 // Append only the file name part of argv[0].
502 line.append (utility::extract_file_name (arguments_[0]));
503
504 for (const char* argument : arguments_.subspan (1))
505 {
506 line.append (" ");
507 line.append (argument);
508 }
509 line.append ("\n");
510
511 write_line (to_console, "%s", line.c_str ());
512 }
513
514 {
515 // Build the "Built with ..." line, including the compiler version
516 // (via __VERSION__); the same line goes to the file and to stdout.
517 std::string line;
518 line.reserve (256);
519 line.append (comment_prefix ());
520 line.append ("Built with ");
521#if defined(__clang__)
522 line.append ("clang " __VERSION__);
523#elif defined(__GNUC__)
524 line.append ("GCC " __VERSION__);
525#elif defined(_MSC_VER)
526 line.append ("MSVC");
527 char msvc_ver[16];
528 snprintf (msvc_ver, sizeof (msvc_ver), " - %d", _MSC_VER);
529 line.append (msvc_ver);
530#else
531 line.append ("an unknown compiler");
532#endif // defined(__clang__)
533#if !(defined(__APPLE__) || defined(__linux__) || defined(__unix__) \
534 || defined(_WIN32))
535 // This is relevant only on bare-metal.
536#if defined(__ARM_PCS_VFP) || defined(__ARM_FP)
537 line.append (", with FP");
538#else
539 line.append (", no FP");
540#endif // defined(__ARM_PCS_VFP) || defined(__ARM_FP)
541#endif // bare-metal only
542#if defined(__cpp_exceptions)
543 line.append (", with exceptions");
544#else
545 line.append (", no exceptions");
546#endif // defined(__cpp_exceptions)
547#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)
548 line.append (", with MICRO_OS_PLUS_DEBUG_ENABLED");
549#endif // defined(MICRO_OS_PLUS_DEBUG_ENABLED)
550#if defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
551 line.append (", with MICRO_OS_PLUS_DIAG_TRACE_ENABLED");
552#endif // defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)

838 void
839 reporter::format_session_totals (const runner& runner,
840 char (&message)[totals_size_])
841 {
842 // The fixed text, four counts, two fragments, and the terminator.
843 static_assert (totals_size_
844 >= 54 + 4 * (std::numeric_limits<size_t>::digits10 + 1)
845 + 2 * (fragment_size - 1),
846 "totals_size_ too small for the session totals");
847
848 const counters& totals = runner.totals ();
849 const size_t suites_count = runner.total_suites_count ();
850
851 char message_skipped_test_cases[fragment_size];
852 format_skipped (totals.skipped_test_cases (), message_skipped_test_cases);
853
854 char message_skipped_suites[fragment_size];
855 format_skipped (runner.skipped_suites_count (), message_skipped_suites);
856
857 snprintf (message, sizeof (message),
858 "%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
859 " check%s passed, %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
860 " failed, in %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
861 " test case%s%s, %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE " "
862 "test suite%s%s",
863 totals.passed_checks (), plural (totals.passed_checks ()),
864 totals.failed_checks (), totals.executed_test_cases (),
865 plural (totals.executed_test_cases ()),
866 message_skipped_test_cases, suites_count, plural (suites_count),
867 message_skipped_suites);
868 }
