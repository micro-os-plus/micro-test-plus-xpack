83 reporter::reporter (std::unique_ptr<std::vector<std::string_view>> argvs)
84 {
85#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
86 trace::printf ("%s\n", __PRETTY_FUNCTION__);
87#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
88
89 std::string_view output_file_sv{};
90
91 argvs_ = std::move (argvs);
92
93 static constexpr std::string_view output_file_prefix{ "--output-file=" };
94 if (argvs_)
95 {
96 const auto& args = *argvs_;
97 for (size_t i = 0; i < args.size (); ++i)
98 {
99 if (args[i] == "--verbose")
100 {
101 verbosity_ = verbosity::verbose;
102 }
103 else if (args[i] == "--quiet")
104 {
105 verbosity_ = verbosity::quiet;
106 }
107 else if (args[i] == "--silent")
108 {
109 verbosity_ = verbosity::silent;
110 }
111 else if (args[i].starts_with (output_file_prefix))
112 {
113 output_file_sv = args[i].substr (output_file_prefix.size ());
114 }
115 else if (args[i]
116 == output_file_prefix.substr (
117 0, output_file_prefix.size () - 1))
118 {
119 if (i + 1 < args.size ())
120 {
121 output_file_sv = args[++i];
122 }
123 else
124 {
125 fprintf (stderr, "error: --output-file option requires a "
126 "file path argument\n");
127 exit (1);
128 }
129 }
130 }
131 }
132
133 if (!output_file_sv.empty ())
134 {
135 // .data() is safe: all string_views are views into argv[]
136 // entries, which are null-terminated C strings.
137 output_file_ = fopen (output_file_sv.data (), "w");
138 if (output_file_ == nullptr)
139 {
140 fprintf (stderr, "error: Failed to open output file '%.*s'\n",
141 static_cast<int> (output_file_sv.size ()),
142 output_file_sv.data ());
143 exit (1);
144 }
145 // The original string is zero terminated, so we can safely use .data()
146 // here.
147 output_file_path_ = output_file_sv.data ();
148 }
149
150 // Pre-allocate buffer to reduce dynamic allocations.
151 buffer_.reserve (128);
152 }

161 reporter::~reporter ()
162 {
163#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
164 trace::printf ("%s\n", __PRETTY_FUNCTION__);
165#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
166
167 if (output_file_ != nullptr)
168 {
169 fflush (output_file_);
170 fclose (output_file_);
171
172 write_console_ ("Test output written to '%s'.\n", output_file_path_);
173

175 output_file_path_ = nullptr;

231 void
232 reporter::write_buffer_to_file_ (void)
233 {
234 if (output_file_ == nullptr)
235 {
236 return;
237 }
238
239 // Pass only the string, do not add an `\n` here.
240 const std::string_view text{ buffer_ };
241 size_t start = 0;
242 size_t pos = text.find ('\033');
243 while (pos != std::string_view::npos)
244 {
245 // Parse `ESC [ <digits and semicolons> m`.
246 size_t end = pos + 1;
247 if (end < text.size () && text[end] == '[')
248 {
249 ++end;
250 while (end < text.size ()
251 && ((text[end] >= '0' && text[end] <= '9')
252 || text[end] == ';'))
253 {
254 ++end;
255 }
256 if (end < text.size () && text[end] == 'm')
257 {
258 // A colour sequence; write the preceding text and skip it.
259 const std::string_view chunk
260 = text.substr (start, pos - start);
261 fwrite (chunk.data (), 1, chunk.size (), output_file_);
262 start = end + 1;
263 }
264 }
265 pos = text.find ('\033', pos + 1);
266 }
267
268 const std::string_view chunk = text.substr (start);
269 fwrite (chunk.data (), 1, chunk.size (), output_file_);
270 }

320 void
321 reporter::write_info_ (void)
322 {
323 if (argvs_ && !argvs_->empty ())
324 {
325 const auto& args = *argvs_;
326 std::string line;
327 line.reserve (256);
328 line.append (get_comment_prefix ());
329 line.append ("Running: ");
330
331 // Append only the file name part of argv[0].
332 const std::string_view arg0 = args[0];
333 const auto sep = arg0.rfind ('/');
334 line.append ((sep != std::string_view::npos) ? arg0.substr (sep + 1)
335 : arg0);
336
337 for (size_t i = 1; i < args.size (); ++i)
338 {
339 line.append (" ");
340 line.append (args[i]);
341 }
342 line.append ("\n");
343
344 write_file_ ("%s", line.c_str ());
345
346#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
347 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
348 if (verbosity_ == verbosity::normal
349 || verbosity_ == verbosity::verbose)
350 write_console_ ("%s", line.c_str ());
351#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
352 }
353
354 {
355 // Build the "Built with ..." line, including the compiler version
356 // (via __VERSION__); the same line goes to the file and to stdout.
357 std::string line;
358 line.reserve (256);
359 line.append (get_comment_prefix ());
360 line.append ("Built with ");
361#if defined(__clang__)
362 line.append ("clang " __VERSION__);
363#elif defined(__GNUC__)
364 line.append ("GCC " __VERSION__);
365#elif defined(_MSC_VER)
366 line.append ("MSVC");
367 char msvc_ver[16];
368 snprintf (msvc_ver, sizeof (msvc_ver), " - %d", _MSC_VER);
369 line.append (msvc_ver);
370#else
371 line.append ("an unknown compiler");
372#endif // defined(__clang__)
373#if !(defined(__APPLE__) || defined(__linux__) || defined(__unix__) \
374 || defined(WIN32))
375 // This is relevant only on bare-metal.
376#if defined(__ARM_PCS_VFP) || defined(__ARM_FP)
377 line.append (", with FP");
378#else
379 line.append (", no FP");
380#endif // defined(__ARM_PCS_VFP) || defined(__ARM_FP)
381#endif // !(defined(__APPLE__) || defined(__linux__) || defined(__unix__) || defined(WIN32))
382#if defined(__EXCEPTIONS)
383 line.append (", with exceptions");
384#else
385 line.append (", no exceptions");
386#endif // defined(__EXCEPTIONS)
387#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)
388 line.append (", with MICRO_OS_PLUS_DEBUG_ENABLED");
389#endif // defined(MICRO_OS_PLUS_DEBUG_ENABLED)
390#if defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
391 line.append (", with MICRO_OS_PLUS_DIAG_TRACE_ENABLED");
392#endif // defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
393
394 write_file_ ("%s\n", line.c_str ());
395
396#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
397 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
398 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
399 {
400 write_console_ ("%s\n", line.c_str ());
401 }
402#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
403 }
