78 reporter::reporter (std::unique_ptr<std::vector<std::string_view>> argvs)
79 {
80#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
81 trace::printf ("%s\n", __PRETTY_FUNCTION__);
82#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
83
84 std::string_view output_file_sv{};
85
86 argvs_ = std::move (argvs);
87
88 static constexpr std::string_view output_file_prefix{ "--output-file=" };
89 if (argvs_)
90 {
91 const auto& args = *argvs_;
92 for (size_t i = 0; i < args.size (); ++i)
93 {
94 if (args[i] == "--verbose")
95 {
96 verbosity_ = verbosity::verbose;
97 }
98 else if (args[i] == "--quiet")
99 {
100 verbosity_ = verbosity::quiet;
101 }
102 else if (args[i] == "--silent")
103 {
104 verbosity_ = verbosity::silent;
105 }
106 else if (args[i].starts_with (output_file_prefix))
107 {
108 output_file_sv = args[i].substr (output_file_prefix.size ());
109 }
110 else if (args[i]
111 == output_file_prefix.substr (
112 0, output_file_prefix.size () - 1))
113 {
114 if (i + 1 < args.size ())
115 {
116 output_file_sv = args[++i];
117 }
118 else
119 {
120 fprintf (stderr, "error: --output-file option requires a "
121 "file path argument\n");
122 exit (1);
123 }
124 }
125 }
126 }
127
128 if (!output_file_sv.empty ())
129 {
130 // .data() is safe: all string_views are views into argv[]
131 // entries, which are null-terminated C strings.
132 output_file_ = fopen (output_file_sv.data (), "w");
133 if (output_file_ == nullptr)
134 {
135 fprintf (stderr, "error: Failed to open output file '%.*s'\n",
136 static_cast<int> (output_file_sv.size ()),
137 output_file_sv.data ());
138 exit (1);
139 }
140 // The original string is zero terminated, so we can safely use .data()
141 // here.
142 output_file_path_ = output_file_sv.data ();
143 }
144
145 // Pre-allocate buffer to reduce dynamic allocations.
146 buffer_.reserve (128);
147 }

156 reporter::~reporter ()
157 {
158#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
159 trace::printf ("%s\n", __PRETTY_FUNCTION__);
160#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
161
162 if (output_file_ != nullptr)
163 {
164 fflush (output_file_);
165 fclose (output_file_);
166
167 printf ("Test output written to '%s'.\n", output_file_path_);
168
169 output_file_ = nullptr;
170 output_file_path_ = nullptr;
171 }
172 }

248 void
249 reporter::write_info_ (void)
250 {
251 if (argvs_ && !argvs_->empty ())
252 {
253 const auto& args = *argvs_;
254 std::string line;
255 line.reserve (256);
256 line.append (get_comment_prefix ());
257 line.append ("Running: ");
258
259 // Append only the file name part of argv[0].
260 const std::string_view arg0 = args[0];
261 const auto sep = arg0.rfind ('/');
262 line.append ((sep != std::string_view::npos) ? arg0.substr (sep + 1)
263 : arg0);
264
265 for (size_t i = 1; i < args.size (); ++i)
266 {
267 line.append (" ");
268 line.append (args[i]);
269 }
270 line.append ("\n");
271
272 if (output_file_ != nullptr)
273 fprintf (output_file_, "%s", line.c_str ());
274
275#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
276 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
277 if (verbosity_ == verbosity::normal
278 || verbosity_ == verbosity::verbose)
279 printf ("%s", line.c_str ());
280#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
281 }
282
283 {
284 // Build the "Built with ..." line. For the output file the compiler
285 // version is omitted; for stdout it is appended via __VERSION__.
286 std::string line;
287 line.reserve (256);
288 line.append (get_comment_prefix ());
289 line.append ("Built with ");
290#if defined(__clang__)
291 line.append ("clang " __VERSION__);
292#elif defined(__GNUC__)
293 line.append ("GCC " __VERSION__);
294#elif defined(_MSC_VER)
295 line.append ("MSVC");
296 char msvc_ver[16];
297 snprintf (msvc_ver, sizeof (msvc_ver), " - %d", _MSC_VER);
298 line.append (msvc_ver);
299#else
300 line.append ("an unknown compiler");
301#endif // defined(__clang__)
302#if !(defined(__APPLE__) || defined(__linux__) || defined(__unix__) \
303 || defined(WIN32))
304 // This is relevant only on bare-metal.
305#if defined(__ARM_PCS_VFP) || defined(__ARM_FP)
306 line.append (", with FP");
307#else
308 line.append (", no FP");
309#endif // defined(__ARM_PCS_VFP) || defined(__ARM_FP)
310#endif // !(defined(__APPLE__) || defined(__linux__) || defined(__unix__) || defined(WIN32))
311#if defined(__EXCEPTIONS)
312 line.append (", with exceptions");
313#else
314 line.append (", no exceptions");
315#endif // defined(__EXCEPTIONS)
316#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)
317 line.append (", with MICRO_OS_PLUS_DEBUG_ENABLED");
318#endif // defined(MICRO_OS_PLUS_DEBUG_ENABLED)
319#if defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
320 line.append (", with MICRO_OS_PLUS_DIAG_TRACE_ENABLED");
321#endif // defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED)
322
323 if (output_file_ != nullptr)
324 {
325 fprintf (output_file_, "%s\n", line.c_str ());
326 }
327
328#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
329 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
330 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
331 {
332 printf ("%s\n", line.c_str ());
333 }
334#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
335 }
