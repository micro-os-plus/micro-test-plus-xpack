173 template <typename Callable_T, typename... Args_T>
174 void
175 runner::add_suite (std::string_view name, Callable_T&& callable,
176 Args_T&&... arguments)
177 {
178#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
179 trace::printf ("%s '%.*s'\n", __PRETTY_FUNCTION__,
180 static_cast<int> (name.size ()), name.data ());
181#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
182
183 // Not std::make_unique(), which cannot access the protected constructor.
184 std::unique_ptr<class suite> child_suite{ new class suite (
185 name, *this, std::forward<Callable_T> (callable),
186 std::forward<Args_T> (arguments)...) };
187
188 register_suite (std::move (child_suite));
189 }
190
191#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_DEPRECATED_ENABLED)
196 // Deprecated since 5.1.0; use add_suite().
197 template <typename Callable_T, typename... Args_T>
198 void
199 runner::suite (std::string_view name, Callable_T&& callable,
200 Args_T&&... arguments)
201 {
202 add_suite (name, std::forward<Callable_T> (callable),
203 std::forward<Args_T> (arguments)...);
204 }
