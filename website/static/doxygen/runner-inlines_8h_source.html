185 template <typename Callable_T, typename... Args_T>
186 requires detail::suite_body<Callable_T, Args_T...>
187 void
188 runner::add_suite (located_name name, Callable_T&& callable,
189 Args_T&&... arguments)
190 {
191#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
192 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
193#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
194
195 // Not std::make_unique(), which cannot access the private constructor.
196 std::unique_ptr<class suite> child_suite{ new class suite (
197 std::move (name), *this, std::forward<Callable_T> (callable),
198 std::forward<Args_T> (arguments)...) };
199
200 register_suite (std::move (child_suite));
201 }
202
203#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_DEPRECATED_ENABLED)

210 void
211 runner::suite (std::string_view name, Callable_T&& callable,
212 Args_T&&... arguments)
