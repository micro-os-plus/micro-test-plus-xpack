155 template <typename Callable_T, typename... Args_T>
156 void
157 runner::add_suite (std::string_view name, Callable_T&& callable,
158 Args_T&&... arguments)
159 {
160#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
161 trace::printf ("%s '%.*s'\n", __PRETTY_FUNCTION__,
162 static_cast<int> (name.size ()), name.data ());
163#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
164
165 // Not std::make_unique(), which cannot access the protected constructor.
166 std::unique_ptr<class suite> child_suite{ new class suite (
167 name, *this, std::forward<Callable_T> (callable),
168 std::forward<Args_T> (arguments)...) };
169
170 register_suite (std::move (child_suite));
171 }
172
177 template <typename Callable_T, typename... Args_T>
178 void
179 runner::suite (std::string_view name, Callable_T&& callable,
180 Args_T&&... arguments)
181 {
182 add_suite (name, std::forward<Callable_T> (callable),
183 std::forward<Args_T> (arguments)...);
184 }
