188 template <typename Callable_T, typename... Args_T>
189 requires detail::suite_body<Callable_T, Args_T...>
190 void
191 runner::add_suite (located_name name, Callable_T&& callable,
192 Args_T&&... arguments)
193 {
194#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
195 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name.name ());
196#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
197
198 // Not std::make_unique(), which cannot access the private constructor.
199 std::unique_ptr<class suite> child_suite{ new class suite (
200 std::move (name), *this, std::forward<Callable_T> (callable),
201 std::forward<Args_T> (arguments)...) };
202
203 register_suite (std::move (child_suite));
204 }
