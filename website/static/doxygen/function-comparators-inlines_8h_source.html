355 template <class First_T, class Second_T, class... Rest_T>
356 requires (extension::checkable<std::remove_cvref_t<First_T>>
357 and extension::checkable<std::remove_cvref_t<Second_T>>
358 and (extension::checkable<std::remove_cvref_t<Rest_T>> and ...))
359 constexpr auto
360 all_of (First_T&& first, Second_T&& second, Rest_T&&... rest)
361 {
362 if constexpr (sizeof...(Rest_T) == 0)
363 {
364 return detail::and_op{ std::forward<First_T> (first),
365 std::forward<Second_T> (second) };
366 }
367 else
368 {
369 return micro_test_plus::all_of (
370 detail::and_op{ std::forward<First_T> (first),
371 std::forward<Second_T> (second) },
372 std::forward<Rest_T> (rest)...);
373 }
374 }
