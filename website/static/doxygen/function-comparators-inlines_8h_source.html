246 all_of (First_T&& first, Second_T&& second, Rest_T&&... rest)
247 {
248 if constexpr (sizeof...(Rest_T) == 0)
249 {
250 return detail::and_op{ std::forward<First_T> (first),
251 std::forward<Second_T> (second) };
