260 template <class First_T, class Second_T, class... Rest_T>
261 requires (extension::checkable<std::remove_cvref_t<First_T>>
262 and extension::checkable<std::remove_cvref_t<Second_T>>
263 and (extension::checkable<std::remove_cvref_t<Rest_T>> and ...))
264 constexpr auto
265 all_of (First_T&& first, Second_T&& second, Rest_T&&... rest)
266 {
267 if constexpr (sizeof...(Rest_T) == 0)
268 {
269 return detail::and_op{ std::forward<First_T> (first),
270 std::forward<Second_T> (second) };
271 }
272 else
273 {
274 return micro_test_plus::all_of (
275 detail::and_op{ std::forward<First_T> (first),
276 std::forward<Second_T> (second) },
277 std::forward<Rest_T> (rest)...);
278 }
279 }
