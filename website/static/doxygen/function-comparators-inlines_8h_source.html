349 template <class First_T, class Second_T, class... Rest_T>
350 requires (extension::checkable<std::remove_cvref_t<First_T>>
351 and extension::checkable<std::remove_cvref_t<Second_T>>
352 and (extension::checkable<std::remove_cvref_t<Rest_T>> and ...))
353 constexpr auto
354 all_of (First_T&& first, Second_T&& second, Rest_T&&... rest)
355 {
356 if constexpr (sizeof...(Rest_T) == 0)
357 {
358 return detail::and_op{ std::forward<First_T> (first),
359 std::forward<Second_T> (second) };
360 }
361 else
362 {
363 return micro_test_plus::all_of (
364 detail::and_op{ std::forward<First_T> (first),
365 std::forward<Second_T> (second) },
366 std::forward<Rest_T> (rest)...);
367 }
368 }
