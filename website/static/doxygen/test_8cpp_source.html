79 void
80 escape_control_characters (std::string& name)
81 {
82 const auto is_control = [] (char c)
83 {
84 const auto unit = static_cast<unsigned char> (c);
85 return (unit < 0x20 && c != '\t') || unit == 0x7F;
86 };
87
88 std::size_t first = 0;
89 while (first < name.size () && !is_control (name[first]))
90 {
91 ++first;
92 }
93 if (first == name.size ())
94 {
95 return;
96 }
97
98 std::string escaped (name, 0, first);
99 escaped.reserve (name.size () + 8);
100 for (std::size_t i = first; i < name.size (); ++i)
101 {
102 if (is_control (name[i]))
103 {
104 append_control_escape (escaped, name[i]);
105 }
106 else
107 {
108 escaped.push_back (name[i]);
109 }
110 }
111 name = std::move (escaped);
112 }

220 bool
221 runnable_base::accepts_test_case (const suite& target, const char* name)
222 {
223 if (runner_.running_suite_ == &target)
224 {
225 return true;
226 }
227
228 if (runner_.aborted_ || runner_.has_option_errors ()
229 || runner_.options_.help)
230 {
231 return false;
232 }
233
234 std::string message{ "test case '" };
235 message.append (name);
236 message.append ("' started on suite '");
237 message.append (target.name ());
238 message.append ("', which is not running");
239
240 suite* running = runner_.running_suite_;
241 if (running != nullptr && !runner_.options_.list_tests)
242 {
243 running->next_test_point_number ();
244 running->test_reporter ().fail_suite (message, *running);
245 running->totals_.increment_failed_checks ();
246 return false;
247 }
248
249 fprintf (stderr, "error: %s\n", message.c_str ());
250 runner_.mark_rejected ();
251 return false;
252 }

305 void
306 runnable_base::after_test_case_create (class test_case& test_case,
307 suite& suite)
308 {
309 if (runner_.aborted_ || runner_.has_option_errors ()
310 || runner_.options_.help)
311 {
312 return;
313 }
314
315 ++child_test_cases_count_;
316
317 // While the test cases are listed, the selected ones are listed and
318 // the others ignored; none is run.
319 if (runner_.options_.list_tests)
320 {
321 if (runner_.is_selected (suite.name (), 0)
322 && runner_.is_selected (test_case.name (),
323 test_case.nesting_depth ()))
324 {
325 runner_.list_test_case (suite, test_case);
326 }
327 return;
328 }
329
330 // Skip the test cases not selected by the filter; the outer test
331 // cases, if any, were selected, otherwise this one was not created.
332 if (!runner_.is_selected (suite.name (), 0)
333 || !runner_.is_selected (test_case.name (),
334 test_case.nesting_depth ()))
335 {
336 test_reporter ().skip_test_case (test_case);
337 totals_.increment_skipped_test_cases ();
338 return;
339 }
340
341 // Run the child test case immediately, as the innermost running test
342 // case of the runner. The outer one is restored by the guard, also
343 // when an exception leaves run() (for example thrown by an abort
344 // handler), so that the runner never keeps a destroyed test case.
345 {
346 struct running_guard
347 {
348 class test_case*& running;
349 class test_case* outer;
350
351 ~running_guard ()
352 {
353 running = outer;
354 }
355 } guard{ runner_.running_test_case_, runner_.running_test_case_ };
356
357 runner_.running_test_case_ = &test_case;
358 // A non-virtual call, so that run() is inlined here; each nested
359 // test case adds this call chain to the stack.
360 test_case.test_case::run ();
361 }
362
363 // This test executed one more test case.
364#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
365 trace::printf ("%s test case '%s' executed one more test case\n",
366 __PRETTY_FUNCTION__, name ());
367#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
368 totals_.increment_executed_test_cases ();
369
370 // Do not accumulate the totals from the child test into the current test
371 // totals, each test case shows only its counters. But a failed child
372 // fails this test too, as in node-tap; the failure reaches all outer
373 // tests, one level at a time, when they end.
374 if (!test_case.totals ().passed ())
375 {
376 totals_.increment_failed_test_cases ();
377 }
378
379#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
380 trace::printf ("%s suite '%s' totals\n", __PRETTY_FUNCTION__,
381 suite.name ());
382#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
383
384 // Accumulate the totals from the child test into the suite totals.
385 suite.totals_ += test_case.totals ();
386 }

525 // Marks this suite as the one that accepts new test cases, until the
526 // body returns.
527 struct running_guard
528 {
529 suite& self;
530 suite* outer;
531
532 ~running_guard ()
533 {
534 self.exchange_running_suite (outer);
535 }
536 } guard{ *this, exchange_running_suite (this) };
537
538 // While the test cases are listed, the body only creates them, so
539 // that they are listed; the suite is not reported. An exception
540 // thrown by the body is reported on stderr, and the listing
541 // continues with the next suite.
542 if (is_listing ())
543 {
544#if defined(__cpp_exceptions)
545 try
546 {
547 callable_ (*this);
548 }
549 catch (...)
550 {
551 if (is_aborted ())
552 {
553 throw;
554 }
555 fprintf (stderr,
556 "error: unexpected exception in the body of suite "
557 "'%s'\n",
558 name_);
559 }
560#else
561 callable_ (*this);
562#endif // defined(__cpp_exceptions)
563 return;
564 }
565
566 detail::reporter& reporter = this->test_reporter ();
567
568 this->stopwatch_.start ();
569 reporter.begin_suite (*this);
570
571#if defined(__cpp_exceptions)
572 try
573 {
574 // Invoke the callable, passing the self reference.
575 callable_ (*this);
576 }
577 catch (const std::exception& e)
578 {
579 if (is_aborted ())
580 {
581 // Thrown by the abort handler; the report is complete.
582 throw;
583 }
584 fail_unexpected_exception (e.what ());
585 }
586 catch (...)
587 {
588 if (is_aborted ())
589 {
590 // Thrown by the abort handler; the report is complete.
591 throw;
592 }
593 fail_unexpected_exception (nullptr);
594 }
595#else
596 // Invoke the callable, passing the self reference.
597 callable_ (*this);
598#endif // defined(__cpp_exceptions)
599
600 this->stopwatch_.stop ();
601 reporter.end_suite (*this);
602 }

655 void
656 suite::link (void) noexcept
657 {
658#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
659 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
660#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
661
662 // std::strcmp() compares as unsigned char, like std::string_view,
663 // without a temporary object (which triggers -Waggregate-return).
664 suite** slot = &first_;
665 while (*slot != nullptr && std::strcmp ((*slot)->name_, name_) <= 0)
666 {
667 slot = &(*slot)->next_;
668 }
669 next_ = *slot;
670 *slot = this;
671 }

677 void
678 suite::unlink (void) noexcept
679 {
680 for (suite** slot = &first_; *slot != nullptr; slot = &(*slot)->next_)
681 {
682 if (*slot == this)
683 {
684 *slot = next_;
685 next_ = nullptr;
686 return;
687 }
688 }
