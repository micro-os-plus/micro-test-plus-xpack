200 void
201 runnable_base::after_test_case_create (class test_case& test_case,
202 suite& suite)
203 {
204 if (runner_.aborted_)
205 {
206 return;
207 }
208
209 ++child_test_cases_count_;
210
211 // While the test cases are listed, the selected ones are listed and
212 // the others ignored; none is run.
213 if (runner_.options_.list_tests)
214 {
215 if (runner_.is_selected (suite.name (), 0)
216 && runner_.is_selected (test_case.name (),
217 test_case.nesting_depth ()))
218 {
219 runner_.list_test_case (suite, test_case);
220 }
221 return;
222 }
223
224 // Skip the test cases not selected by the filter; the outer test
225 // cases, if any, were selected, otherwise this one was not created.
226 if (!runner_.is_selected (suite.name (), 0)
227 || !runner_.is_selected (test_case.name (),
228 test_case.nesting_depth ()))
229 {
230 test_reporter ().skip_test_case (test_case);
231 totals_.increment_skipped_test_cases ();
232 return;
233 }
234
235 // Run the child test case immediately, as the innermost running test
236 // case of the runner. The outer one is restored by the guard, also
237 // when an exception leaves run() (for example thrown by an abort
238 // handler), so that the runner never keeps a destroyed test case.
239 {
240 struct running_guard
241 {
242 class test_case*& running;
243 class test_case* outer;
244
245 ~running_guard ()
246 {
247 running = outer;
248 }
249 } guard{ runner_.running_test_case_, runner_.running_test_case_ };
250
251 runner_.running_test_case_ = &test_case;
252 // A non-virtual call, so that run() is inlined here; each nested
253 // test case adds this call chain to the stack.
254 test_case.test_case::run ();
255 }
256
257 // This test executed one more test case.
258#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
259 trace::printf ("%s test case '%s' executed one more test case\n",
260 __PRETTY_FUNCTION__, name ());
261#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
262 totals_.increment_executed_test_cases ();
263
264 // Do not accumulate the totals from the child test into the current test
265 // totals, each test case shows only its counters. But a failed child
266 // fails this test too, as in node-tap; the failure reaches all outer
267 // tests, one level at a time, when they end.
268 if (!test_case.totals ().passed ())
269 {
270 totals_.increment_failed_test_cases ();
271 }
272
273#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
274 trace::printf ("%s suite '%s' totals\n", __PRETTY_FUNCTION__,
275 suite.name ());
276#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
277
278 // Accumulate the totals from the child test into the suite totals.
279 suite.totals_ += test_case.totals ();
280 }

373 void
374 suite::run (void)
375 {
376#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
377 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
378#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
379
380 // While the test cases are listed, the body only creates them, so
381 // that they are listed; the suite is not reported. An exception
382 // thrown by the body is reported on stderr, and the listing
383 // continues with the next suite.
384 if (is_listing ())
385 {
386#if defined(__cpp_exceptions)
387 try
388 {
389 callable_ (*this);
390 }
391 catch (...)
392 {
393 if (is_aborted ())
394 {
395 throw;
396 }
397 fprintf (stderr,
398 "error: unexpected exception in the body of suite "
399 "'%s'\n",
400 name_);
401 }
402#else
403 callable_ (*this);
404#endif // defined(__cpp_exceptions)
405 return;
406 }
407
408 detail::reporter& reporter = this->test_reporter ();
409
410 this->stopwatch_.start ();
411 reporter.begin_suite (*this);
412
413#if defined(__cpp_exceptions)
414 try
415 {
416 // Invoke the callable, passing the self reference followed by the
417 // variadic arguments.
418 callable_ (*this);
419 }
420 catch (const std::exception& e)
421 {
422 if (is_aborted ())
423 {
424 // Thrown by the abort handler; the report is complete.
425 throw;
426 }
427 fail_unexpected_exception (e.what ());
428 }
429 catch (...)
430 {
431 if (is_aborted ())
432 {
433 // Thrown by the abort handler; the report is complete.
434 throw;
435 }
436 fail_unexpected_exception (nullptr);
437 }
438#else
439 // Invoke the callable, passing the self reference followed by the variadic
440 // arguments.
441 callable_ (*this);
442#endif // defined(__cpp_exceptions)
443
444 this->stopwatch_.stop ();
445 reporter.end_suite (*this);
446 }

499 void
500 suite::link (void) noexcept
501 {
502#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
503 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
504#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
505
506 // std::strcmp() compares as unsigned char, like std::string_view,
507 // without a temporary object (which triggers -Waggregate-return).
508 suite** slot = &first_;
509 while (*slot != nullptr && std::strcmp ((*slot)->name_, name_) <= 0)
510 {
511 slot = &(*slot)->next_;
512 }
513 next_ = *slot;
514 *slot = this;
515 }

521 void
522 suite::unlink (void) noexcept
523 {
524 for (suite** slot = &first_; *slot != nullptr; slot = &(*slot)->next_)
525 {
526 if (*slot == this)
527 {
528 *slot = next_;
529 next_ = nullptr;
530 return;
531 }
532 }
