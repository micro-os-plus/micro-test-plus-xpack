79 void
80 escape_control_characters (std::string& name)
81 {
82 const auto is_control = [] (char c)
83 {
84 const auto unit = static_cast<unsigned char> (c);
85 return (unit < 0x20 && c != '\t') || unit == 0x7F;
86 };
87
88 std::size_t first = 0;
89 while (first < name.size () && !is_control (name[first]))
90 {
91 ++first;
92 }
93 if (first == name.size ())
94 {
95 return;
96 }
97
98 std::string escaped (name, 0, first);
99 escaped.reserve (name.size () + 8);
100 for (std::size_t i = first; i < name.size (); ++i)
101 {
102 if (is_control (name[i]))
103 {
104 append_control_escape (escaped, name[i]);
105 }
106 else
107 {
108 escaped.push_back (name[i]);
109 }
110 }
111 name = std::move (escaped);
112 }

238 bool
239 runnable_base::accepts_test_case (const suite& target,
240 const located_name& name)
241 {
242 if (runner_.running_suite_ == &target)
243 {
244 return true;
245 }
246
247 if (runner_.aborted_ || runner_.has_option_errors ()
248 || runner_.options_.help)
249 {
250 return false;
251 }
252
253 std::string message{ "test case '" };
254 message.append (name.name ());
255 message.append ("' started on suite '");
256 message.append (target.name ());
257 message.append ("', which is not running");
258
259 if (runner_.running_test_case_ != nullptr
260 && !runner_.options_.list_tests)
261 {
262 runner_.running_test_case_->report_rejected_test (message,
263 name.location ());
264 return false;
265 }
266
267 suite* running = runner_.running_suite_;
268 if (running != nullptr && !runner_.options_.list_tests)
269 {
270 running->next_test_point_number ();
271 running->test_reporter ().fail_suite (message, *running);
272 running->totals_.increment_failed_checks ();
273 return false;
274 }
275
276 runner_.reject_call (message, &name.location ());
277 return false;
278 }

349 void
350 runnable_base::after_test_case_create (class test_case& test_case,
351 suite& suite)
352 {
353 if (runner_.aborted_ || runner_.has_option_errors ()
354 || runner_.options_.help)
355 {
356 return;
357 }
358
359 ++child_test_cases_count_;
360
361 // While the test cases are listed, the selected ones are listed and
362 // the others ignored; none is run.
363 if (runner_.options_.list_tests)
364 {
365 if (runner_.is_selected (suite.name (), 0)
366 && runner_.is_selected (test_case.name (),
367 test_case.nesting_depth ()))
368 {
369 runner_.list_test_case (suite, test_case);
370 }
371 return;
372 }
373
374 // Skip the test cases not selected by the filter; the outer test
375 // cases, if any, were selected, otherwise this one was not created.
376 if (!runner_.is_selected (suite.name (), 0)
377 || !runner_.is_selected (test_case.name (),
378 test_case.nesting_depth ()))
379 {
380 test_reporter ().skip_test_case (test_case);
381 totals_.increment_skipped_test_cases ();
382 return;
383 }
384
385 // Run the child test case immediately, as the innermost running test
386 // case of the runner. The outer one is restored by the guard, also
387 // when an exception leaves run() (for example thrown by an abort
388 // handler), so that the runner never keeps a destroyed test case.
389 {
390 struct running_guard
391 {
392 class test_case*& running;
393 class test_case* outer;
394
395 ~running_guard ()
396 {
397 running = outer;
398 }
399 } guard{ runner_.running_test_case_, runner_.running_test_case_ };
400
401 runner_.running_test_case_ = &test_case;
402 // A non-virtual call, so that run() is inlined here; each nested
403 // test case adds this call chain to the stack.
404 test_case.test_case::run ();
405 }
406
407 // This test executed one more test case.
408#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
409 trace::printf ("%s test case '%s' executed one more test case\n",
410 __PRETTY_FUNCTION__, name ());
411#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
412 totals_.increment_executed_test_cases ();
413
414 // Do not accumulate the totals from the child test into the current test
415 // totals, each test case shows only its counters. But a failed child
416 // fails this test too, as in node-tap; the failure reaches all outer
417 // tests, one level at a time, when they end.
418 if (!test_case.totals ().passed ())
419 {
420 totals_.increment_failed_test_cases ();
421 }
422
423#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
424 trace::printf ("%s suite '%s' totals\n", __PRETTY_FUNCTION__,
425 suite.name ());
426#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
427
428 // Accumulate the totals from the child test into the suite totals.
429 suite.totals_ += test_case.totals ();
430 }

604 suite::run (void)

606#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)

608#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED

610 // Marks this suite as the one that accepts new test cases, until the
611 // body returns.
612 struct running_guard
613 {
614 suite& self;
615 suite* outer;
616
617 ~running_guard ()
618 {
619 self.exchange_running_suite (outer);
620 }
621 } guard{ *this, exchange_running_suite (this) };
622
623 // While the test cases are listed, the body only creates them, so
624 // that they are listed; the suite is not reported. An exception
625 // thrown by the body is reported on stderr and in the listing, and
626 // the listing continues with the next suite.
627 if (is_listing ())
628 {
629#if defined(__cpp_exceptions)
630 try
631 {
632 callable_ (*this);
633 }
634 catch (...)
635 {
636 if (is_aborted ())
637 {
638 throw;
639 }
640 reject_body_exception ();
641 }
642#else
643 callable_ (*this);
644#endif // defined(__cpp_exceptions)
645 return;
646 }
647
648 detail::reporter& reporter = this->test_reporter ();
649
650 this->stopwatch_.start ();
651 reporter.begin_suite (*this);
652
653#if defined(__cpp_exceptions)
654 try
655 {
656 // Invoke the callable, passing the self reference.
657 callable_ (*this);
658 }
659 catch (const std::exception& e)
660 {
661 if (is_aborted ())
662 {
663 // Thrown by the abort handler; the report is complete.
664 throw;
665 }
666 fail_unexpected_exception (e.what ());
667 }
668 catch (...)
669 {
670 if (is_aborted ())
671 {
672 // Thrown by the abort handler; the report is complete.
673 throw;
674 }
675 fail_unexpected_exception (nullptr);
676 }
677#else
678 // Invoke the callable, passing the self reference.
679 callable_ (*this);
680#endif // defined(__cpp_exceptions)
681
682 this->stopwatch_.stop ();
683 reporter.end_suite (*this);
684 }

737 void
738 suite::link (void) noexcept
739 {
740#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
741 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, name_);
742#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
743
744 // std::strcmp() compares as unsigned char, like std::string_view,
745 // without a temporary object (which triggers -Waggregate-return).
746 suite** slot = &first_;
747 while (*slot != nullptr && std::strcmp ((*slot)->name_, name_) <= 0)
748 {
749 slot = &(*slot)->next_;
750 }
751 next_ = *slot;
752 *slot = this;
753 }

759 void
760 suite::unlink (void) noexcept
761 {
762 for (suite** slot = &first_; *slot != nullptr; slot = &(*slot)->next_)
763 {
764 if (*slot == this)
765 {
766 *slot = next_;
767 next_ = nullptr;
768 return;
769 }
770 }
