205 void
206 runnable_base::after_subtest_create_ (
207 std::unique_ptr<class subtest> child_test, suite& suite)
208 {
209 // Transfer ownership into the vector.
210 children_subtests_.push_back (std::move (child_test));
211
212 // Run the child test case immediately.
213 class subtest& subtest = *children_subtests_.back ();
214 subtest.run ();
215
216 // This test executed one more subtest.
217#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
218#if defined(__GNUC__)
219#pragma GCC diagnostic push
220#if defined(__clang__)
221#pragma clang diagnostic ignored "-Wunsafe-buffer-usage-in-libc-call"
222#endif
223#endif
224 trace::printf ("%s subtest '%s' executed one more subtest\n",
225 __PRETTY_FUNCTION__, name ());
226#if defined(__GNUC__)
227#pragma GCC diagnostic pop
228#endif
229#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
230 totals ().increment_executed_subtests ();
231 // Do not accumulate the totals from the child test into the current test
232 // totals, each subtest shows only its counters.
233
234#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
235#if defined(__GNUC__)
236#pragma GCC diagnostic push
237#if defined(__clang__)
238#pragma clang diagnostic ignored "-Wunsafe-buffer-usage-in-libc-call"
239#endif
240#endif
241 trace::printf ("%s suite '%s' totals\n", __PRETTY_FUNCTION__,
242 suite.name ());
243#if defined(__GNUC__)
244#pragma GCC diagnostic pop
245#endif
246#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
247
248 // Accumulate the totals from the child test into the suite totals.
249 suite.totals () += subtest.totals ();
250 }
