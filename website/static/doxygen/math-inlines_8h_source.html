173 template <class T, char... Cs>
174 consteval auto
175 num (void) -> T
176 {
177 static_assert (
178 ((Cs == '.' or Cs == '\'' or (Cs >= '0' and Cs <= '9')) and ...),
179 "µTest++: a floating-point literal must be written with decimal "
180 "digits and a decimal point, without an exponent");
181 static_assert (
182 count_literal_digits<false, Cs...> ()
183 <= static_cast<std::size_t> (std::numeric_limits<T>::digits10),
184 "µTest++: the floating-point literal has too many "
185 "digits before the decimal point");
186 T result{};
187 for (const char c : { Cs... })
188 {
189 if (c == '.')
190 {
191 break;
192 }
193 if (c >= '0' and c <= '9')
194 {
195 result = static_cast<T> (result * static_cast<T> (10)
196 + static_cast<T> (c - '0'));
197 }
198 }
199 return result;
200 }

297 template <class T, class Value_T>
298 constexpr auto
299 count_decimals (Value_T value) -> T
300 {
301 constexpr auto precision = Value_T (1e-7);
302 // From this magnitude upwards all values are integers.
303 constexpr auto integral_limit
304 = min_value (Value_T (1) / std::numeric_limits<Value_T>::epsilon (),
305 Value_T (1ULL << 63));
306
307 value = abs (value);
308
309#if defined(__GNUC__)
310#pragma GCC diagnostic push
311#pragma GCC diagnostic ignored "-Wfloat-equal"
312#endif // defined(__GNUC__)
313 if (value == Value_T (0))
314 {
315 // A zero has no decimals.
316 return T{};
317 }
318#if defined(__GNUC__)
319#pragma GCC diagnostic pop
320#endif // defined(__GNUC__)
321
322 T result{};
323 for (;;)
324 {
325 value *= 10;
326 ++result;
327
328 // Also true for infinities and NaNs.
329 if (not(value < integral_limit))
330 {
331 break;
332 }
333
334 const auto fraction = value
335 - static_cast<Value_T> (
336 static_cast<unsigned long long> (value));
337 // Below one, the distance is to one, not to zero, so that the
338 // leading zeros of a small value are counted.
339 const auto distance
340 = (value < Value_T (1))
341 ? Value_T (1) - fraction
342 : min_value (fraction, Value_T (1) - fraction);
343 if (distance <= precision)
344 {
345 break;
346 }
347 }
348
349 return result;
350 }

365 template <class Value_T>
366 constexpr auto
367 decimal_epsilon (Value_T value) -> Value_T
368 {
369 const auto decimals = count_decimals<unsigned long long> (value);
370 if (decimals == 0
371 or decimals > static_cast<unsigned long long> (
372 std::numeric_limits<Value_T>::max_exponent10))
373 {
374 return Value_T{};
375 }
376 return Value_T (1) / pow (Value_T (10), decimals);
377 }

393 template <char... Cs>
394 consteval auto
395 parse_integer_literal (void) -> integer_literal
396 {
397 constexpr std::array<char, sizeof...(Cs)> cs{ Cs... };
398 constexpr unsigned long long max_value
399 = std::numeric_limits<unsigned long long>::max ();
400
401 integer_literal result{ 0, true, false };
402
403 unsigned long long base = 10;
404 std::size_t i = 0;
405 if (cs.size () > 1 && cs[0] == '0')
406 {
407 if (cs[1] == 'x' || cs[1] == 'X')
408 {
409 base = 16;
410 i = 2;
411 }
412 else if (cs[1] == 'b' || cs[1] == 'B')
413 {
414 base = 2;
415 i = 2;
416 }
417 else
418 {
419 base = 8;
420 i = 1;
421 }
422 }
423
424 for (; i < cs.size (); ++i)
425 {
426 const char c = cs[i];
427 if (c == '\'')
428 {
429 continue;
430 }
431
432 unsigned long long digit = 0;
433 if (c >= '0' && c <= '9')
434 {
435 digit = static_cast<unsigned long long> (c - '0');
436 }
437 else if (base == 16 && c >= 'a' && c <= 'f')
438 {
439 digit = static_cast<unsigned long long> (c - 'a' + 10);
440 }
441 else if (base == 16 && c >= 'A' && c <= 'F')
442 {
443 digit = static_cast<unsigned long long> (c - 'A' + 10);
444 }
445 else
446 {
447 // A decimal point, an exponent, or a suffix character.
448 result.is_integer = false;
449 return result;
450 }
451
452 if (digit >= base)
453 {
454 // Possible only in a floating-point literal, like `09.5`.
455 result.is_integer = false;
456 return result;
457 }
458
459 if (result.value > (max_value - digit) / base)
460 {
461 result.is_too_large = true;
462 return result;
463 }
464
465 result.value = result.value * base + digit;
466 }
467
468 return result;
469 }

487 template <class T, char... Cs>
488 consteval auto
489 integer_magnitude (void) -> unsigned long long
490 {
491 constexpr integer_literal literal = parse_integer_literal<Cs...> ();
492 constexpr unsigned long long max_magnitude
493 = static_cast<unsigned long long> (std::numeric_limits<T>::max ())
494 + (std::numeric_limits<T>::is_signed ? 1u : 0u);
495 constexpr bool fits = literal.is_integer && !literal.is_too_large
496 && literal.value <= max_magnitude;
497
498 static_assert (literal.is_integer,
499 "µTest++: an integer literal must not have a decimal "
500 "point or an exponent");
501 static_assert (!literal.is_integer || !literal.is_too_large,
502 "µTest++: the integer literal is too large");
503 static_assert (!literal.is_integer || literal.is_too_large || fits,
504 "µTest++: the integer literal value does not fit in the "
505 "type of the literal");
506
507 return fits ? literal.value : 0;
508 }
