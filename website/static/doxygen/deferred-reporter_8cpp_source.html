89 deferred_reporter_base::~deferred_reporter_base ()
90 {
91#if defined(MICRO_OS_PLUS_TRACE) \
92 && defined(MICRO_OS_PLUS_TRACE_MICRO_TEST_PLUS_CONSTRUCTORS)
93 trace::printf ("%s\n", __PRETTY_FUNCTION__);
94#endif // MICRO_OS_PLUS_TRACE_MICRO_TEST_PLUS_CONSTRUCTORS
95
96 if (value_) [[likely]]
97 {
98 subtest_.totals ().increment_successful_checks ();
99 }
100 else
101 {
102 subtest_.totals ().increment_failed_checks ();
103 }
104
105 if (abort_ && !value_) [[unlikely]]
106 {
107 subtest_.reporter ().write_buffer_to_stdout ();
108 subtest_.reporter ().flush ();
109 subtest_.abort (location_);
110 }
111 }
