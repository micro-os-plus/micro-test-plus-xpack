242 void
243 reporter_human::end_suite (const suite& suite)
244 {
245#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
246 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, suite.name ());
247#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
248
249 char message_time[elapsed_time_size_];
250 format_elapsed_time (suite.stopwatch_, message_time);
251
252 // At this point, the buffer may contain output from the test case, which
253 // should be displayed.
254 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
255 {
256 char message_totals[totals_size_];
257 format_suite_totals (suite, message_totals);
258
259 if (suite.totals ().executed_test_cases () > 0)
260 {
261 write_console ("\n");
262 }
263
264 if (suite.totals ().passed ()) [[likely]]
265 {
266 // Successful test suite.
267
268 write_buffer_to_file ();
269
270 if (verbosity_ == verbosity::verbose)
271 {
272 // With verbosity, show the full output accumulated in the
273 // buffer.
274 write_buffer_to_console ();
275 }
276
277 write_line (true, "%s✓%s %s - passed (%s)%s\n", colours_.pass,
278 colours_.none, suite.name (), message_totals,
279 message_time);
280 }
281 else
282 {
283 // Failed test suite.
284
285 write_buffer_to_file ();
286
287 // Show the full output accumulated in the buffer for a failed
288 // suite, as it may contain useful information about the failure.
289 write_buffer_to_console ();
290
291 write_line (true, "%s✗%s %s - %sFAILED%s (%s)%s\n", colours_.fail,
292 colours_.none, suite.name (), colours_.fail,
293 colours_.none, message_totals, message_time);
294 }
295 }
296
297 flush_streams ();
298
299 // Clear residual content when less verbose.
300 clear_buffer ();
301
302 add_empty_line_ = true;
303 }

372 reporter_human::begin_test_case (const test_case& test_case)
373 {
374#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
375 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, test_case.name ());

377
378 commit_outer_results ();
379
380 open_test_cases_.push_back (

382
383 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
384
385 const bool to_console = (verbosity_ == verbosity::verbose);
386
387 if (to_console && add_empty_line_)
388 {
389 write_console ("\n");
390 }
391
392 write_line (to_console, "%s• %s\n", indent.c_str (), test_case.name ());
393
394 if (to_console)
395 {
396 add_empty_line_ = false;
397 }

421 void
422 reporter_human::end_test_case (const test_case& test_case)
423 {
424#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
425 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
426 __PRETTY_FUNCTION__, test_case.name (),
427 test_case.nesting_depth ());
428#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
429
430 // At this point, the buffer may contain output from the test case, which
431 // should be displayed.
432 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
433 {
434 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
435
436 // The name is not included, since its length is not bounded;
437 // it is passed to each write.
438 char message_totals[totals_size_];
439 format_test_case_totals (test_case, message_totals);
440
441 if (add_empty_line_)
442 {
443 write_console ("\n");
444 }
445
446 if (test_case.totals ().passed ()) [[likely]]
447 {
448 // Successful test case.
449
450 write_buffer_to_file ();
451
452 if (verbosity_ == verbosity::verbose)
453 {
454 // With verbosity, show the full output accumulated in the
455 // buffer.
456 write_buffer_to_console ();
457 }
458
459 write_line (true, "%s%s✓%s %s - passed (%s)\n", indent.c_str (),
460 colours_.pass, colours_.none, test_case.name (),
461 message_totals);
462
463 add_empty_line_ = (verbosity_ == verbosity::verbose);
464 }
465 else
466 {
467 // Failed test case.
468
469 write_buffer_to_file ();
470
471 if (verbosity_ == verbosity::normal)
472 {
473 // This test case and the outer ones, whose names may have
474 // been displayed already, by `commit_buffer()` or by a
475 // failed sibling.
476 display_open_names ();
477 }
478
479 // Show full output accumulated in the buffer for failed
480 // test cases, as it may contain useful information about the
481 // failure.
482 write_buffer_to_console ();
483
484 write_line (true, "%s%s✗%s %s - %sFAILED%s (%s)\n",
485 indent.c_str (), colours_.fail, colours_.none,
486 test_case.name (), colours_.fail, colours_.none,
487 message_totals);
488
489 add_empty_line_ = true;
490 }
491 }
492
493 flush_streams ();
494
495 // Clear residual content when less verbose.
496 clear_buffer ();
497
498 if (!open_test_cases_.empty ())
499 {
500 open_test_cases_.pop_back ();
501 }
502 }

584 reporter_human::output_fail_prefix (

719 void
720 reporter_human::abort_session (const reflection::source_location& location,
721 [[maybe_unused]] bool is_requirement)
722 {
723#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
724 trace::printf ("%s\n", __PRETTY_FUNCTION__);
725#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
726
727 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
728 {
729 write_buffer_to_file ();
730 }
731
732 if (verbosity_ != verbosity::silent)
733 {
734 // Whether names or results are displayed before the final line.
735 bool has_results = !buffer_.empty ();
736 for (const open_test_case& open : open_test_cases_)
737 {
738 has_results = has_results || !open.name_displayed;
739 }
740
741 if (add_empty_line_)
742 {
743 write_console ("\n");
744 }
745
746 display_open_names ();
747
748 write_buffer_to_console ();
749
750 // Separate the final line by an empty line, as the summary, unless
751 // it was already written above.
752 if (has_results || !add_empty_line_)
753 {
754 write_console ("\n");
755 }
756
757 // The type of `line()` depends on the toolchain (`uint_least32_t`
758 // for `std::source_location`); widen it implicitly, without a cast
759 // that would be useless on some platforms.
760 const unsigned long line = location.line ();
761 const char* file_name
762 = utility::extract_file_name (location.file_name ());
763
764 write_line (true, "%s✗%s Test execution aborted at %s:%lu\n",
765 colours_.fail, colours_.none, file_name, line);
766 }
767
768 flush_streams ();
769
770 clear_buffer ();
771 }

789 void
790 reporter_human::commit_buffer (void)
791 {
792 // As in `end_test_case()`, the results reach the output file only with
793 // normal or verbose verbosity.
794 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
795 {
796 write_buffer_to_file ();
797 }
798
799 if (verbosity_ == verbosity::verbose)
800 {

810 {

815
816 display_open_names ();
817
818 write_buffer_to_console ();

820 add_empty_line_ = true;
