254 void
255 reporter_human::end_suite (const suite& suite)
256 {
257#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
258 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, suite.name ());
259#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
260
261 const bool to_console = route (record_kind::suite_result).console;
262 const bool passed = suite.totals ().passed ();
263
264 char message_time[elapsed_time_size_];
265 format_elapsed_time (suite.stopwatch_, message_time);
266
267 char message_totals[totals_size_];
268 format_suite_totals (suite, message_totals);
269
270 // At this point, the buffer may contain output from the test case, which
271 // should be displayed.
272 const std::string_view lines = console_lines (!passed);
273
274 write_buffer_to_file ();
275
276 if (to_console)
277 {
278 if (suite.totals ().executed_test_cases () > 0)
279 {
280 write_console ("\n");
281 }
282
283 write_console_text (lines);
284 }
285 else if (!lines.empty ())
286 {
287 // The failure of the suite body, with quiet verbosity.
288 if (add_empty_line_)
289 {
290 write_console ("\n");
291 }
292
293 display_open_names (true);
294
295 write_console_text (lines);
296 }
297
298 if (passed) [[likely]]
299 {
300 // Successful test suite.
301
302 write_line (to_console, "%s✓%s %s - passed (%s)%s\n", colours_.pass,
303 colours_.none, suite.name (), message_totals,
304 message_time);
305 }
306 else
307 {
308 // Failed test suite.
309
310 write_line (to_console, "%s✗%s %s - %sFAILED%s (%s)%s\n",
311 colours_.fail, colours_.none, suite.name (), colours_.fail,
312 colours_.none, message_totals, message_time);
313 }
314
315 flush_streams ();
316
317 // Clear residual content when less verbose.
318 clear_buffer ();
319
320 add_empty_line_ = true;
321
322 current_suite_ = nullptr;
323 }

451 void
452 reporter_human::end_test_case (const test_case& test_case)
453 {
454#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
455 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
456 __PRETTY_FUNCTION__, test_case.name (),
457 test_case.nesting_depth ());
458#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
459
460 const bool to_console = route (record_kind::test_case_result).console;
461 const bool passed = test_case.totals ().passed ();
462
463 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
464
465 // The name is not included, since its length is not bounded;
466 // it is passed to each write.
467 char message_totals[totals_size_];
468 format_test_case_totals (test_case, message_totals);
469
470 // At this point, the buffer may contain output from the test case, which
471 // should be displayed.
472 const std::string_view lines = console_lines (!passed);
473
474 write_buffer_to_file ();
475
476 const bool shows = to_console || !lines.empty ();
477 if (shows)
478 {
479 if (add_empty_line_)
480 {
481 write_console ("\n");
482 }
483
484 // The names of this test case and of the outer ones, whose names
485 // may have been displayed already, by `commit_buffer()` or by a
486 // failed sibling; for a passed test case, only the outer ones.
487 display_open_names (!passed);
488
489 // With verbosity, the full output accumulated in the buffer;
490 // for a failed test case, as it may contain useful information
491 // about the failure.
492 write_console_text (lines);
493 }
494
495 if (passed) [[likely]]
496 {
497 // Successful test case.
498
499 write_line (to_console, "%s%s✓%s %s - passed (%s)\n", indent.c_str (),
500 colours_.pass, colours_.none, test_case.name (),
501 message_totals);
502 }
503 else
504 {
505 // Failed test case.
506
507 write_line (to_console, "%s%s✗%s %s - %sFAILED%s (%s)\n",
508 indent.c_str (), colours_.fail, colours_.none,
509 test_case.name (), colours_.fail, colours_.none,
510 message_totals);
511 }
512
513 if (shows)
514 {
515 add_empty_line_
516 = !passed || route (record_kind::test_case_heading).console;
517 }
518
519 flush_streams ();
520
521 // Clear residual content when less verbose.
522 clear_buffer ();
523
524 if (!open_test_cases_.empty ())
525 {
526 open_test_cases_.pop_back ();
527 }
528 }

595 reporter_human::output_fail_prefix (

750 void
751 reporter_human::abort_session (const reflection::source_location& location,
752 [[maybe_unused]] bool is_requirement)
753 {
754#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
755 trace::printf ("%s\n", __PRETTY_FUNCTION__);
756#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
757
758 const bool to_console = route (record_kind::abort).console;
759
760 // The type of `line()` depends on the toolchain (`uint_least32_t`
761 // for `std::source_location`); widen it implicitly, without a cast
762 // that would be useless on some platforms.
763 const unsigned long line = location.line ();
764 const char* file_name = utility::extract_file_name (location.file_name ());
765
766 const std::string_view lines = console_lines (true);
767
768 write_buffer_to_file ();
769
770 if (to_console)
771 {
772 // Whether headings or results are displayed before the final line.
773 bool has_results = !lines.empty ()
774 || (current_suite_ != nullptr && !suite_displayed_);
775 for (const open_test_case& open : open_test_cases_)
776 {
777 has_results = has_results || !open.name_displayed;
778 }
779
780 if (add_empty_line_)
781 {
782 write_console ("\n");
783 }
784
785 display_open_names (true);
786
787 write_console_text (lines);
788
789 // Separate the final line by an empty line, as the summary, unless
790 // it was already written above.
791 if (has_results || !add_empty_line_)
792 {
793 write_console ("\n");
794 }
795 }
796
797 write_line (to_console, "%s✗%s Test execution aborted at %s:%lu\n",
798 colours_.fail, colours_.none, file_name, line);
799
800 flush_streams ();
801
802 announce_output_file ();
803
804 clear_buffer ();
805
806 // Should the abort handler throw, nothing may refer to the objects
807 // that are unwound.
808 open_test_cases_.clear ();
809 current_suite_ = nullptr;
810 }
