253 void
254 reporter_human::end_suite (const suite& suite)
255 {
256#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
257 trace::printf ("%s '%s'\n", __PRETTY_FUNCTION__, suite.name ());
258#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
259
260 const bool to_console = route (record_kind::suite_result).console;
261 const bool passed = suite.totals ().passed ();
262
263 char message_time[elapsed_time_size_];
264 format_elapsed_time (suite.stopwatch_, message_time);
265
266 char message_totals[totals_size_];
267 format_suite_totals (suite, message_totals);
268
269 // At this point, the buffer may contain output from the test case, which
270 // should be displayed.
271 const std::string_view lines = console_lines (!passed);
272
273 write_buffer_to_file ();
274
275 if (to_console)
276 {
277 if (suite.totals ().executed_test_cases () > 0)
278 {
279 write_console ("\n");
280 }
281
282 write_console_text (lines);
283 }
284 else if (!lines.empty ())
285 {
286 // The failure of the suite body, with quiet verbosity.
287 if (add_empty_line_)
288 {
289 write_console ("\n");
290 }
291
292 display_open_names (true);
293
294 write_console_text (lines);
295 }
296
297 if (passed) [[likely]]
298 {
299 // Successful test suite.
300
301 write_line (to_console, "%s✓%s %s - passed (%s)%s\n", colours_.pass,
302 colours_.none, suite.name (), message_totals,
303 message_time);
304 }
305 else
306 {
307 // Failed test suite.
308
309 write_line (to_console, "%s✗%s %s - %sFAILED%s (%s)%s\n",
310 colours_.fail, colours_.none, suite.name (), colours_.fail,
311 colours_.none, message_totals, message_time);
312 }
313
314 flush_streams ();
315
316 // Clear residual content when less verbose.
317 clear_buffer ();
318
319 add_empty_line_ = true;
320
321 current_suite_ = nullptr;
322 }

450 void
451 reporter_human::end_test_case (const test_case& test_case)
452 {
453#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
454 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
455 __PRETTY_FUNCTION__, test_case.name (),
456 test_case.nesting_depth ());
457#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
458
459 const bool to_console = route (record_kind::test_case_result).console;
460 const bool passed = test_case.totals ().passed ();
461
462 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
463
464 // The name is not included, since its length is not bounded;
465 // it is passed to each write.
466 char message_totals[totals_size_];
467 format_test_case_totals (test_case, message_totals);
468
469 // At this point, the buffer may contain output from the test case, which
470 // should be displayed.
471 const std::string_view lines = console_lines (!passed);
472
473 write_buffer_to_file ();
474
475 const bool shows = to_console || !lines.empty ();
476 if (shows)
477 {
478 if (add_empty_line_)
479 {
480 write_console ("\n");
481 }
482
483 // The names of this test case and of the outer ones, whose names
484 // may have been displayed already, by `commit_buffer()` or by a
485 // failed sibling; for a passed test case, only the outer ones.
486 display_open_names (!passed);
487
488 // With verbosity, the full output accumulated in the buffer;
489 // for a failed test case, as it may contain useful information
490 // about the failure.
491 write_console_text (lines);
492 }
493
494 if (passed) [[likely]]
495 {
496 // Successful test case.
497
498 write_line (to_console, "%s%s✓%s %s - passed (%s)\n", indent.c_str (),
499 colours_.pass, colours_.none, test_case.name (),
500 message_totals);
501 }
502 else
503 {
504 // Failed test case.
505
506 write_line (to_console, "%s%s✗%s %s - %sFAILED%s (%s)\n",
507 indent.c_str (), colours_.fail, colours_.none,
508 test_case.name (), colours_.fail, colours_.none,
509 message_totals);
510 }
511
512 if (shows)
513 {
514 add_empty_line_
515 = !passed || route (record_kind::test_case_heading).console;
516 }
517
518 flush_streams ();
519
520 // Clear residual content when less verbose.
521 clear_buffer ();
522
523 if (!open_test_cases_.empty ())
524 {
525 open_test_cases_.pop_back ();
526 }
527 }

737 void
738 reporter_human::abort_session (const reflection::source_location& location,
739 [[maybe_unused]] bool is_requirement)
740 {
741#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
742 trace::printf ("%s\n", __PRETTY_FUNCTION__);
743#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
744
745 const bool to_console = route (record_kind::abort).console;
746
747 // The type of `line()` depends on the toolchain (`uint_least32_t`
748 // for `std::source_location`); widen it implicitly, without a cast
749 // that would be useless on some platforms.
750 const unsigned long line = location.line ();
751 const char* file_name = utility::extract_file_name (location.file_name ());
752
753 const std::string_view lines = console_lines (true);
754
755 write_buffer_to_file ();
756
757 if (to_console)
758 {
759 // Whether headings or results are displayed before the final line.
760 bool has_results = !lines.empty ()
761 || (current_suite_ != nullptr && !suite_displayed_);
762 for (const open_test_case& open : open_test_cases_)
763 {
764 has_results = has_results || !open.name_displayed;
765 }
766
767 if (add_empty_line_)
768 {
769 write_console ("\n");
770 }
771
772 display_open_names (true);
773
774 write_console_text (lines);
775
776 // Separate the final line by an empty line, as the summary, unless
777 // it was already written above.
778 if (has_results || !add_empty_line_)
779 {
780 write_console ("\n");
781 }
782 }
783
784 write_line (to_console, "%s✗%s Test execution aborted at %s:%lu\n",
785 colours_.fail, colours_.none, file_name, line);
786
787 flush_streams ();
788
789 clear_buffer ();
790
791 // Should the abort handler throw, nothing may refer to the objects
792 // that are unwound.
793 open_test_cases_.clear ();
794 current_suite_ = nullptr;
795 }

863 void
864 reporter_human::display_open_names (bool include_last)
865 {
866 if (!suite_displayed_ && current_suite_ != nullptr)
867 {
868 if (!add_empty_line_)
869 {
870 write_console ("\n");
871 }
872
873 write_console ("• %s\n\n", current_suite_->name ());
874
875 suite_displayed_ = true;
876 add_empty_line_ = true;
877 }
878
879 bool is_first = true;
880 const size_t count = include_last || open_test_cases_.empty ()
881 ? open_test_cases_.size ()
882 : open_test_cases_.size () - 1;
883 for (size_t i = 0; i < count; ++i)
884 {
885 open_test_case& open = open_test_cases_[i];
886 if (open.name_displayed)
887 {
888 continue;
889 }
890
891 if (is_first && !add_empty_line_)
892 {
893 write_console ("\n");
894 }
895 is_first = false;
896
897 std::string indent (indent_size_ * open.test->nesting_depth (), ' ');
898 write_console ("%s• %s\n", indent.c_str (), open.test->name ());
899
900 open.name_displayed = true;
901 }
