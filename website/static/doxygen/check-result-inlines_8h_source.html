105 check_result::init (const Expr_T& expr, check_kind kind,
106 const reflection::source_location& location,
107 test_case& test_case, expression_formatter& expression,
108 bool writes_passed)
109 {
110 start (static_cast<bool> (expr), kind, extension::is_op<Expr_T>,

113#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)

115#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED

117 // The same address is used by the destructor, to retrieve the text,
118 // even if other checks used the formatter in the meantime.

125 {
126 expression << expr;
127 }

129 {
130 expression << static_cast<bool> (expr);
131 }

178 template <class T>
179 requires (extension::printable<T> or detail::lazy_message<T>)
180 check_result&&
181 check_result::operator<< (const T& msg) &&
182 {
183 if constexpr (detail::lazy_message<T>)
184 {
185 if (value_)
186 {
187 return std::move (*this);
188 }
189 }
190 else
191 {
192 if (!is_written_)
193 {
194 return std::move (*this);
195 }
196 }
197
198 expression_formatter formatter{ std::move (info_.message) };
199 if constexpr (detail::lazy_message<T>)
200 {
201 formatter << std::invoke (msg);
202 }
203 else
204 {
205 formatter << msg;
206 }
207 info_.message = std::move (formatter.buffer_);
208 return std::move (*this);
209 }
