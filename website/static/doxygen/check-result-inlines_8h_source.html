181 template <class T>
182 requires (extension::printable<T> or detail::lazy_message<T>)
183 auto&
184 check_result::operator<< (const T& msg)
185 {
186 if constexpr (detail::lazy_message<T>)
187 {
188 if (value_)
189 {
190 return *this;
191 }
192 }
193 else
194 {
195 if (!is_written_)
196 {
197 return *this;
198 }
199 }
200
201 expression_formatter formatter{ std::move (info_.message) };
202 if constexpr (detail::lazy_message<T>)
203 {
204 formatter << std::invoke (msg);
205 }
206 else
207 {
208 formatter << msg;
209 }
210 info_.message = std::move (formatter.buffer_);
211 return *this;
212 }
