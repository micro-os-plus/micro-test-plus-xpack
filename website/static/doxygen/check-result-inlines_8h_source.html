100 template <class Expr_T>
101 void
102 check_result::init (const Expr_T& expr, check_kind kind,
103 const reflection::source_location& location,
104 test_case& test_case, expression_formatter& expression,
105 bool writes_passed)
106 {
107 start (static_cast<bool> (expr), kind, extension::is_op<Expr_T>,
108 location, test_case, writes_passed);
109
110#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED)
111 trace::printf ("%s\n", __PRETTY_FUNCTION__);
112#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_CONSTRUCTORS_ENABLED
113
114 // The same address is used by the destructor, to retrieve the text,
115 // even if other checks used the formatter in the meantime.
116 expression.begin_check (this);
117
118 if (is_written_)
119 {
120 if constexpr (extension::is_op<Expr_T>
121 or std::convertible_to<Expr_T, bool>)
122 {
123 expression << expr;
124 }
125 else
126 {
127 expression << static_cast<bool> (expr);
128 }
129 }
130 }

175 template <class T>
176 requires detail::message_argument<T>
177 check_result&&
178 check_result::operator<< (const T& msg) &&
179 {
180 if constexpr (detail::lazy_message<T>)
181 {
182 if (value_)
183 {
184 return std::move (*this);
185 }
186 }
187 else
188 {
189 if (!is_written_)
190 {
191 return std::move (*this);
192 }
193 }
194
195 expression_formatter formatter{ std::move (info_.message) };
196 if constexpr (detail::lazy_message<T>)
197 {
198 formatter << std::invoke (msg);
199 }
200 else
201 {
202 formatter << msg;
203 }
204 info_.message = std::move (formatter.buffer_);
205 return std::move (*this);
206 }
207
208#if !defined(__DOXYGEN__)
214 template <class T>
215 requires (not detail::message_argument<T>)
216 check_result&&
217 check_result::operator<< ([[maybe_unused]] const T& msg) &&
218 {
219 if constexpr (std::is_class_v<T> and std::invocable<const T&>)
220 {
221 static_assert (
222 not std::is_same_v<T, T>,
223 "a message that is a callable object must be callable "
224 "without arguments and return a value that can be "
225 "written, not void");
226 }
227 else
228 {

231 "written: define an operator<< (mt::"
232 "expression_formatter&, const T&) in the namespace "
233 "of the type, or pass a callable object that returns "
234 "such a value");
235 }
236 return std::move (*this);
