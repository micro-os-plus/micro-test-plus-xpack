651 template <class Lhs_T, class Rhs_T, class Epsilon_T>
652 constexpr bool
653 differ_by_at_most (Lhs_T lhs, Rhs_T rhs, Epsilon_T epsilon)
654 {
655 if constexpr (std::is_integral_v<Lhs_T> and std::is_integral_v<Rhs_T>
656 and std::is_integral_v<Epsilon_T>)
657 {
658 bool overflow = false;
659 const auto distance
660 = [&overflow] (auto larger,
661 auto smaller) constexpr -> std::uintmax_t
662 {
663 if constexpr (std::is_signed_v<decltype (smaller)>)
664 {
665 if (smaller < 0 and std::cmp_greater_equal (larger, 0))
666 {
667 // The magnitude of the negative value, without
668 // negating the minimum of its type.
669 const auto magnitude
670 = static_cast<std::uintmax_t> (-(smaller + 1)) + 1u;
671 const auto upper = static_cast<std::uintmax_t> (larger);
672 if (upper > std::numeric_limits<std::uintmax_t>::max ()
673 - magnitude)
674 {
675 overflow = true;
676 return std::numeric_limits<std::uintmax_t>::max ();
677 }
678 return upper + magnitude;
679 }
680 }
681 return static_cast<std::uintmax_t> (larger)
682 - static_cast<std::uintmax_t> (smaller);
683 };
684
685 const std::uintmax_t difference = std::cmp_less (lhs, rhs)
686 ? distance (rhs, lhs)
687 : distance (lhs, rhs);
688 return !overflow && std::cmp_less_equal (difference, epsilon);
689 }
690 else
691 {
692 using float_type = std::common_type_t<Lhs_T, Rhs_T, Epsilon_T>;
693 const auto left = static_cast<float_type> (lhs);
694 const auto right = static_cast<float_type> (rhs);
695
696#if defined(__GNUC__)
697#pragma GCC diagnostic push
698#pragma GCC diagnostic ignored "-Wfloat-equal"
699#endif // defined(__GNUC__)
700 // The exact comparison is deliberate, for the infinities.
701 const float_type distance
702 = (left == right) ? float_type{} : math::abs (left - right);
703#if defined(__GNUC__)
704#pragma GCC diagnostic pop
705#endif // defined(__GNUC__)
706
707 return distance <= static_cast<float_type> (epsilon);
708 }
709 }

1039 template <class T>
1040 requires std::is_arithmetic_v<T>
1041 void
1042 append_number (std::string& buffer, const T v)
1043 {
1044 if constexpr (std::is_same_v<T, long double>)
1045 {
1046#if defined(_WIN32) \
1047 || (defined(__SIZEOF_LONG_DOUBLE__) \
1048 && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
1049 // On Windows (all toolchains: MinGW, Clang, MSVC), the C runtime
1050 // does not handle the %Lg printf specifier correctly for 80-bit
1051 // long double, producing garbage output. On platforms where long
1052 // double has the same width as double (e.g. 32-bit Arm), the cast
1053 // is lossless. In both cases, cast to double and use
1054 // std::to_chars.
1055 char buf[32];
1056 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
1057 static_cast<double> (v));
1058 if (ec == std::errc{})
1059 buffer.append (buf, ptr);
1060#else
1061 // On other platforms (80-bit extended precision on x86-64,
1062 // 128-bit quad precision on AArch64 and RISC-V), std::to_chars
1063 // for long double may be unavailable (e.g., with lld).
1064 // Use snprintf as a portable fallback; %Lg is supported by the C
1065 // libraries used on these platforms (e.g. glibc, the macOS libc,
1066 // and newlib, the latter with the precision of double on AArch64
1067 // and RISC-V). Use the smallest precision that reads back as the
1068 // same value; the buffer fits the 36 digits of the 128-bit
1069 // format, with the sign, the point, and the exponent.
1070 char buf[48];
1071 const int saved_errno = errno;
1072 // The subnormal values have fewer significant digits.
1073 const int min_precision
1074 = (v > -LDBL_MIN && v < LDBL_MIN) ? 1 : LDBL_DIG;
1075 bool is_exact = false;
1076 for (int precision = min_precision; precision <= LDBL_DECIMAL_DIG;
1077 ++precision)
1078 {
1079 snprintf (buf, sizeof (buf), "%.*Lg", precision, v);
1080 // A NaN is never equal to itself, and has no digits.
1081 if (v != v || strtold (buf, nullptr) == v)
1082 {
1083 is_exact = true;
1084 break;
1085 }
1086 }
1087 errno = saved_errno;
1088 if (is_exact)
1089 {
1090 buffer.append (buf);
1091 }
1092 else
1093 {
1094 // The C library formats or reads long double with the
1095 // precision of double (e.g. newlib on AArch64 and RISC-V),
1096 // and its long text would only show the error of the
