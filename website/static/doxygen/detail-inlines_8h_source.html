556 template <comparison Op, class Lhs_T, class Rhs_T>
557 constexpr void
558 reject_comparison (void) noexcept
559 {
560 if constexpr (not extension::printable<Lhs_T>
561 or not extension::printable<Rhs_T>)
562 {
563 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
564 "an operand of the comparator cannot be written in "
565 "the report; define an operator<< (mt::"
566 "expression_formatter&, const T&) in the namespace "
567 "of its type");
568 }
569 else if constexpr (Op == comparison::equal)
570 {
571 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
572 "the operands of eq () cannot be compared with ==");
573 }
574 else if constexpr (Op == comparison::not_equal)
575 {
576 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
577 "the operands of ne () cannot be compared with !=");
578 }
579 else if constexpr (Op == comparison::greater)
580 {
581 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
582 "the operands of gt () cannot be compared with >");
583 }
584 else if constexpr (Op == comparison::greater_equal)
585 {
586 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
587 "the operands of ge () cannot be compared with >=");
588 }
589 else if constexpr (Op == comparison::less)
590 {
591 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
592 "the operands of lt () cannot be compared with <");
593 }
594 else
595 {
596 static_assert (not std::is_same_v<Lhs_T, Lhs_T>,
597 "the operands of le () cannot be compared with <=");

939 template <class Lhs_T, class Rhs_T, class Epsilon_T>
940 constexpr bool
941 differ_by_at_most (Lhs_T lhs, Rhs_T rhs, Epsilon_T epsilon)
942 {
943 if constexpr (std::is_integral_v<Lhs_T> and std::is_integral_v<Rhs_T>
944 and std::is_integral_v<Epsilon_T>)
945 {
946 bool overflow = false;
947 const auto distance
948 = [&overflow] (auto larger,
949 auto smaller) constexpr -> std::uintmax_t
950 {
951 if constexpr (std::is_signed_v<decltype (smaller)>)
952 {
953 if (smaller < 0 and std::cmp_greater_equal (larger, 0))
954 {
955 // The magnitude of the negative value, without
956 // negating the minimum of its type.
957 const auto magnitude
958 = static_cast<std::uintmax_t> (-(smaller + 1)) + 1u;
959 const auto upper = static_cast<std::uintmax_t> (larger);
960 if (upper > std::numeric_limits<std::uintmax_t>::max ()
961 - magnitude)
962 {
963 overflow = true;
964 return std::numeric_limits<std::uintmax_t>::max ();
965 }
966 return upper + magnitude;
967 }
968 }
969 return static_cast<std::uintmax_t> (larger)
970 - static_cast<std::uintmax_t> (smaller);
971 };
972
973 const std::uintmax_t difference = std::cmp_less (lhs, rhs)
974 ? distance (rhs, lhs)
975 : distance (lhs, rhs);
976 return !overflow && std::cmp_less_equal (difference, epsilon);
977 }
978 else
979 {
980 using float_type = std::common_type_t<Lhs_T, Rhs_T, Epsilon_T>;
981 const auto left = static_cast<float_type> (lhs);
982 const auto right = static_cast<float_type> (rhs);
983
984#if defined(__GNUC__)
985#pragma GCC diagnostic push
986#pragma GCC diagnostic ignored "-Wfloat-equal"
987#endif // defined(__GNUC__)
988 // The exact comparison is deliberate, for the infinities.
989 const float_type distance
990 = (left == right) ? float_type{} : math::abs (left - right);
991#if defined(__GNUC__)
992#pragma GCC diagnostic pop
993#endif // defined(__GNUC__)
994
995 return distance <= static_cast<float_type> (epsilon);
996 }
997 }

1346 template <class T>
1347 requires std::is_arithmetic_v<T>
1348 void
1349 append_number (std::string& buffer, const T v)
1350 {
1351 if constexpr (std::is_same_v<T, long double>)
1352 {
1353#if defined(_WIN32) \
1354 || (defined(__SIZEOF_LONG_DOUBLE__) \
1355 && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
1356 // On Windows (all toolchains: MinGW, Clang, MSVC), the C runtime
1357 // does not handle the %Lg printf specifier correctly for 80-bit
1358 // long double, producing garbage output. On platforms where long
1359 // double has the same width as double (e.g. 32-bit Arm), the cast
1360 // is lossless. In both cases, cast to double and use
1361 // std::to_chars.
1362 char buf[32];
1363 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
1364 static_cast<double> (v));
1365 if (ec == std::errc{})
1366 buffer.append (buf, ptr);
1367#else
1368 // On other platforms (80-bit extended precision on x86-64,
1369 // 128-bit quad precision on AArch64 and RISC-V), std::to_chars
1370 // for long double may be unavailable (e.g., with lld).
1371 // Use snprintf as a portable fallback; %Lg is supported by the C
1372 // libraries used on these platforms (e.g. glibc, the macOS libc,
1373 // and newlib, the latter with the precision of double on AArch64
1374 // and RISC-V). Use the smallest precision that reads back as the
1375 // same value; the buffer fits the 36 digits of the 128-bit
1376 // format, with the sign, the point, and the exponent.
1377 char buf[48];
1378 const int saved_errno = errno;
1379 // The subnormal values have fewer significant digits.
1380 const int min_precision
1381 = (v > -LDBL_MIN && v < LDBL_MIN) ? 1 : LDBL_DIG;
1382 bool is_exact = false;
1383 for (int precision = min_precision; precision <= LDBL_DECIMAL_DIG;
1384 ++precision)
1385 {
1386 snprintf (buf, sizeof (buf), "%.*Lg", precision, v);
1387 // A NaN is never equal to itself, and has no digits.
1388 if (v != v || strtold (buf, nullptr) == v)
1389 {
1390 is_exact = true;
1391 break;
1392 }
1393 }
1394 errno = saved_errno;
1395 if (is_exact)
1396 {
1397 buffer.append (buf);
1398 }
1399 else
1400 {
1401 // The C library formats or reads long double with the
1402 // precision of double (e.g. newlib on AArch64 and RISC-V),
1403 // and its long text would only show the error of the
1404 // conversion; use the shortest form of the double value,
1405 // as on the platforms above.
1406 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
1407 static_cast<double> (v));
1408 if (ec == std::errc{})
1409 buffer.append (buf, ptr);
1410 }
1411#endif // long double formatting
1412 }
1413 else
1414 {
1415 char buf[32];
1416 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf), v);
1417 if (ec == std::errc{})
1418 buffer.append (buf, ptr);
1419 }
1420 }

1436 * implicitly to `const volatile void*`, which accepts any
1437 * qualification, and the qualifiers are then removed with
1438 * `const_cast`; a `reinterpret_cast` to `void*` would not compile for
1439 * pointers to `const` objects. The object is not accessed. Function
1440 * pointers, which do not convert to `void*` implicitly, use
1441 * `reinterpret_cast`.
1442 */

1461 p = const_cast<void*> (cvp);
1462 }

1469#endif // defined(__clang__)
1470#endif // defined(__GNUC__)

1473 snprintf (buf, sizeof (buf), "%p", p);

1714 template <class T>
1715 constexpr auto
1716 as_operand (T&& t) noexcept -> decltype (auto)
1717 {
1718 using type = std::remove_cvref_t<T>;
1719 // The cv-qualifiers of an array are those of its elements, which
1720 // std::remove_cvref_t removes; keep volatile for the element type.
1721 using element_type = std::remove_const_t<
1722 std::remove_extent_t<std::remove_reference_t<T>>>;
1723
1724 if constexpr (std::is_array_v<type> && character<element_type>
1725 && not std::is_volatile_v<element_type>)
1726 {
1727 return basic_c_string<std::remove_cv_t<std::remove_extent_t<type>>>{
1728 t, std::extent_v<type>
1729 };
1730 }
1731 else if constexpr (character_pointer<type>)
1732 {
1733 return basic_c_string<std::remove_cv_t<std::remove_pointer_t<type>>>{
1734 t
1735 };
1736 }
1737 else if constexpr (std::is_array_v<type>)
1738 {
1739 // Other arrays (e.g. byte buffers) are compared by content.
1740 return array_operand<element_type>{
1741 static_cast<const element_type*> (t), std::extent_v<type>
1742 };
1743 }
1744 else if constexpr (std::is_function_v<type>)
1745 {
1746 // Functions are compared by address, as function pointers.
1747 return &t;
1748 }
1749 else
1750 {
1751 return std::forward<T> (t);
1752 }
