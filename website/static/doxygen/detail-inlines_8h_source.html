191 template <class Lhs_T, class Rhs_T>
192 constexpr eq_<Lhs_T, Rhs_T>::eq_ (const Lhs_T& lhs, const Rhs_T& rhs)
193 : binary_op_<Lhs_T, Rhs_T>{ lhs, rhs, [&]
194 {
195 // This lambda is called in the constructor to evaluate the
196 // comparison. Its result is implicitly converted to bool via
197 // the operator bool() of whatever type the branch returns.
198 // This is intentional: all result types (integral_constant,
199 // comparator objects, plain bool) define operator bool().
200 using std::operator==;
201 using std::operator<;
202
203#if defined(__GNUC__)
204#pragma GCC diagnostic push
205
206#pragma GCC diagnostic ignored "-Wfloat-equal"
207#pragma GCC diagnostic ignored "-Wconversion"
208#pragma GCC diagnostic ignored "-Wdouble-promotion"
209#pragma GCC diagnostic ignored "-Wsign-compare"
210#if defined(__clang__)
211#pragma clang diagnostic ignored "-Wimplicit-int-float-conversion"
212#pragma clang diagnostic ignored "-Wpedantic"
213#endif // defined(__clang__)
214#endif // defined(__GNUC__)
215
216 if constexpr (type_traits::has_value<Lhs_T>
217 and type_traits::has_value<Rhs_T>)
218 {
219 // If both types have values (like numeric constants),
220 // compare them directly.
221 return Lhs_T::value == Rhs_T::value;
222 }
223 else if constexpr (type_traits::has_epsilon<Lhs_T>
224 and type_traits::has_epsilon<Rhs_T>)
225 {
226 // If both values have precision, compare them using
227 // the smallest precision.
228 return math::abs (get (lhs) - get (rhs))
229 < math::min_value (lhs.epsilon, rhs.epsilon);
230 }
231 else if constexpr (type_traits::has_epsilon<Lhs_T>)
232 {
233 // If only the left operand has precision, use it.
234 return math::abs (get (lhs) - get (rhs)) < lhs.epsilon;
235 }
236 else if constexpr (type_traits::has_epsilon<Rhs_T>)
237 {
238 // If only the right operand has precision, use it.
239 return math::abs (get (lhs) - get (rhs)) < rhs.epsilon;
240 }
241 else
242 {
243 // Call the generic getters, which might
244 // either call the type get() or return the value.
245 return get (lhs) == get (rhs);
246 }
247
248#if defined(__GNUC__)
249#pragma GCC diagnostic pop
250#endif // defined(__GNUC__)
251 }() }
252 {
253 }

676 template <class T>
677 requires std::is_arithmetic_v<T>
678 void
679 append_number_ (std::string& buffer, const T v)
680 {
681 char buf[32];
682 if constexpr (std::is_same_v<T, long double>)
683 {
684#if defined(_WIN32) \
685 || (defined(__SIZEOF_LONG_DOUBLE__) \
686 && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
687 // On Windows (all toolchains: MinGW, Clang, MSVC), the C runtime
688 // does not handle the %Lg printf specifier correctly for 80-bit
689 // long double, producing garbage output. On platforms where long
690 // double has the same width as double (e.g. 32-bit Arm), the cast
691 // is lossless. In both cases, cast to double and use
692 // std::to_chars.
693 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
694 static_cast<double> (v));
695 if (ec == std::errc{})
696 buffer.append (buf, ptr);
697#else
698 // On other platforms (80-bit extended precision on x86-64,
699 // 128-bit quad precision on AArch64 and RISC-V), std::to_chars
700 // for long double may be unavailable (e.g., with lld).
701 // Use snprintf as a portable fallback; %Lg is supported correctly
702 // by the C libraries used on these platforms (e.g. glibc, the
703 // macOS libc, and newlib).
704 snprintf (buf, sizeof (buf), "%Lg", v);
705 buffer.append (buf);
706#endif // defined(_WIN32) || (defined(__SIZEOF_LONG_DOUBLE__) && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
707 }
708 else
709 {
710 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf), v);
711 if (ec == std::errc{})
712 buffer.append (buf, ptr);
713 }
714 }
