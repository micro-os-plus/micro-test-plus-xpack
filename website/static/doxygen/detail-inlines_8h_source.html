205 template <class Lhs_T, class Rhs_T>
206 requires (extension::has_epsilon<Lhs_T> or extension::has_epsilon<Rhs_T>)
207 constexpr bool
208 within_tolerance (const Lhs_T& lhs, const Rhs_T& rhs)
209 {
210#if defined(__GNUC__)
211#pragma GCC diagnostic push
212#pragma GCC diagnostic ignored "-Wfloat-equal"
213#pragma GCC diagnostic ignored "-Wconversion"
214#pragma GCC diagnostic ignored "-Wdouble-promotion"
215#pragma GCC diagnostic ignored "-Wsign-compare"
216#if defined(__clang__)
217#pragma clang diagnostic ignored "-Wimplicit-int-float-conversion"
218#pragma clang diagnostic ignored "-Wpedantic"
219#endif // defined(__clang__)
220#endif // defined(__GNUC__)
221 const auto difference
222 = math::abs (detail::unwrap (lhs) - detail::unwrap (rhs));
223 if constexpr (extension::has_epsilon<Lhs_T>
224 and extension::has_epsilon<Rhs_T>)
225 {
226 // If both values have precision, use the smallest one.
227 return difference < math::min_value (lhs.epsilon, rhs.epsilon);
228 }
229 else if constexpr (extension::has_epsilon<Lhs_T>)
230 {
231 return difference < lhs.epsilon;
232 }
233 else
234 {
235 return difference < rhs.epsilon;
236 }
237#if defined(__GNUC__)
238#pragma GCC diagnostic pop
239#endif // defined(__GNUC__)
240 }

369 template <class L, class R>
370 constexpr compare_op<Lhs_T, Rhs_T, Compare_T>::compare_op (L&& lhs,
371 R&& rhs,
372 Compare_T)
373 : binary_op<Lhs_T, Rhs_T>{ std::forward<L> (lhs),
374 std::forward<R> (rhs), [&] () -> bool
375 {
376 // The result is computed before the arguments are moved, since
377 // the base constructor is entered only after all its arguments
378 // are evaluated.
379 using lhs_type = std::remove_cvref_t<L>;
380 using rhs_type = std::remove_cvref_t<R>;
381
382 if constexpr (detail::has_value<lhs_type>
383 and detail::has_value<rhs_type>)
384 {

390 Compare_T::within_tolerance_result;

394 ? Compare_T::within_tolerance_result
395 : Compare_T::apply (detail::unwrap (lhs),
396 detail::unwrap (rhs));
397 }
398 else
399 {
400 return Compare_T::apply (detail::unwrap (lhs),
401 detail::unwrap (rhs));

425 template <class Lhs_T, class Rhs_T, class Epsilon_T>
426 constexpr bool
427 differ_by_less_than (Lhs_T lhs, Rhs_T rhs, Epsilon_T epsilon)
428 {
429 if constexpr (std::is_integral_v<Lhs_T> and std::is_integral_v<Rhs_T>
430 and std::is_integral_v<Epsilon_T>)
431 {
432 const auto distance
433 = [] (auto larger, auto smaller) constexpr -> std::uintmax_t
434 {
435 if constexpr (std::is_signed_v<decltype (smaller)>)
436 {
437 if (smaller < 0 and std::cmp_greater_equal (larger, 0))
438 {
439 // The magnitude of the negative value, without
440 // negating the minimum of its type.
441 const auto magnitude
442 = static_cast<std::uintmax_t> (-(smaller + 1)) + 1u;
443 const auto upper = static_cast<std::uintmax_t> (larger);
444 if (upper > std::numeric_limits<std::uintmax_t>::max ()
445 - magnitude)
446 {
447 return std::numeric_limits<std::uintmax_t>::max ();
448 }
449 return upper + magnitude;
450 }
451 }
452 return static_cast<std::uintmax_t> (larger)
453 - static_cast<std::uintmax_t> (smaller);
454 };
455
456 const std::uintmax_t difference = std::cmp_less (lhs, rhs)
457 ? distance (rhs, lhs)
458 : distance (lhs, rhs);
459 return std::cmp_less (difference, epsilon);
460 }
461 else
462 {
463 using float_type = std::common_type_t<Lhs_T, Rhs_T, Epsilon_T>;
464 return math::abs (static_cast<float_type> (lhs)
465 - static_cast<float_type> (rhs))
466 < static_cast<float_type> (epsilon);
467 }
468 }

725 template <class T>
726 requires std::is_arithmetic_v<T>
727 void
728 append_number (std::string& buffer, const T v)
729 {
730 if constexpr (std::is_same_v<T, long double>)
731 {
732#if defined(_WIN32) \
733 || (defined(__SIZEOF_LONG_DOUBLE__) \
734 && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
735 // On Windows (all toolchains: MinGW, Clang, MSVC), the C runtime
736 // does not handle the %Lg printf specifier correctly for 80-bit
737 // long double, producing garbage output. On platforms where long
738 // double has the same width as double (e.g. 32-bit Arm), the cast
739 // is lossless. In both cases, cast to double and use
740 // std::to_chars.
741 char buf[32];
742 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
743 static_cast<double> (v));
744 if (ec == std::errc{})
745 buffer.append (buf, ptr);
746#else
747 // On other platforms (80-bit extended precision on x86-64,
748 // 128-bit quad precision on AArch64 and RISC-V), std::to_chars
749 // for long double may be unavailable (e.g., with lld).
750 // Use snprintf as a portable fallback; %Lg is supported by the C
751 // libraries used on these platforms (e.g. glibc, the macOS libc,
752 // and newlib, the latter with the precision of double on AArch64
753 // and RISC-V). Use the smallest precision that reads back as the
754 // same value; the buffer fits the 36 digits of the 128-bit
755 // format, with the sign, the point, and the exponent.
756 char buf[48];
757 const int saved_errno = errno;
758 // The subnormal values have fewer significant digits.
759 const int min_precision
760 = (v > -LDBL_MIN && v < LDBL_MIN) ? 1 : LDBL_DIG;
761 bool is_exact = false;
762 for (int precision = min_precision; precision <= LDBL_DECIMAL_DIG;
763 ++precision)
764 {
765 snprintf (buf, sizeof (buf), "%.*Lg", precision, v);
766 // A NaN is never equal to itself, and has no digits.
767 if (v != v || strtold (buf, nullptr) == v)
768 {
769 is_exact = true;
770 break;
771 }
772 }
773 errno = saved_errno;
774 if (is_exact)
775 {
776 buffer.append (buf);
777 }
778 else
779 {
780 // The C library formats or reads long double with the
781 // precision of double (e.g. newlib on AArch64 and RISC-V),
782 // and its long text would only show the error of the
783 // conversion; use the shortest form of the double value,
784 // as on the platforms above.
785 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
786 static_cast<double> (v));
787 if (ec == std::errc{})
788 buffer.append (buf, ptr);
789 }
790#endif // defined(_WIN32) || (defined(__SIZEOF_LONG_DOUBLE__) && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
791 }
792 else
793 {
794 char buf[32];
795 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf), v);
796 if (ec == std::errc{})
797 buffer.append (buf, ptr);
798 }
799 }

822 template <class T>
823 void
824 append_pointer (std::string& buffer, T* v)
825 {
826 if (v == nullptr)
827 {
828 buffer.append ("0x0");
829 return;
830 }
831
832 void* p;
833 if constexpr (std::is_function_v<T>)
834 {
835 p = reinterpret_cast<void*> (v);
836 }
837 else
838 {
839 const volatile void* cvp = v;
840 p = const_cast<void*> (cvp);

843#if defined(__GNUC__)
844#pragma GCC diagnostic push
845
846#if defined(__clang__)
847#pragma clang diagnostic ignored "-Wunsafe-buffer-usage-in-libc-call"
848#endif // defined(__clang__)
849#endif // defined(__GNUC__)

852 snprintf (buf, sizeof (buf), "%p", p);
853 buffer.append (buf);

856#pragma GCC diagnostic pop
857#endif // defined(__GNUC__)

983 template <class T>
984 constexpr auto
985 as_operand (T&& t) noexcept -> decltype (auto)
986 {
987 using type = std::remove_cvref_t<T>;
988
989 if constexpr (std::is_array_v<type>
990 && std::is_same_v<
991 std::remove_cv_t<std::remove_extent_t<type>>, char>)
992 {
993 return c_string{ t, std::extent_v<type> };
994 }
995 else if constexpr (std::is_pointer_v<type>
996 && std::is_same_v<
997 std::remove_cv_t<std::remove_pointer_t<type>>,
998 char>)
999 {
1000 return c_string{ t };
1001 }
1002 else if constexpr (std::is_array_v<type>)
1003 {
1004 // Other arrays (e.g. byte buffers) are compared by address, as
1005 // pointers to their first element.
1006 return static_cast<const std::remove_extent_t<type>*> (t);
1007 }
1008 else if constexpr (std::is_function_v<type>)
1009 {
1010 // Functions are compared by address, as function pointers.
1011 return &t;
1012 }
1013 else
1014 {
1015 return std::forward<T> (t);
1016 }
1017 }
