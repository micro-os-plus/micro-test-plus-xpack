206 template <class Lhs_T, class Rhs_T>
207 requires (extension::has_epsilon<Lhs_T> or extension::has_epsilon<Rhs_T>)
208 constexpr bool
209 within_tolerance (const Lhs_T& lhs, const Rhs_T& rhs)
210 {
211#if defined(__GNUC__)
212#pragma GCC diagnostic push
213#pragma GCC diagnostic ignored "-Wfloat-equal"
214#pragma GCC diagnostic ignored "-Wconversion"
215#pragma GCC diagnostic ignored "-Wdouble-promotion"
216#pragma GCC diagnostic ignored "-Wsign-compare"
217#if defined(__clang__)
218#pragma clang diagnostic ignored "-Wimplicit-int-float-conversion"
219#pragma clang diagnostic ignored "-Wpedantic"
220#endif // defined(__clang__)
221#endif // defined(__GNUC__)
222 const auto difference
223 = math::abs (detail::unwrap (lhs) - detail::unwrap (rhs));
224 if constexpr (extension::has_epsilon<Lhs_T>
225 and extension::has_epsilon<Rhs_T>)
226 {
227 // If both values have precision, use the smallest one.
228 return difference < math::min_value (lhs.epsilon, rhs.epsilon);
229 }
230 else if constexpr (extension::has_epsilon<Lhs_T>)
231 {
232 return difference < lhs.epsilon;
233 }
234 else
235 {
236 return difference < rhs.epsilon;
237 }
238#if defined(__GNUC__)
239#pragma GCC diagnostic pop
240#endif // defined(__GNUC__)
241 }

369 template <class Lhs_T, class Rhs_T, class Compare_T>
370 template <class L, class R>
371 constexpr compare_op<Lhs_T, Rhs_T, Compare_T>::compare_op (L&& lhs,
372 R&& rhs,
373 Compare_T)
374 : binary_op<Lhs_T, Rhs_T>{ std::forward<L> (lhs),
375 std::forward<R> (rhs), [&] () -> bool
376 {
377 // The result is computed before the arguments are moved, since
378 // the base constructor is entered only after all its arguments
379 // are evaluated.
380 using lhs_type = std::remove_cvref_t<L>;
381 using rhs_type = std::remove_cvref_t<R>;
382
383 if constexpr (detail::has_value<lhs_type>
384 and detail::has_value<rhs_type>)
385 {
386 return Compare_T::apply (lhs_type::value, rhs_type::value);
387 }
388 else if constexpr ((extension::has_epsilon<lhs_type>
389 or extension::has_epsilon<rhs_type>)
390 and requires {
391 Compare_T::within_tolerance_result;
392 })
393 {
394 return detail::within_tolerance (lhs, rhs)
395 ? Compare_T::within_tolerance_result
396 : Compare_T::apply (detail::unwrap (lhs),
397 detail::unwrap (rhs));
398 }
399 else
400 {
401 return Compare_T::apply (detail::unwrap (lhs),
402 detail::unwrap (rhs));
403 }
404 }() }

426 template <class Lhs_T, class Rhs_T, class Epsilon_T>
427 constexpr bool
428 differ_by_less_than (Lhs_T lhs, Rhs_T rhs, Epsilon_T epsilon)
429 {
430 if constexpr (std::is_integral_v<Lhs_T> and std::is_integral_v<Rhs_T>
431 and std::is_integral_v<Epsilon_T>)
432 {
433 const auto distance
434 = [] (auto larger, auto smaller) constexpr -> std::uintmax_t
435 {
436 if constexpr (std::is_signed_v<decltype (smaller)>)
437 {
438 if (smaller < 0 and std::cmp_greater_equal (larger, 0))
439 {
440 // The magnitude of the negative value, without
441 // negating the minimum of its type.
442 const auto magnitude
443 = static_cast<std::uintmax_t> (-(smaller + 1)) + 1u;
444 const auto upper = static_cast<std::uintmax_t> (larger);
445 if (upper > std::numeric_limits<std::uintmax_t>::max ()
446 - magnitude)
447 {
448 return std::numeric_limits<std::uintmax_t>::max ();
449 }
450 return upper + magnitude;
451 }
452 }
453 return static_cast<std::uintmax_t> (larger)
454 - static_cast<std::uintmax_t> (smaller);
455 };
456
457 const std::uintmax_t difference = std::cmp_less (lhs, rhs)
458 ? distance (rhs, lhs)
459 : distance (lhs, rhs);
460 return std::cmp_less (difference, epsilon);
461 }
462 else
463 {
464 using float_type = std::common_type_t<Lhs_T, Rhs_T, Epsilon_T>;
465 return math::abs (static_cast<float_type> (lhs)
466 - static_cast<float_type> (rhs))
467 < static_cast<float_type> (epsilon);
468 }
469 }

726 template <class T>
727 requires std::is_arithmetic_v<T>
728 void
729 append_number (std::string& buffer, const T v)
730 {
731 if constexpr (std::is_same_v<T, long double>)
732 {
733#if defined(_WIN32) \
734 || (defined(__SIZEOF_LONG_DOUBLE__) \
735 && __SIZEOF_LONG_DOUBLE__ == __SIZEOF_DOUBLE__)
736 // On Windows (all toolchains: MinGW, Clang, MSVC), the C runtime
737 // does not handle the %Lg printf specifier correctly for 80-bit
738 // long double, producing garbage output. On platforms where long
739 // double has the same width as double (e.g. 32-bit Arm), the cast
740 // is lossless. In both cases, cast to double and use
741 // std::to_chars.
742 char buf[32];
743 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
744 static_cast<double> (v));
745 if (ec == std::errc{})
746 buffer.append (buf, ptr);
747#else
748 // On other platforms (80-bit extended precision on x86-64,
749 // 128-bit quad precision on AArch64 and RISC-V), std::to_chars
750 // for long double may be unavailable (e.g., with lld).
751 // Use snprintf as a portable fallback; %Lg is supported by the C
752 // libraries used on these platforms (e.g. glibc, the macOS libc,
753 // and newlib, the latter with the precision of double on AArch64
754 // and RISC-V). Use the smallest precision that reads back as the
755 // same value; the buffer fits the 36 digits of the 128-bit
756 // format, with the sign, the point, and the exponent.
757 char buf[48];
758 const int saved_errno = errno;
759 // The subnormal values have fewer significant digits.
760 const int min_precision
761 = (v > -LDBL_MIN && v < LDBL_MIN) ? 1 : LDBL_DIG;
762 bool is_exact = false;
763 for (int precision = min_precision; precision <= LDBL_DECIMAL_DIG;
764 ++precision)
765 {
766 snprintf (buf, sizeof (buf), "%.*Lg", precision, v);
767 // A NaN is never equal to itself, and has no digits.
768 if (v != v || strtold (buf, nullptr) == v)
769 {
770 is_exact = true;
771 break;
772 }
773 }
774 errno = saved_errno;
775 if (is_exact)
776 {
777 buffer.append (buf);
778 }
779 else
780 {
781 // The C library formats or reads long double with the
782 // precision of double (e.g. newlib on AArch64 and RISC-V),
783 // and its long text would only show the error of the
784 // conversion; use the shortest form of the double value,
785 // as on the platforms above.
786 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf),
787 static_cast<double> (v));
788 if (ec == std::errc{})
789 buffer.append (buf, ptr);
790 }
791#endif // long double formatting
792 }
793 else
794 {
795 char buf[32];
796 const auto [ptr, ec] = std::to_chars (buf, buf + sizeof (buf), v);
797 if (ec == std::errc{})
798 buffer.append (buf, ptr);
799 }
800 }

826 {

828 {
829 buffer.append ("0x0");

1063 template <class T>
1064 constexpr auto
1065 as_operand (T&& t) noexcept -> decltype (auto)
1066 {
1067 using type = std::remove_cvref_t<T>;
1068
1069 if constexpr (std::is_array_v<type>
1070 && std::is_same_v<
1071 std::remove_cv_t<std::remove_extent_t<type>>, char>)
1072 {
1073 return c_string{ t, std::extent_v<type> };
1074 }
1075 else if constexpr (std::is_pointer_v<type>
1076 && std::is_same_v<
1077 std::remove_cv_t<std::remove_pointer_t<type>>,
1078 char>)
1079 {
1080 return c_string{ t };
1081 }
1082 else if constexpr (std::is_array_v<type>)
1083 {
1084 // Other arrays (e.g. byte buffers) are compared by content.
1085 return array_operand<std::remove_extent_t<type>>{
1086 static_cast<const std::remove_extent_t<type>*> (t),
1087 std::extent_v<type>
1088 };
1089 }
1090 else if constexpr (std::is_function_v<type>)
1091 {
1092 // Functions are compared by address, as function pointers.
1093 return &t;
1094 }
1095 else
1096 {
1097 return std::forward<T> (t);
1098 }
1099 }
