103 std::string
104 tap_text (std::string_view text, bool escape)
105 {
106 std::string result;
107 result.reserve (text.size ());
108
109 for (size_t i = 0; i < text.size (); ++i)
110 {
111 const char c = text[i];
112 if (c == '\n' || c == '\r')
113 {
114 // The `\r\n` pair is a single line break.
115 if (c == '\r' && i + 1 < text.size () && text[i + 1] == '\n')
116 {
117 ++i;
118 }
119 result.push_back (' ');
120 }
121 else if (escape && (c == '\\' || c == '#'))
122 {
123 result.push_back ('\\');
124 result.push_back (c);
125 }
126 else
127 {
128 result.push_back (c);
129 }
130 }
131
132 return result;
133 }

153 std::string
154 yaml_double_quoted (std::string_view text)
155 {
156 constexpr std::string_view digits{ "0123456789ABCDEF" };
157
158 std::string result;
159 result.reserve (text.size () + 2);
160
161 result.push_back ('"');
162 for (const char c : text)
163 {
164 const auto u = static_cast<unsigned char> (c);
165 switch (c)
166 {
167 case '\\':
168 result.append ("\\\\");
169 break;
170 case '"':
171 result.append ("\\\"");
172 break;
173 case '\0':
174 result.append ("\\0");
175 break;
176 case '\a':
177 result.append ("\\a");
178 break;
179 case '\b':
180 result.append ("\\b");
181 break;
182 case '\t':
183 result.append ("\\t");
184 break;
185 case '\n':
186 result.append ("\\n");
187 break;
188 case '\v':
189 result.append ("\\v");
190 break;
191 case '\f':
192 result.append ("\\f");
193 break;
194 case '\r':
195 result.append ("\\r");
196 break;
197 case '\x1B':
198 result.append ("\\e");
199 break;
200 default:
201 if (u < 0x20 || u == 0x7F)
202 {
203 result.append ("\\x");
204 result.push_back (digits[static_cast<std::size_t> (u >> 4)]);
205 result.push_back (
206 digits[static_cast<std::size_t> (u & 0x0F)]);
207 }
208 else
209 {
210 result.push_back (c);
211 }
212 break;
213 }
214 }
215 result.push_back ('"');
216
217 return result;
218 }

385 "%s '%s' +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
386 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
387 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE

418 if (to_console)

448 write_line (to_console,

450 "\n%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
451 " - %s # { %s, %s%s }\n",

454 message_time);
455
456 if (verbosity_ == verbosity::quiet)
457 {
458 // Only the test point of the suite, counted by the session plan.
459 write_console ("%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
460 " - %s # { %s, %s%s }\n",
461 result, suite.own_index_, name.c_str (), status,
462 message_totals, message_time);
463 }
464
465 flush_streams ();
466
467 clear_buffer ();
468
469 add_empty_line_ = true;

592 void
593 reporter_tap::end_test_case (const test_case& test_case)
594 {
595#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
596 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
597 " +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
598 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
599 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
600 ", xs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
601 " | cs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
602 __PRETTY_FUNCTION__, test_case.name (),
603 test_case.nesting_depth (),
604 test_case.totals ().passed_checks (),
605 test_case.totals ().failed_checks (),
606 test_case.totals ().executed_checks (),
607 test_case.totals ().executed_test_cases (),
608 test_case.child_test_cases_count_);
609#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
610
611 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
612 std::string indent2 (indent_size_ * (test_case.nesting_depth () + 1), ' ');
613
614 // The nested test cases skipped by the filter keep their index, and are
615 // counted in the plan.
616 size_t plan_count = test_case.totals ().executed_checks ()
617 + test_case.totals ().executed_test_cases ()
618 + test_case.totals ().skipped_test_cases ();
619
620 // The name and the indentation are not included, since their lengths
621 // are not bounded; they are passed to each write.
622 const bool passed = test_case.totals ().passed ();
623 const char* result = passed ? "ok" : "not ok";
624 const char* status = passed ? "passed" : "FAILED";
625
626 char message_totals[totals_size_];
627 format_test_case_totals (test_case, message_totals);
628
629 const std::string name = tap_text (test_case.name (), true);
630
631 write_buffer_to_file ();
632
633 const bool to_console = (verbosity_ == verbosity::normal
634 || verbosity_ == verbosity::verbose);
635
636 // At this point, the buffer may contain output from the test case, which
637 // should be displayed.
638 if (to_console)
639 {
640 if (add_empty_line_)
641 {
642 write_console ("\n");
643 }
644
645 if (passed) [[likely]]
646 {
647 // Successful test case.
648 if (verbosity_ == verbosity::verbose)
649 {
650 // With verbosity, show full TAP output accumulated in the
651 // buffer.
652 write_buffer_to_console ();
653 }
654 }
655 else
656 {
657 // Failed test case.
658
659 // Show full TAP output accumulated in the buffer for failed
660 // test cases, as it may contain useful information about the
661 // failure.
662 write_buffer_to_console ();
663 }
664 }
665
666 if (passed && verbosity_ == verbosity::normal)
667 {
668 // Without verbosity, show only the summary line and count only
669 // test cases, not checks, as the TAP output is not shown; the
670 // output file has all the lines, so its plan counts the checks.
671 write_file ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
672 indent2.c_str (), plan_count);
673
674 write_console ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
675 indent2.c_str (),
676 test_case.totals ().executed_test_cases ()
677 + test_case.totals ().skipped_test_cases ());
678 }
679 else
680 {
681 write_line (to_console,
682 "%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
683 indent2.c_str (), plan_count);
684 }
685
686 write_line (to_console,
687 "%s%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
688 " - %s # { %s, %s }\n",
689 indent.c_str (), result, test_case.own_index_, name.c_str (),
690 status, message_totals);
691
692 flush_streams ();
693
694 clear_buffer ();
695
696 add_empty_line_ = true;
697 }
