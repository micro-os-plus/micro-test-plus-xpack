103 std::string
104 tap_text (std::string_view text, bool escape)
105 {
106 std::string result;
107 result.reserve (text.size ());
108
109 for (size_t i = 0; i < text.size (); ++i)
110 {
111 const char c = text[i];
112 if (c == '\n' || c == '\r')
113 {
114 // The `\r\n` pair is a single line break.
115 if (c == '\r' && i + 1 < text.size () && text[i + 1] == '\n')
116 {
117 ++i;
118 }
119 result.push_back (' ');
120 }
121 else if (escape && (c == '\\' || c == '#'))
122 {
123 result.push_back ('\\');
124 result.push_back (c);
125 }
126 else
127 {
128 result.push_back (c);
129 }
130 }
131
132 return result;
133 }

293 std::string
294 yaml_double_quoted (std::string_view text)
295 {
296 constexpr std::string_view digits{ "0123456789ABCDEF" };
297
298 std::string result;
299 result.reserve (text.size () + 2);
300
301 result.push_back ('"');
302 for (const char c : text)
303 {
304 const auto u = static_cast<unsigned char> (c);
305 switch (c)
306 {
307 case '\\':
308 result.append ("\\\\");
309 break;
310 case '"':
311 result.append ("\\\"");
312 break;
313 case '\0':
314 result.append ("\\0");
315 break;
316 case '\a':
317 result.append ("\\a");
318 break;
319 case '\b':
320 result.append ("\\b");
321 break;
322 case '\t':
323 result.append ("\\t");
324 break;
325 case '\n':
326 result.append ("\\n");
327 break;
328 case '\v':
329 result.append ("\\v");
330 break;
331 case '\f':
332 result.append ("\\f");
333 break;
334 case '\r':
335 result.append ("\\r");
336 break;
337 case '\x1B':
338 result.append ("\\e");
339 break;
340 default:
341 if (u < 0x20 || u == 0x7F)
342 {
343 result.append ("\\x");
344 result.push_back (digits[static_cast<std::size_t> (u >> 4)]);
345 result.push_back (
346 digits[static_cast<std::size_t> (u & 0x0F)]);
347 }
348 else
349 {
350 result.push_back (c);
351 }
352 break;
353 }
354 }
355 result.push_back ('"');
356
357 return result;
358 }

527 void

529 {
530#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
531 trace::printf (
532 "%s '%s' +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
533 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
534 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
535 ", xs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
536 " | cs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
537 __PRETTY_FUNCTION__, suite.name (), suite.totals ().passed_checks (),
538 suite.totals ().failed_checks (), suite.totals ().executed_checks (),
539 suite.totals ().executed_test_cases (), suite.child_test_cases_count_);
540#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
541
542 std::string indent (indent_size_, ' ');
543
544 // The name is not included, since its length is not bounded; it is
545 // passed to each write.
546 const bool passed = suite.totals ().passed ();
547 const char* result = passed ? "ok" : "not ok";
548 const char* status = passed ? "passed" : "FAILED";
549
550 char message_totals[totals_size_];
551 format_suite_totals (suite, message_totals);
552
553 char message_time[elapsed_time_size_];
554 format_elapsed_time (suite.stopwatch_, message_time);
555
556 const tap_name name{ suite.name (), true };
557
558 write_buffer_to_file ();
559
560 const bool to_console = route (record_kind::suite_result).console;
561
562 // At this point, the buffer may contain output from the test case, which
563 // should be displayed.
564 if (to_console)
565 {
566 if (add_empty_line_ && suite.totals ().executed_test_cases () > 0)
567 {
568 write_console ("\n");
569 }
570
571 // With verbosity, the full TAP output accumulated in the buffer;
572 // for a failed suite, as it may contain useful information about
573 // the failure.
574 write_console_text (console_lines (!passed));
575 }
576 else
577 {
578 write_quiet_failures ();
579 }
580
581 // The plan counts the test cases and the failure of the suite body,
582 // if any, which is numbered after them.
583 write_line (to_console,
584 "%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
585 "\n%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
586 " - %s # { %s, %s%s }\n",
587 indent.c_str (), suite.test_point_number_, result,
588 suite.own_index_, name.c_str (), status, message_totals,
589 message_time);
590
591 if (!to_console && route (record_kind::total).console)
592 {
593 // Quiet: only the test point of the suite, counted by the session
594 // plan.
595 write_console ("%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
596 " - %s # { %s, %s%s }\n",
597 result, suite.own_index_, name.c_str (), status,
598 message_totals, message_time);
599 }
600
601 flush_streams ();
602
603 clear_buffer ();
604
605 add_empty_line_ = true;
606
607 current_suite_ = nullptr;
608 }

747 void
748 reporter_tap::end_test_case (const test_case& test_case)
749 {
750#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
751 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
752 " +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
753 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
754 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
755 ", xs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
756 " | cs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
757 __PRETTY_FUNCTION__, test_case.name (),
758 test_case.nesting_depth (),
759 test_case.totals ().passed_checks (),
760 test_case.totals ().failed_checks (),
761 test_case.totals ().executed_checks (),
762 test_case.totals ().executed_test_cases (),
763 test_case.child_test_cases_count_);
764#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
765
766 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
767 std::string indent2 (indent_size_ * (test_case.nesting_depth () + 1), ' ');
768
769 // The nested test cases skipped by the filter keep their index, and are
770 // counted in the plan.
771 size_t plan_count = test_case.totals ().executed_checks ()
772 + test_case.totals ().executed_test_cases ()
773 + test_case.totals ().skipped_test_cases ();
774
775 // The name and the indentation are not included, since their lengths
776 // are not bounded; they are passed to each write.
777 const bool passed = test_case.totals ().passed ();
778 const char* result = passed ? "ok" : "not ok";
779 const char* status = passed ? "passed" : "FAILED";
780
781 char message_totals[totals_size_];
782 format_test_case_totals (test_case, message_totals);
783
784 const tap_name name{ test_case.name (), true };
785
786 write_buffer_to_file ();
787
788 const bool to_console = route (record_kind::test_case_result).console;
789
790 // A passed test case without anything displayed under it is a single
791 // test point.
792 const bool is_single_point
793 = to_console && passed
794 && (open_test_cases_.empty ()
795 || !open_test_cases_.back ().name_displayed);
796
797 if (to_console)
798 {
799 if (is_single_point)
800 {
801 // The headings of the outer test cases, to keep the test point
802 // in their block.
803 display_open_test_cases (false);
804 }
805 else
806 {
807 // The heading of this test case too, if not displayed yet.
808 display_open_test_cases (true);
809
810 if (add_empty_line_)
811 {
812 write_console ("\n");
813 }
814
815 // The full TAP output accumulated in the buffer, as the
816 // plan counts it.
817 write_console_text (console_lines (true));
818
819 write_console ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
820 "\n",
821 indent2.c_str (), plan_count);
822 }
823 }
824 else
825 {
826 write_quiet_failures ();
827 }
828
829 write_file ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
830 indent2.c_str (), plan_count);
831
832 write_line (to_console,
833 "%s%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
834 " - %s # { %s, %s }\n",
835 indent.c_str (), result, test_case.own_index_, name.c_str (),
836 status, message_totals);
837
838 flush_streams ();
839
840 clear_buffer ();
841
842 add_empty_line_ = true;
843
844 if (!open_test_cases_.empty ())
845 {
846 open_test_cases_.pop_back ();
847 }
848 }

1228 {
1229 if (failures_.empty ())
1230 {
1231 return;
1232 }
1233
1234 std::string path;
1235 if (current_suite_ != nullptr)
1236 {
1237 path = tap_name{ current_suite_->name (), false }.c_str ();
1238 }
1239 for (const open_test_case& open : open_test_cases_)
1240 {
1241 if (!path.empty ())
1242 {
1243 path.append (" > ");
1244 }
1245 path.append (tap_name{ open.test->name (), false }.c_str ());
1246 }
1247 write_console ("%s%s\n", comment_prefix (), path.c_str ());
1248

1251 {

1253 const size_t length
1254 = (end == std::string_view::npos) ? rest.size () : end + 1;
1255 write_console ("%s", comment_prefix ());
1256 write_console_text (rest.substr (0, length));
1257 rest.remove_prefix (length);
1258 }
