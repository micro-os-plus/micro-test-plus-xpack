134 printf ("\n");

164 void
165 reporter_tap::end_session (runner& runner)
166 {
167#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
168 trace::printf ("%s\n", __PRETTY_FUNCTION__);
169#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
170
171 size_t total_suites_count = runner.total_suites_count ();
172
173 uint32_t milliseconds = 0;
174 uint32_t microseconds = 0;
175 if (runner.timings ().has_timestamps ())
176 {
177 runner.timings ().compute_elapsed_time (milliseconds, microseconds);
178 }
179
180 char message_summary[32];
181 snprintf (message_summary, sizeof (message_summary), "1..%zu",
182 total_suites_count);
183
184 char message_totals[160];
185 snprintf (message_totals, sizeof (message_totals),
186 "total: %zu check%s passed, %zu failed, in %zu test "
187 "case%s, %zu test suite%s",
188 runner.totals ().successful_checks (),
189 runner.totals ().successful_checks () == 1 ? "" : "s",
190 runner.totals ().failed_checks (),
191 runner.totals ().executed_subtests (),
192 runner.totals ().executed_subtests () == 1 ? "" : "s",
193 total_suites_count, total_suites_count == 1 ? "" : "s");
194
195 char message_time[120] = "";
196 if (milliseconds > 0 || microseconds > 0)
197 {
198 snprintf (message_time, sizeof (message_time),
199 ", time: %" PRIu32 ".%03" PRIu32 " ms", milliseconds,
200 microseconds);
201 }
202
203 if (output_file_ != nullptr)
204 {
205 fprintf (output_file_, "%s\n# { %s%s }\n", message_summary,
206 message_totals, message_time);
207 }
208
209 if (verbosity_ != verbosity::silent)
210 {
211 if (add_empty_line_)
212 {
213 printf ("\n");
214 }
215
216 if (verbosity_ != verbosity::quiet)
217 {
218 printf ("%s\n", message_summary);
219 }
220 else
221 {
222 // With quiet verbosity, there are no ok/not ok lines, so the TAP
223 // plan should look like a skipped test.
224 printf ("1..0\n");
225 }
226
227 printf ("# { %s%s }\n", message_totals, message_time);
228
229 flush ();
230 }
231 }

283 void
284 reporter_tap::end_suite (suite& suite)
285 {
286#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
287 trace::printf (
288 "%s '%s' +%zu -%zu in xc%zu, xs%zu | cs%zu\n", __PRETTY_FUNCTION__,
289 suite.name (), suite.totals ().successful_checks (),
290 suite.totals ().failed_checks (), suite.totals ().executed_checks (),
291 suite.totals ().executed_subtests (),
292 suite.children_subtests_count ());
293#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
294
295 uint32_t milliseconds = 0;
296 uint32_t microseconds = 0;
297 if (suite.timings ().has_timestamps ())
298 {
299 suite.timings ().compute_elapsed_time (milliseconds, microseconds);
300 }
301
302 std::string indent (indent_size, ' ');
303
304 char message_summary[40];
305 snprintf (message_summary, sizeof (message_summary), "%s1..%zu",
306 indent.c_str (), suite.children_subtests_count ());
307 char message_totals[120];
308 if (suite.totals ().was_successful ()) [[likely]]
309 {
310 snprintf (message_totals, sizeof (message_totals),
311 "ok %zu - %s # { passed, %zu check%s in %zu "
312 "test case%s",
313 suite.own_index (), suite.name (),
314 suite.totals ().successful_checks (),
315 suite.totals ().successful_checks () == 1 ? "" : "s",
316 suite.totals ().executed_subtests (),
317 suite.totals ().executed_subtests () == 1 ? "" : "s");
318 }
319 else
320 {
321 snprintf (message_totals, sizeof (message_totals),
322 "not ok %zu - %s # { FAILED, %zu check%s "
323 "passed, %zu failed, in %zu test case%s",
324 suite.own_index (), suite.name (),
325 suite.totals ().successful_checks (),
326 suite.totals ().successful_checks () == 1 ? "" : "s",
327 suite.totals ().failed_checks (),
328 suite.totals ().executed_subtests (),
329 suite.totals ().executed_subtests () == 1 ? "" : "s");
330 }
331
332 char message_time[120] = "";
333 if (milliseconds > 0 || microseconds > 0)
334 {
335 snprintf (message_time, sizeof (message_time),
336 ", time: %" PRIu32 ".%03" PRIu32 " ms", milliseconds,
337 microseconds);
338 }
339
340 if (output_file_ != nullptr)
341 {
342 write_buffer_to_file_ ();
343
344 fprintf (output_file_, "%s\n%s%s }\n", message_summary, message_totals,
345 message_time);
346 }
347
348 // At this point, the buffer may contain output from the test case, which
349 // should be displayed.
350 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
351 {
352 if (add_empty_line_ && suite.totals ().executed_subtests () > 0)
353 {
354 printf ("\n");
355 }
356
357 if (suite.totals ().was_successful ()) [[likely]]
358 {
359 // Successful test suite.
360
361 if (verbosity_ == verbosity::verbose)
362 {
363 // With verbosity, show full TAP output accumulated in the
364 // buffer.
365 write_buffer_to_stdout ();
366 }
367
368 printf ("%s\n%s%s }\n", message_summary, message_totals,
369 message_time);
370 }
371 else
372 {
373 // Failed test suite.
374
375 // Show full TAP output accumulated in the buffer for failed suite
376 // cases, as it may contain useful information about the failure.
377 write_buffer_to_stdout ();
378
379 printf ("%s\n%s%s }\n", message_summary, message_totals,
380 message_time);
381 }
382
383 flush ();
384 }
385
386 buffer_.clear ();
387
388 add_empty_line_ = true;
389 }

451 void
452 reporter_tap::end_subtest (subtest& subtest)
453 {
454#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
455 trace::printf ("%s '%s' i%zu +%zu -%zu in xc%zu, xs%zu | cs%zu\n",
456 __PRETTY_FUNCTION__, subtest.name (),
457 subtest.nesting_depth (),
458 subtest.totals ().successful_checks (),
459 subtest.totals ().failed_checks (),
460 subtest.totals ().executed_checks (),
461 subtest.totals ().executed_subtests (),
462 subtest.children_subtests_count ());
463#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
464
465 std::string indent (indent_size * subtest.nesting_depth (), ' ');
466 std::string indent2 (indent_size * (subtest.nesting_depth () + 1), ' ');
467
468 char message_summary[40];
469 snprintf (message_summary, sizeof (message_summary), "%s1..%zu",
470 indent2.c_str (),
471 subtest.totals ().executed_checks ()
472 + subtest.totals ().executed_subtests ());
473
474 char message_totals[120];
475 if (subtest.totals ().was_successful ()) [[likely]]
476 {
477 snprintf (message_totals, sizeof (message_totals),
478 "%sok %zu - %s # { passed, %zu check%s }", indent.c_str (),
479 subtest.own_index (), subtest.name (),
480 subtest.totals ().successful_checks (),
481 subtest.totals ().successful_checks () == 1 ? "" : "s");
482 }
483 else
484 {
485 snprintf (message_totals, sizeof (message_totals),
486 "%snot ok %zu - %s # { FAILED, %zu check%s "
487 "passed, %zu failed }",
488 indent.c_str (), subtest.own_index (), subtest.name (),
489 subtest.totals ().successful_checks (),
490 subtest.totals ().successful_checks () == 1 ? "" : "s",
491 subtest.totals ().failed_checks ());
492 }
493
494 if (output_file_ != nullptr)
495 {
496 write_buffer_to_file_ ();
497
498 fprintf (output_file_, "%s\n%s\n", message_summary, message_totals);
499 }
500
501 // At this point, the buffer may contain output from the subtest, which
502 // should be displayed.
503 if (verbosity_ == verbosity::normal || verbosity_ == verbosity::verbose)
504 {
505 if (add_empty_line_)
506 {
507 printf ("\n");
508 }
509
510 if (subtest.totals ().was_successful ()) [[likely]]
511 {
512 // Successful subtest.
513 if (verbosity_ == verbosity::verbose)
514 {
515 // With verbosity, show full TAP output accumulated in the
516 // buffer.
517 write_buffer_to_stdout ();
518
519 printf ("%s\n", message_summary);
520 }
521 else
522 {
523 // Without verbosity, show only the summary line
524 // and count only subtests, not checks, as the TAP output is
525 // not shown.
526 printf ("%s1..%zu\n", indent2.c_str (),
527 subtest.totals ().executed_subtests ());
528 }
529
530 printf ("%s\n", message_totals);
531 }
532 else
533 {
534 // Failed subtest.
535
536 // Show full TAP output accumulated in the buffer for failed
537 // subtests, as it may contain useful information about the
538 // failure.
539 write_buffer_to_stdout ();
540
541 printf ("%s\n%s\n", message_summary, message_totals);
542 }
543
544 flush ();
545 }
546
547 buffer_.clear ();
548
549 add_empty_line_ = true;
550 }

615 void
616 reporter_tap::output_fail_prefix_ (
617 std::string& message, const bool has_expression,
618 [[maybe_unused]] const reflection::source_location& location,
619 subtest& subtest)
620 {
621 size_t level = subtest.nesting_depth ();
622 *this << indent (level + 1) << "not ok "
623 << static_cast<int> (subtest.current_subtest_index ());
624
625 if (!message.empty ())
626 {
627 *this << " - " << message.c_str ();
628 }
629 *this << endl;
630
631 // https://testanything.org/tap-version-14-specification.html
632 // 2-space indentation for YAML diagnostics.
633 *this << indent (level + 1) << " ---";
634 if (has_expression)
635 {
636 *this << endl;
637 *this << indent (level + 1) << " condition: ";
638 }
639 }
640
647 void
648 reporter_tap::output_fail_suffix_ (
649 const reflection::source_location& location, bool abort,
650 subtest& subtest)
651 {
652 size_t level = subtest.nesting_depth ();
653 if (abort)
654 {
655 *this << " aborted...";
656 }
657 *this << endl;
658
659 // https://testanything.org/tap-version-14-specification.html
660 // 2-space indentation for YAML diagnostics.
661
662 *this << indent (level + 1) << " at:" << endl;
663 *this << indent (level + 1)
664 << " filename: " << reflection::short_name (location.file_name ())
665 << endl;
666 *this << indent (level + 1) << " line: " << location.line () << endl;
667
668 *this << indent (level + 1) << " ..." << endl;
669
670 flush ();
671 }
