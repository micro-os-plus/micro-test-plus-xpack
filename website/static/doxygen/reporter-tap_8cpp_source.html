103 std::string
104 tap_text (std::string_view text, bool escape)
105 {
106 std::string result;
107 result.reserve (text.size ());
108
109 for (size_t i = 0; i < text.size (); ++i)
110 {
111 const char c = text[i];
112 if (c == '\n' || c == '\r')
113 {
114 // The `\r\n` pair is a single line break.
115 if (c == '\r' && i + 1 < text.size () && text[i + 1] == '\n')
116 {
117 ++i;
118 }
119 result.push_back (' ');
120 }
121 else if (escape && (c == '\\' || c == '#'))
122 {
123 result.push_back ('\\');
124 result.push_back (c);
125 }
126 else
127 {
128 result.push_back (c);
129 }
130 }
131
132 return result;
133 }

153 std::string
154 yaml_double_quoted (std::string_view text)
155 {
156 constexpr std::string_view digits{ "0123456789ABCDEF" };
157
158 std::string result;
159 result.reserve (text.size () + 2);
160
161 result.push_back ('"');
162 for (const char c : text)
163 {
164 const auto u = static_cast<unsigned char> (c);
165 switch (c)
166 {
167 case '\\':
168 result.append ("\\\\");
169 break;
170 case '"':
171 result.append ("\\\"");
172 break;
173 case '\0':
174 result.append ("\\0");
175 break;
176 case '\a':
177 result.append ("\\a");
178 break;
179 case '\b':
180 result.append ("\\b");
181 break;
182 case '\t':
183 result.append ("\\t");
184 break;
185 case '\n':
186 result.append ("\\n");
187 break;
188 case '\v':
189 result.append ("\\v");
190 break;
191 case '\f':
192 result.append ("\\f");
193 break;
194 case '\r':
195 result.append ("\\r");
196 break;
197 case '\x1B':
198 result.append ("\\e");
199 break;
200 default:
201 if (u < 0x20 || u == 0x7F)
202 {
203 result.append ("\\x");
204 result.push_back (digits[static_cast<std::size_t> (u >> 4)]);
205 result.push_back (
206 digits[static_cast<std::size_t> (u & 0x0F)]);
207 }
208 else
209 {
210 result.push_back (c);
211 }
212 break;
213 }
214 }
215 result.push_back ('"');
216
217 return result;
218 }

386 reporter_tap::end_suite (const suite& suite)
387 {
388#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
389 trace::printf (
390 "%s '%s' +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
391 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
392 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
393 ", xs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
394 " | cs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
395 __PRETTY_FUNCTION__, suite.name (), suite.totals ().passed_checks (),
396 suite.totals ().failed_checks (), suite.totals ().executed_checks (),
397 suite.totals ().executed_test_cases (), suite.child_test_cases_count_);
398#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
399
400 std::string indent (indent_size_, ' ');
401
402 // The name is not included, since its length is not bounded; it is
403 // passed to each write.
404 const bool passed = suite.totals ().passed ();
405 const char* result = passed ? "ok" : "not ok";
406 const char* status = passed ? "passed" : "FAILED";
407
408 char message_totals[totals_size_];
409 format_suite_totals (suite, message_totals);
410
411 char message_time[elapsed_time_size_];
412 format_elapsed_time (suite.stopwatch_, message_time);
413
414 const std::string name = tap_text (suite.name (), true);
415
416 write_buffer_to_file ();
417
418 const bool to_console = route (record_kind::suite_result).console;
419
420 // At this point, the buffer may contain output from the test case, which
421 // should be displayed.
422 if (to_console)
423 {
424 if (add_empty_line_ && suite.totals ().executed_test_cases () > 0)
425 {
426 write_console ("\n");
427 }
428
429 // With verbosity, the full TAP output accumulated in the buffer;
430 // for a failed suite, as it may contain useful information about
431 // the failure.
432 write_console_text (console_lines (!passed));
433 }
434 else
435 {
436 write_quiet_failures ();
437 }
438
439 // The plan counts the test cases and the failure of the suite body,
440 // if any, which is numbered after them.
441 write_line (to_console,
442 "%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
443 "\n%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
444 " - %s # { %s, %s%s }\n",
445 indent.c_str (), suite.test_point_number_, result,
446 suite.own_index_, name.c_str (), status, message_totals,
447 message_time);
448
449 if (!to_console && route (record_kind::total).console)
450 {
451 // Quiet: only the test point of the suite, counted by the session
452 // plan.
453 write_console ("%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
454 " - %s # { %s, %s%s }\n",
455 result, suite.own_index_, name.c_str (), status,
456 message_totals, message_time);
457 }
458
459 flush_streams ();
460
461 clear_buffer ();
462
463 add_empty_line_ = true;
464
465 current_suite_ = nullptr;
466 }

605 void
606 reporter_tap::end_test_case (const test_case& test_case)
607 {
608#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
609 trace::printf ("%s '%s' i%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
610 " +%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
611 " -%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
612 " in xc%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
613 ", xs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
614 " | cs%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
615 __PRETTY_FUNCTION__, test_case.name (),
616 test_case.nesting_depth (),
617 test_case.totals ().passed_checks (),
618 test_case.totals ().failed_checks (),
619 test_case.totals ().executed_checks (),
620 test_case.totals ().executed_test_cases (),
621 test_case.child_test_cases_count_);
622#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
623
624 std::string indent (indent_size_ * test_case.nesting_depth (), ' ');
625 std::string indent2 (indent_size_ * (test_case.nesting_depth () + 1), ' ');
626
627 // The nested test cases skipped by the filter keep their index, and are
628 // counted in the plan.
629 size_t plan_count = test_case.totals ().executed_checks ()
630 + test_case.totals ().executed_test_cases ()
631 + test_case.totals ().skipped_test_cases ();
632
633 // The name and the indentation are not included, since their lengths
634 // are not bounded; they are passed to each write.
635 const bool passed = test_case.totals ().passed ();
636 const char* result = passed ? "ok" : "not ok";
637 const char* status = passed ? "passed" : "FAILED";
638
639 char message_totals[totals_size_];
640 format_test_case_totals (test_case, message_totals);
641
642 const std::string name = tap_text (test_case.name (), true);
643
644 write_buffer_to_file ();
645
646 const bool to_console = route (record_kind::test_case_result).console;
647
648 // A passed test case without anything displayed under it is a single
649 // test point.
650 const bool is_single_point
651 = to_console && passed
652 && (open_test_cases_.empty ()
653 || !open_test_cases_.back ().name_displayed);
654
655 if (to_console)
656 {
657 if (is_single_point)
658 {
659 // The headings of the outer test cases, to keep the test point
660 // in their block.
661 display_open_headers (false);
662 }
663 else
664 {
665 // The heading of this test case too, if not displayed yet.
666 display_open_headers (true);
667
668 if (add_empty_line_)
669 {
670 write_console ("\n");
671 }
672
673 // The full TAP output accumulated in the buffer, as the
674 // plan counts it.
675 write_console_text (console_lines (true));
676
677 write_console ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
678 "\n",
679 indent2.c_str (), plan_count);
680 }
681 }
682 else
683 {
684 write_quiet_failures ();
685 }
686
687 write_file ("%s1..%" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE "\n",
688 indent2.c_str (), plan_count);
689
690 write_line (to_console,
691 "%s%s %" MICRO_OS_PLUS_MICRO_TEST_PLUS_PRI_SIZE
692 " - %s # { %s, %s }\n",
693 indent.c_str (), result, test_case.own_index_, name.c_str (),
694 status, message_totals);
695
696 flush_streams ();
697
698 clear_buffer ();
699
700 add_empty_line_ = true;
701
702 if (!open_test_cases_.empty ())
703 {
704 open_test_cases_.pop_back ();
705 }
706 }

946 void
947 reporter_tap::abort_session (const reflection::source_location& location,
948 bool is_requirement)
949 {
950#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
951 trace::printf ("%s\n", __PRETTY_FUNCTION__);
952#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
953
954 // The type of `line()` depends on the toolchain (`uint_least32_t` for
955 // `std::source_location`); widen it implicitly, without a cast that
956 // would be useless on some platforms.
957 const unsigned long line = location.line ();
958 // The reason is escaped, as the descriptions.
959 const std::string file_name
960 = tap_text (utility::extract_file_name (location.file_name ()), true);
961 const char* reason
962 = is_requirement ? "Requirement failed" : "Test execution aborted";
963
964 write_buffer_to_file ();
965
966 const bool to_console = route (record_kind::abort).console;
967
968 if (to_console)
969 {
970 if (route (record_kind::suite_result).console)
971 {
972 display_open_headers (true);
973
974 if (add_empty_line_)
975 {
976 write_console ("\n");
977 }
978
979 write_console_text (console_lines (true));
980 }
981 else
982 {
983 write_quiet_failures ();
984 }
985 }
986
987 write_line (to_console, "Bail out! %s at %s:%lu\n", reason,
988 file_name.c_str (), line);
989
990 flush_streams ();
991
992 clear_buffer ();
993
994 // Should the abort handler throw, nothing may refer to the objects
995 // that are unwound.
996 open_test_cases_.clear ();
997 current_suite_ = nullptr;

1049 void
1050 reporter_tap::display_open_headers (bool include_last)
1051 {
1052 const size_t count = include_last || open_test_cases_.empty ()
1053 ? open_test_cases_.size ()
1054 : open_test_cases_.size () - 1;
1055 for (size_t i = 0; i < count; ++i)
1056 {
1057 open_test_case& open = open_test_cases_[i];
1058 if (open.name_displayed)
1059 {
1060 continue;
1061 }
1062
1063 if (add_empty_line_)
1064 {
1065 write_console ("\n");
1066 }
1067
