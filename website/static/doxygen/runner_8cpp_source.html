168 suite&
169 runner::initialise (int argc, char* argv[], const char* top_suite_name)
170 {
171#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
172 trace::printf ("%s\n", __PRETTY_FUNCTION__);
173
174#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
175 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
176#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)
177 trace::printf ("argv[");
178 for (int i = 0; i < argc; ++i)
179 {
180 if (i > 0)
181 {
182 trace::printf (", ");
183 }
184 trace::printf ("'%s'", argv[i]);
185 }
186 trace::puts ("]");
187#endif // defined(MICRO_OS_PLUS_DEBUG_ENABLED)
188#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
189#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
190
191 if (strlen (top_suite_name) > 0)
192 {
193 // If provided by this call, use it, possibly override the
194 // deprecated constructor.
195 top_suite_name_ = top_suite_name;
196 top_suite_.name (top_suite_name_.c_str ());
197 }
198 else if (strlen (top_suite_.name ()) == 0)
199 {
200 // If not provided by the constructor or by this call, try to extract a
201 // name from argv[0], which is commonly the executable name. If that
202 // fails, use a default name.
203 if (argc > 0 && argv != nullptr && argv[0] != nullptr)
204 {
205 std::string_view top_suite_name_view{ utility::extract_file_name (
206 argv[0]) };
207
208 const auto dot_pos = top_suite_name_view.rfind ('.');
209 if (dot_pos != std::string_view::npos)
210 {
211 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
212 }
213
214 top_suite_name_ = top_suite_name_view;
215 }
216 else
217 {
218 top_suite_name_ = "default suite";
219 }
220 top_suite_.name (top_suite_name_.c_str ());
221 }
222
223 std::vector<std::string_view> argvs (argv, argv + argc);
224
225 std::string_view reporter_name{ "tap" };
226 static constexpr std::string_view reporter_prefix{ "--reporter=" };
227 for (size_t i = 0; i < argvs.size (); ++i)
228 {
229 if (argvs[i].starts_with (reporter_prefix))
230 {
231 reporter_name = argvs[i].substr (reporter_prefix.size ());
232 }
233 else if (argvs[i]
234 == reporter_prefix.substr (0, reporter_prefix.size () - 1))
235 {
236 if (i + 1 < argvs.size ())
237 {
238 reporter_name = argvs[++i];
239 }
240 else
241 {
242 fprintf (stderr, "error: --reporter option requires a "
243 "reporter name argument\n");
244 exit (1);
245 }
246 }
247 }
248
249 // Initialise and configure the reporter.
250 if (reporter_name == "human")
251 {
252 reporter_ = std::make_unique<reporter_human> (
253 std::make_unique<std::vector<std::string_view>> (
254 std::move (argvs)));
255 }
256 else if (reporter_name == "tap")
257 {
258 reporter_ = std::make_unique<reporter_tap> (
259 std::make_unique<std::vector<std::string_view>> (
260 std::move (argvs)));
261 }
262 else
263 {
264 fprintf (stderr, "error: unknown reporter '%.*s'\n",
265 static_cast<int> (reporter_name.size ()),
266 reporter_name.data ());
267 exit (1);
268 }
269
270 // ------------------------------------------------------------------------
271
272 timings_.timestamp_begin ();
273 reporter_->begin_session (*this);
274
275 top_suite_.timings ().timestamp_begin ();
276 reporter_->begin_suite (top_suite_);
277
278 return top_suite_;
279 }

314 void
315 runner::run_suites_ (void)
316 {
317 // Use selection sort with unique_ptr::swap (returns void) to avoid
318 // std::sort triggering -Waggregate-return via std::move_backward,
319 // which returns a class-type iterator when operating on unique_ptr
320 // elements.
321 const size_t n = children_suites_.size ();
322 for (size_t i = 0; i < n; ++i)
323 {
324 size_t min_idx = i;
325 for (size_t j = i + 1; j < n; ++j)
326 {
327 if (std::string_view{ children_suites_[j]->name () }
328 < std::string_view{ children_suites_[min_idx]->name () })
329 min_idx = j;
330 }
331 if (min_idx != i)
332 children_suites_[i].swap (children_suites_[min_idx]);
333 }
334
335 for (size_t i = 0; i < n; ++i)
336 {
337 auto* suite_ptr = children_suites_[i].get ();
338
339 // +1 for 1-based index, +1 for top suite
340 suite_ptr->own_index (i + 1 + 1);
341
342 // Run the child suite immediately.
343 suite_ptr->run ();
344
345 // Accumulate the totals from the child suite into the runner
346 // totals.
347 // DO NOT increment executed_subtests here.
348 totals_ += suite_ptr->totals ();
349 }
350 }

361 int
362 runner::exit_code (void)
363 {
364#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
365 trace::printf ("%s\n", __PRETTY_FUNCTION__);
366#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
367
368 if (reporter_ == nullptr)
369 {
370 fprintf (stderr, "error: test runner not initialised\n");
371 return 1;
372 }
373
374 top_suite_.timings ().timestamp_end ();
375 reporter_->end_suite (top_suite_);
376 totals_ += top_suite_.totals ();
377
378 run_suites_ ();
379
380 timings_.timestamp_end ();
381 reporter_->end_session (*this);
382
383 const int result = totals_.was_successful () ? 0 : 1;
384
385#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
386 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, result);
387#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
388
389 return result;
390 }

517 void
518 static_runner::run_suites_ (void)
519 {
520#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
521 trace::printf ("%s\n", __PRETTY_FUNCTION__);
522#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
523
524 runner::run_suites_ ();
525
526 if (static_children_suites_ != nullptr)
527 {
528 // Use selection sort with std::swap on raw pointers (returns void)
529 // to avoid std::sort triggering -Waggregate-return via
530 // std::move_backward returning a class-type iterator.
531 const size_t n = static_children_suites_->size ();
532 auto& suites = *static_children_suites_;
533 for (size_t i = 0; i < n; ++i)
534 {
535 size_t min_idx = i;
536 for (size_t j = i + 1; j < n; ++j)
537 {
538 if (std::string_view{ suites[j]->name () }
539 < std::string_view{ suites[min_idx]->name () })
540 min_idx = j;
541 }
542 if (min_idx != i)
543 std::swap (suites[i], suites[min_idx]);
544 }
545
546 for (size_t i = 0; i < n; ++i)
547 {
548 auto* suite_ptr = suites[i];
549
550 suite_ptr->own_index (i + 1 + suites_count ());
551
552 // Run the child suite immediately.
553 suite_ptr->run ();
554
555 // Accumulate the totals from the static suite into the runner
556 // totals.
557 // DO NOT increment executed_subtests here.
558 totals_ += suite_ptr->totals ();
559 }
560 }
