151 {

208 suite&
209 runner::initialise (const runner_options& options,
210 const char* top_suite_name)
211 {
212#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
213 trace::printf ("%s\n", __PRETTY_FUNCTION__);
214#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
215
216 if (aborted_)
217 {
218 fprintf (stderr, "error: test execution aborted\n");
219 return top_suite_;
220 }
221
222 if (session_state_ != session_state::created)
223 {
224 fprintf (stderr, "error: test runner already initialised\n");
225 return top_suite_;
226 }
227
228 if (top_suite_name != nullptr && strlen (top_suite_name) > 0)
229 {
230 // If provided by this call, use it, possibly override the
231 // deprecated constructor.
232 top_suite_name_ = top_suite_name;
233 top_suite_.name_ = top_suite_name_.c_str ();
234 }
235 else if (strlen (top_suite_.name ()) == 0)
236 {
237 // If not provided by the constructor or by this call, try to extract a
238 // name from argv[0], which is commonly the executable name. If that
239 // fails, use a default name.
240 if (!options.arguments.empty () && options.arguments[0] != nullptr)
241 {
242 std::string_view top_suite_name_view{ utility::extract_file_name (
243 options.arguments[0]) };
244
245 const auto dot_pos = top_suite_name_view.rfind ('.');
246 if (dot_pos != std::string_view::npos)
247 {
248 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
249 }
250
251 top_suite_name_ = top_suite_name_view;
252 }
253 else
254 {
255 top_suite_name_ = "default suite";
256 }
257 top_suite_.name_ = top_suite_name_.c_str ();
258 }
259
260 verbosity_ = options.verbosity_level;
261
262 // With an unknown reporter, do not proceed: no output file is created,
263 // nothing is reported, no test is run, and `run()` returns 2. A silent
264 // reporter without output file is still created, since the suites and
265 // the test cases refer to it.
266 if (has_invalid_reporter_)
267 {
268 runner_options silent_options{};
269 silent_options.verbosity_level = verbosity::silent;
270 verbosity_ = verbosity::silent;
271 reporter_ = std::make_unique<detail::reporter_tap> (silent_options);
272 exit_code_ = 2;
273 session_state_ = session_state::finished;
274 return top_suite_;
275 }
276
277 // Initialise and configure the reporter; it keeps its own copy of the
278 // verbosity, taken from the same options.
279 if (options.selected_reporter == reporter_kind::human)
280 {
281 reporter_ = std::make_unique<detail::reporter_human> (options);
282 }
283 else
284 {
285 reporter_ = std::make_unique<detail::reporter_tap> (options);
286 }
287
288 if (reporter_->has_output_file_error ())
289 {
290 has_option_errors_ = true;
291 }
292
293 // Copy the strings, not only the views, since the patterns are used
294 // by run(), after the caller's strings may have been destroyed. Copy
295 // them one by one, since the vector assignment uses std::copy, which
296 // triggers -Waggregate-return in the standard library headers on some
297 // targets.
298 filter_patterns_.reserve (options.filter_patterns.size ());
299 for (const std::string_view pattern : options.filter_patterns)
300 {
301 filter_patterns_.emplace_back (pattern);
302 }
303
304 // ------------------------------------------------------------------------
305
306 // When listing, the session is not reported; only the header of the
307 // list is written, and the test cases are listed as they are created.
308 list_tests_ = options.list_tests;
309 if (list_tests_)
310 {
311 reporter_->write_line (verbosity_ != verbosity::silent, "%s\n",
312 filter_patterns_.empty ()
313 ? "All available test cases:"
314 : "Matching test cases:");
315 report_option_errors ();
316 session_state_ = session_state::initialised;
317 return top_suite_;
318 }
319
320 stopwatch_.start ();
321 reporter_->begin_session (*this);
322 report_option_errors ();
323
324 top_suite_.stopwatch_.start ();
325 reporter_->begin_suite (top_suite_);
326
327 session_state_ = session_state::initialised;
328
329 return top_suite_;
330 }

401 runner_options
402 runner::parse_options (int argc, char* argv[])
403 {
404 runner_options opts{};
405
406 if (argc <= 0 || argv == nullptr)
407 {
408 return opts;
409 }
410
411 opts.arguments = { argv, static_cast<size_t> (argc) };
412
413 static constexpr std::string_view reporter_option{ "--reporter" };
414 static constexpr std::string_view output_file_option{ "--output-file" };
415 static constexpr std::string_view color_option{ "--color" };
416 static constexpr std::string_view use_colour_option{ "--use-colour" };
417 static constexpr std::string_view filter_option{ "--filter" };
418
419 // Get the value of an option given as `--option=<value>` or as
420 // `--option <value>`; the latter advances the index past the value.
421 // Return false if the argument is not this option, or if its value is
422 // missing, which is reported as an error.
423 auto get_value = [&] (std::string_view option, size_t& i,
424 std::string_view& value) -> bool
425 {
426 const std::string_view argument{ opts.arguments[i] };
427 if (argument == option)
428 {
429 if (i + 1 >= opts.arguments.size ())
430 {
431 add_option_error ("", option, " option requires an argument");
432 return false;
433 }
434 value = opts.arguments[++i];
435 return true;
436 }
437 if (argument.starts_with (option)
438 && argument.substr (option.size ()).starts_with ('='))
439 {
440 value = argument.substr (option.size () + 1);
441 return true;
442 }
443 return false;
444 };
445
446 // Whether the argument is one of the options with a value, possibly
447 // with the value missing, which was already reported.
448 auto is_known_option = [&] (std::string_view argument) -> bool
449 {
450 for (const std::string_view option :
451 { reporter_option, output_file_option, use_colour_option,
452 filter_option })
453 {
454 if (argument == option
455 || (argument.starts_with (option)
456 && argument.substr (option.size ()).starts_with ('=')))
457 {
458 return true;
459 }
460 }
461 return false;
462 };
463
464 // Set the colour mode from the value of `--color` or of
465 // `--use-colour`; both accept the values and the synonyms of the GNU
466 // tools, which include those of Catch2 and Boost UT (`yes`, `no`, and
467 // `auto`). An unknown mode is reported as an error.
468 auto set_colour_mode = [&] (std::string_view value)
469 {
470 if (value == "always" || value == "yes" || value == "force")
471 {
472 opts.colour = colour_mode::always;
473 }
474 else if (value == "never" || value == "no" || value == "none")
475 {
476 opts.colour = colour_mode::never;
477 }
478 else if (value == "auto" || value == "tty" || value == "if-tty")
479 {
480 opts.colour = colour_mode::automatic;
481 }
482 else
483 {
484 add_option_error ("unknown colour mode '", value, "'");
485 }
486 };
487
488 for (size_t i = 1; i < opts.arguments.size (); ++i)
489 {
490 const std::string_view argument{ opts.arguments[i] };
491 std::string_view value{};
492
493 if (argument == "--verbose")
494 {
495 opts.verbosity_level = verbosity::verbose;
496 }
497 else if (argument == "--quiet")
498 {
499 opts.verbosity_level = verbosity::quiet;
500 }
501 else if (argument == "--silent")
502 {
503 opts.verbosity_level = verbosity::silent;
504 }
505 else if (argument == "--list-tests" || argument == "--list")
506 {
507 opts.list_tests = true;
508 }
509 else if (get_value (reporter_option, i, value))
510 {
511 if (value == "human")
512 {
513 opts.selected_reporter = reporter_kind::human;
514 }
515 else if (value == "tap")
516 {
517 opts.selected_reporter = reporter_kind::tap;
518 }
519 else
520 {
521 add_option_error ("unknown reporter '", value, "'");
522 has_invalid_reporter_ = true;
523 }
524 }
525 else if (get_value (output_file_option, i, value))
526 {
527 // The value is a suffix of a null-terminated argv[] string,
528 // so it is null-terminated too. An empty path disables the
529 // output file.
530 opts.output_file_path = value.empty () ? nullptr : value.data ();
531 }
532 else if (get_value (use_colour_option, i, value))
533 {
534 set_colour_mode (value);
535 }
536 else if (argument == color_option)
537 {
538 // As in the GNU tools, the option alone means `always`, and
539 // the next argument is never taken as its value.
540 opts.colour = colour_mode::always;
541 }
542 else if (argument.starts_with (color_option)
543 && argument.substr (color_option.size ()).starts_with ('='))
544 {
545 set_colour_mode (argument.substr (color_option.size () + 1));
546 }
547 else if (get_value (filter_option, i, value))
548 {
549 // The view refers to the argv[] string, which remains valid
550 // until the program ends.
551 opts.filter_patterns.push_back (value);
552 }
553 else if (argument.starts_with ("--") && !is_known_option (argument))
554 {
555 // The other arguments are left to the application, but a
556 // mistyped option must not silently disable the tests.
557 add_option_error ("unknown option '", argument, "'");
558 }
559 }
560
561 return opts;
562 }

574 bool
575 runner::is_selected (const char* name, size_t level) const
576 {
577 if (level >= filter_patterns_.size ())
578 {
579 return true;
580 }
581
582 std::string_view alternatives{ filter_patterns_[level] };
583 while (true)
584 {
585 const size_t separator = alternatives.find ('|');
586 if (utility::is_match (name, alternatives.substr (0, separator)))
587 {
588 return true;
589 }
590 if (separator == std::string_view::npos)
591 {
592 return false;
593 }
594 alternatives.remove_prefix (separator + 1);
595 }
596 }

733 void
734 runner::run_suites (void)
735 {
736 // The top suite has index 1.
737 size_t own_index = 1;
738 for (class suite* suite_ptr = suite::first_; suite_ptr != nullptr;
739 suite_ptr = suite_ptr->next_)
740 {
741 // Skip the suites run by other runners.
742 if (&suite_ptr->test_runner () != this)
743 {
744 continue;
745 }
746
747 suite_ptr->own_index_ = ++own_index;
748
749 if (!is_selected (suite_ptr->name (), 0))
750 {
751 ++skipped_suites_count_;
752 if (!list_tests_)
753 {
754 reporter_->skip_suite (*suite_ptr);
755 }
756 continue;
757 }
758
759 suite_ptr->run ();
760
761 // Accumulate the totals from the child suite into the runner
762 // totals.
763 // DO NOT increment executed_test_cases here.
764 totals_ += suite_ptr->totals ();
765 }
766 }

793 int
794 runner::run (void)
795 {
796#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
797 trace::printf ("%s\n", __PRETTY_FUNCTION__);
798#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
799
800 if (aborted_)
801 {
802 fprintf (stderr, "error: test execution aborted\n");
803 return 1;
804 }
805
806 if (session_state_ == session_state::finished)
807 {
808 // The session has ended; do not run the suites again.
809 return exit_code_;
810 }
811
812 if (session_state_ == session_state::running)
813 {
814 fprintf (stderr, "error: test runner already running\n");
815 return 1;
816 }
817
818 if (session_state_ != session_state::initialised)
819 {
820 fprintf (stderr, "error: test runner not initialised\n");
821 return 1;
822 }
823
824 session_state_ = session_state::running;
825 started_next_ = started_first_;
826 started_first_ = this;
827
828 // When listing, the suites run only to create their test cases, which
829 // are listed, and the number of listed test cases ends the list.
830 if (list_tests_)
831 {
832 run_suites ();
833
834 const unsigned long test_cases = listed_test_cases_count_;
835 const unsigned long suites = listed_suites_count_;
836 reporter_->write_line (verbosity_ != verbosity::silent,
837 "%lu test case%s in %lu test suite%s\n",
838 test_cases, (test_cases == 1) ? "" : "s",
839 suites, (suites == 1) ? "" : "s");
840 reporter_->flush_streams ();
841
842 exit_code_ = has_option_errors_ ? 2 : 0;
843 session_state_ = session_state::finished;
844 return exit_code_;
845 }
846
847 top_suite_.stopwatch_.stop ();
848 reporter_->end_suite (top_suite_);
849 totals_ += top_suite_.totals ();
850
851 run_suites ();
852
853 // A filter that selects no test case disables the tests; report it.
854 if (!filter_patterns_.empty () && totals_.executed_test_cases () == 0)
855 {
856 add_option_error ("the filters selected no test case");
857 reporter_->write_line (verbosity_ != verbosity::silent,
858 "%serror: the filters selected no test case\n",
859 reporter_->comment_prefix ());
860 }
861
862 stopwatch_.stop ();
863 reporter_->end_session (*this);
864
865 if (has_option_errors_)
866 {
867 exit_code_ = 2;
868 }
869 else
870 {
871 exit_code_ = (totals_.passed () && !has_rejected_suites_) ? 0 : 1;
872 }
873 session_state_ = session_state::finished;
874
875#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
876 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, exit_code_);
877#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
878
879 return exit_code_;
880 }

937 void
938 runner::abort (const reflection::source_location& sl, bool is_requirement)
939 {
940#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
941 trace::printf ("%s\n", __PRETTY_FUNCTION__);
942#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
943
944 if (!aborted_)
945 {
946 aborted_ = true;
947
948 if (session_state_ == session_state::initialised
949 || session_state_ == session_state::running)
950 {
951 reporter_->abort_session (sl, is_requirement);
952 }
953 else
954 {
955 // The type of `line()` depends on the toolchain
956 // (`uint_least32_t` for `std::source_location`); widen it
957 // implicitly, without a cast that would be useless on some
958 // platforms.
959 const unsigned long line = sl.line ();
960 fprintf (stderr, "error: test execution aborted at %s:%lu\n",
961 utility::extract_file_name (sl.file_name ()), line);
962 }
963 }
964
965 fflush (stdout);
966 fflush (stderr);
967
968 if (abort_handler_ != nullptr)
969 {
970 abort_handler_ (sl);
971 }
972
973 ::abort ();
974 }
