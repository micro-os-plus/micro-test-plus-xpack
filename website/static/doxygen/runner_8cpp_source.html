272 void
273 runner::initialise (runner_options options)
274 {
275#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
276 trace::printf ("%s\n", __PRETTY_FUNCTION__);
277#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
278
279 if (aborted_)
280 {
281 fprintf (stderr, "error: test execution aborted\n");
282 return;
283 }
284
285 if (session_state_ != session_state::created)
286 {
287 reject_call ("test runner already initialised");
288 return;
289 }
290
291 // Move the vectors, since their copy assignment uses std::copy, which
292 // triggers -Waggregate-return in the standard library headers on some
293 // targets.
294 options_ = std::move (options);
295
296 // With errors in the options, do not proceed: the errors are written on
297 // the console and in the output file, if any, which is truncated, so
298 // that it never keeps the results of an earlier run; no test is run, and
299 // `run_suites()` returns 2. The same for `--help`, which writes the
300 // usage instead of the errors, only on the console, and `run_suites()`
301 // returns 0. A reporter is still created, since the suites and the test
302 // cases refer to it.
303 if (options_.has_errors () || options_.help)
304 {
305 runner_options console_options{ options_ };
306 if (!options_.has_errors ())
307 {
308 console_options.output_file_path.clear ();
309 }
310 if (console_options.selected_reporter == reporter_kind::human)
311 {
312 reporter_
313 = std::make_unique<detail::reporter_human> (console_options);
314 }
315 else
316 {
317 reporter_
318 = std::make_unique<detail::reporter_tap> (console_options);
319 }
320
321 for (const std::string& error : options_.errors ())
322 {
323 fprintf (stderr, "error: %s\n", error.c_str ());
324 reporter_->write_line (
325 options_.verbosity_level != verbosity::silent, "%s%s\n",
326 reporter_->error_prefix (), error.c_str ());
327 }
328 if (options_.has_unknown_option ())
329 {
330 reporter_->write_line (options_.verbosity_level
331 != verbosity::silent,
332 "%sUse --help to list the options.\n",
333 reporter_->comment_prefix ());
334 }
335 if (options_.help && !options_.has_errors ())
336 {
337 // Requested, so written also with the silent verbosity.
338 const char* program = options_.arguments ().empty ()
339 ? "<program>"
340 : utility::extract_file_name (
341 options_.arguments ()[0].c_str ());
342 reporter_->write_line (true,
343 "Usage: %s [options] [--] "
344 "[arguments]\n%s",
345 program, help_options);
346 }
347 reporter_->flush_streams ();
348
349 exit_code_ = options_.has_errors () ? 2 : 0;
350 session_state_ = session_state::declined;
351 return;
352 }
353
354 // Initialise and configure the reporter; it keeps its own copy of the
355 // verbosity, taken from the same options.
356 if (options_.selected_reporter == reporter_kind::human)
357 {
358 reporter_ = std::make_unique<detail::reporter_human> (options_);
359 }
360 else
361 {
362 reporter_ = std::make_unique<detail::reporter_tap> (options_);
363 }
364
365 // If the output file cannot be opened, do not run the tests either.
366 if (reporter_->has_output_file_error ())
367 {
368 reporter_->write_line (options_.verbosity_level != verbosity::silent,
369 "%sfailed to open output file '%s'\n",
370 reporter_->error_prefix (),
371 options_.output_file_path.c_str ());
372 reporter_->flush_streams ();
373
374 exit_code_ = 2;
375 session_state_ = session_state::declined;
376 return;
377 }
378
379 // ------------------------------------------------------------------------
380
381 // When listing, the session is not reported; only the header of the
382 // list is written, and the test cases are listed as they are created.
383 if (options_.list_tests)
384 {
385 reporter_->write_line (true, "%s\n",
386 options_.filter_patterns.empty ()
387 ? "All available test cases:"
388 : "Matching test cases:");
389 session_state_ = session_state::initialised;
390 return;
391 }
392
393 stopwatch_.start ();
394 reporter_->begin_session (*this);
395
396 session_state_ = session_state::initialised;
397 }

419 suite&
420 runner::add_top_suite (const located_name& top_suite_name)
421 {
422 const std::string_view requested_name{ top_suite_name.name () };
423
424#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
425 trace::printf ("%s\n", __PRETTY_FUNCTION__);
426#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
427
428 if (aborted_)
429 {
430 fprintf (stderr, "error: test execution aborted\n");
431 return top_suite_;
432 }
433
434 if (session_state_ == session_state::created)
435 {
436 initialise (runner_options{});
437 }
438
439 if (has_option_errors () || options_.help)
440 {
441 return top_suite_;
442 }
443
444 if (has_top_suite_)
445 {
446 reject_call ("top suite already added", &top_suite_name.location ());
447 return top_suite_;
448 }
449
450 if (!requested_name.empty ())
451 {
452 // If provided by this call, use it, possibly override the
453 // deprecated constructor.
454 top_suite_.located_name_ = top_suite_name;
455 top_suite_.name_ = top_suite_.located_name_.name ();
456 }
457 else if (strlen (top_suite_.name ()) == 0)
458 {
459 // If not provided by the constructor or by this call, try to extract a
460 // name from argv[0], which is commonly the executable name. If that
461 // fails, use a default name.
462 if (!options_.arguments ().empty ())
463 {
464 std::string_view top_suite_name_view{ utility::extract_file_name (
465 options_.arguments ()[0].c_str ()) };
466
467 const auto dot_pos = top_suite_name_view.rfind ('.');
468 if (dot_pos != std::string_view::npos)
469 {
470 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
471 }
472
473 top_suite_.located_name_
474 = located_name{ top_suite_name_view,
475 top_suite_name.location () };
476 }
477 else
478 {
479 top_suite_.located_name_
480 = located_name{ "default suite", top_suite_name.location () };
481 }
482 top_suite_.name_ = top_suite_.located_name_.name ();
483 }
484
485 // Named first, so that the rejection names it.
486 if (session_state_ != session_state::initialised)
487 {
488 reject_suite (top_suite_.name (), top_suite_name.location ());
489 return top_suite_;
490 }
491
492 has_top_suite_ = true;
493
494 // The top suite runs until run_suites() is called; its test cases
495 // are accepted only meanwhile.
496 running_suite_ = &top_suite_;
497
498 // When listing, the top suite is not reported; its test cases are
499 // listed as they are created.
500 if (!options_.list_tests)
501 {
502 top_suite_.stopwatch_.start ();
503 reporter_->begin_suite (top_suite_);
504 }
505
506 return top_suite_;
507 }

529 bool
530 runner::is_selected (const char* name, size_t level) const
531 {
532 if (level >= options_.filter_patterns.size ())
533 {
534 return true;
535 }
536
537 std::string_view alternatives{ options_.filter_patterns[level] };
538 while (true)
539 {
540 const size_t separator = alternatives.find ('|');
541 if (utility::is_match (name, alternatives.substr (0, separator)))
542 {
543 return true;
544 }
545 if (separator == std::string_view::npos)
546 {
547 return false;
548 }
549 alternatives.remove_prefix (separator + 1);
550 }
551 }

659 void
660 runner::reject_call (const std::string& message,
661 const reflection::source_location* location)
662 {
663 fprintf (stderr, "error: %s\n", message.c_str ());
664 mark_rejected ();
665
666 if (reporter_ == nullptr || session_state_ == session_state::declined
667 || aborted_)
668 {
669 return;
670 }
671
672 if (options_.list_tests)
673 {
674 reporter_->write_line (options_.verbosity_level != verbosity::silent,
675 "%serror: %s\n", reporter_->comment_prefix (),
676 message.c_str ());
677 reporter_->flush_streams ();
678 return;
679 }
680
681 if (running_test_case_ != nullptr)
682 {
683 running_test_case_->report_rejected_test (
684 message, location != nullptr
685 ? *location
686 : running_test_case_->located_name_.location ());
687 return;
688 }
689
690 if (running_suite_ != nullptr)
691 {
692 running_suite_->next_test_point_number ();
693 reporter_->fail_suite (message, *running_suite_);
694 running_suite_->totals_.increment_failed_checks ();
695 return;
696 }
697
698 reporter_->write_line (options_.verbosity_level != verbosity::silent,
699 "%serror: %s\n", reporter_->comment_prefix (),
700 message.c_str ());
701 reporter_->flush_streams ();
702 }

777 void
778 runner::run_child_suites (void)
779 {
780 // The top suite, if added, has index 1.
781 size_t own_index = has_top_suite_ ? 1 : 0;
782 for (micro_test_plus::suite* suite_ptr = suite::first_;
783 suite_ptr != nullptr; suite_ptr = suite_ptr->next_)
784 {
785 // Skip the suites run by other runners.
786 if (&suite_ptr->test_runner () != this)
787 {
788 continue;
789 }
790
791 suite_ptr->own_index_ = ++own_index;
792
793 if (!is_selected (suite_ptr->name (), 0))
794 {
795 ++skipped_suites_count_;
796 if (!options_.list_tests)
797 {
798 reporter_->skip_suite (*suite_ptr);
799 }
800 continue;
801 }
802
803 suite_ptr->run ();
804
805 // Accumulate the totals from the child suite into the runner
806 // totals.
807 // DO NOT increment executed_test_cases here.
808 totals_ += suite_ptr->totals ();
809 }
810 }

839 int
840 runner::run_suites (void)
841 {
842#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
843 trace::printf ("%s\n", __PRETTY_FUNCTION__);
844#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
845
846 if (aborted_)
847 {
848 fprintf (stderr, "error: test execution aborted\n");
849 return 1;
850 }
851
852 if (session_state_ == session_state::finished
853 || session_state_ == session_state::declined)
854 {
855 // The session has ended; do not run the suites again.
856 return exit_code_;
857 }
858
859 if (session_state_ == session_state::running)
860 {
861 fprintf (stderr, "error: test runner already running\n");
862 return 1;
863 }
864
865 if (session_state_ == session_state::created)
866 {
867 initialise (runner_options{});
868 }
869
870 session_state_ = session_state::running;
871 running_suite_ = nullptr;
872 started_next_ = started_first_;
873 started_first_ = this;
874
875 // When listing, the suites run only to create their test cases, which
876 // are listed, and the number of listed test cases ends the list.
877 if (options_.list_tests)
878 {
879 run_child_suites ();
880
881 const unsigned long test_cases = listed_test_cases_count_;
882 const unsigned long suites = listed_suites_count_;
883 reporter_->write_line (true, "%lu test case%s in %lu test suite%s\n",
884 test_cases, (test_cases == 1) ? "" : "s",
885 suites, (suites == 1) ? "" : "s");
886
887 // As for a run, filters that select no test case are an error.
888 const bool listed_nothing = total_suites_count () != 0
889 && !options_.filter_patterns.empty ()
890 && listed_test_cases_count_ == 0;
891 if (listed_nothing)
892 {
893 fprintf (stderr, "error: the filters selected no test case\n");
894 reporter_->write_line (
895 options_.verbosity_level != verbosity::silent,
896 "%serror: the filters selected no test case\n",
897 reporter_->comment_prefix ());
898 }
899 reporter_->flush_streams ();
900
901 if (listed_nothing || has_option_errors ())
902 {
903 exit_code_ = 2;
904 }
905 else
906 {
907 exit_code_ = has_rejected_calls_ ? 1 : 0;
908 }
909 session_state_ = session_state::finished;
910 return exit_code_;
911 }
912
913 if (has_top_suite_)
914 {
915 top_suite_.stopwatch_.stop ();
916 reporter_->end_suite (top_suite_);
917 totals_ += top_suite_.totals ();
918 }
919
920 run_child_suites ();
921
922 // A session without suites runs nothing; report it.
923 const bool has_no_suites = (total_suites_count () == 0);
924 if (has_no_suites)
925 {
926 fprintf (stderr, "error: no test suites\n");
927 reporter_->write_line (options_.verbosity_level != verbosity::silent,
928 "%serror: no test suites\n",
929 reporter_->comment_prefix ());
930 }
931
932 // A filter that selects no test case disables the tests; report it.
933 const bool selected_nothing = !has_no_suites
934 && !options_.filter_patterns.empty ()
935 && totals_.executed_test_cases () == 0;
936 if (selected_nothing)
937 {
938 fprintf (stderr, "error: the filters selected no test case\n");
939 reporter_->write_line (options_.verbosity_level != verbosity::silent,
940 "%serror: the filters selected no test case\n",
941 reporter_->comment_prefix ());
942 }
943
944 stopwatch_.stop ();
945 reporter_->end_session (*this);
946
947 if (selected_nothing || has_option_errors ())
948 {
949 exit_code_ = 2;
950 }
951 else
952 {
953 exit_code_
954 = (totals_.passed () && !has_rejected_calls_ && !has_no_suites)
955 ? 0
956 : 1;
957 }
958 session_state_ = session_state::finished;
959
960#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
961 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, exit_code_);
962#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
963
964 return exit_code_;
965 }

1024 void
1025 runner::abort (const reflection::source_location& sl, bool is_requirement)
1026 {
1027#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
1028 trace::printf ("%s\n", __PRETTY_FUNCTION__);
1029#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
1030
1031 if (!aborted_)
1032 {
1033 aborted_ = true;
1034
1035 if (session_state_ == session_state::initialised
1036 || session_state_ == session_state::running)
1037 {
1038 reporter_->abort_session (sl, is_requirement);
1039 }
1040 else
1041 {
1042 // The type of `line()` depends on the toolchain
1043 // (`uint_least32_t` for `std::source_location`); widen it
1044 // implicitly, without a cast that would be useless on some
1045 // platforms.
1046 const unsigned long line = sl.line ();
