160 suite&
161 runner::initialise (int argc, char* argv[], const char* top_suite_name)
162 {
163#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
164 trace::printf ("%s\n", __PRETTY_FUNCTION__);
165#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
166
167#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \
168 && defined(MICRO_OS_PLUS_DIAG_TRACE_ENABLED))
169#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)
170 trace::printf ("argv[");
171 for (int i = 0; i < argc; ++i)
172 {
173 if (i > 0)
174 {
175 trace::printf (", ");
176 }
177 trace::printf ("'%s'", argv[i]);
178 }
179 trace::puts ("]");
180#endif // defined(MICRO_OS_PLUS_DEBUG_ENABLED)
181#endif // !defined(MICRO_OS_PLUS_STARTUP_ENABLED)
182
183 if (strlen (top_suite_name) > 0)
184 {
185 // If provided by this call, use it, possibly override the
186 // deprecated constructor.
187 top_suite_name_ = top_suite_name;
188 top_suite_.name (top_suite_name_.c_str ());
189 }
190 else if (strlen (top_suite_.name ()) == 0)
191 {
192 // If not provided by the constructor or by this call, try to extract a
193 // name from argv[0], which is commonly the executable name. If that
194 // fails, use a default name.
195 if (argc > 0 && argv != nullptr && argv[0] != nullptr)
196 {
197 std::string_view top_suite_name_view{ utility::extract_file_name (
198 argv[0]) };
199
200 const auto dot_pos = top_suite_name_view.rfind ('.');
201 if (dot_pos != std::string_view::npos)
202 {
203 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
204 }
205
206 top_suite_name_ = top_suite_name_view;
207 }
208 else
209 {
210 top_suite_name_ = "default suite";
211 }
212 top_suite_.name (top_suite_name_.c_str ());
213 }
214
215 std::vector<std::string_view> argvs (argv, argv + argc);
216
217 std::string_view reporter_name{ "tap" };
218 static constexpr std::string_view reporter_prefix{ "--reporter=" };
219 for (size_t i = 0; i < argvs.size (); ++i)
220 {
221 if (argvs[i].starts_with (reporter_prefix))
222 {
223 reporter_name = argvs[i].substr (reporter_prefix.size ());
224 }
225 else if (argvs[i]
226 == reporter_prefix.substr (0, reporter_prefix.size () - 1))
227 {
228 if (i + 1 < argvs.size ())
229 {
230 reporter_name = argvs[++i];
231 }
232 else
233 {
234 fprintf (stderr, "error: --reporter option requires a "
235 "reporter name argument\n");
236 exit (1);
237 }
238 }
239 }
240
241 // Initialise and configure the reporter.
242 if (reporter_name == "human")
243 {
244 reporter_ = std::make_unique<reporter_human> (
245 std::make_unique<std::vector<std::string_view>> (
246 std::move (argvs)));
247 }
248 else if (reporter_name == "tap")
249 {
250 reporter_ = std::make_unique<reporter_tap> (
251 std::make_unique<std::vector<std::string_view>> (
252 std::move (argvs)));
253 }
254 else
255 {
256 fprintf (stderr, "error: unknown reporter '%.*s'\n",
257 static_cast<int> (reporter_name.size ()),
258 reporter_name.data ());
259 exit (1);
260 }
261
262 // ------------------------------------------------------------------------
263
264 timings_.timestamp_begin ();
265 reporter_->begin_session (*this);
266
267 top_suite_.timings ().timestamp_begin ();
268 reporter_->begin_suite (top_suite_);
269
270 return top_suite_;
271 }

302 void
303 runner::run_suites_ (void)
304 {
305 // Use selection sort with unique_ptr::swap (returns void) to avoid
306 // std::sort triggering -Waggregate-return via std::move_backward,
307 // which returns a class-type iterator when operating on unique_ptr
308 // elements.
309 const size_t n = children_suites_.size ();
310 for (size_t i = 0; i < n; ++i)
311 {
312 size_t min_idx = i;
313 for (size_t j = i + 1; j < n; ++j)
314 {
315 if (std::string_view{ children_suites_[j]->name () }
316 < std::string_view{ children_suites_[min_idx]->name () })
317 min_idx = j;
318 }
319 if (min_idx != i)
320 children_suites_[i].swap (children_suites_[min_idx]);
321 }
322
323 for (size_t i = 0; i < n; ++i)
324 {
325 auto* suite_ptr = children_suites_[i].get ();
326
327 // +1 for 1-based index, +1 for top suite
328 suite_ptr->own_index (i + 1 + 1);
329
330 // Run the child suite immediately.
331 suite_ptr->run ();
332
333 // Accumulate the totals from the child suite into the runner
334 // totals.
335 // DO NOT increment executed_subtests here.
336 totals_ += suite_ptr->totals ();
337 }
338 }

349 int
350 runner::exit_code (void)
351 {
352#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
353 trace::printf ("%s\n", __PRETTY_FUNCTION__);
354#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
355
356 if (reporter_ == nullptr)
357 {
358 fprintf (stderr, "error: test runner not initialised\n");
359 return 1;
360 }
361
362 top_suite_.timings ().timestamp_end ();
363 reporter_->end_suite (top_suite_);
364 totals_ += top_suite_.totals ();
365
366 run_suites_ ();
367
368 timings_.timestamp_end ();
369 reporter_->end_session (*this);
370
371 const int result = totals_.was_successful () ? 0 : 1;
372
373#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
374 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, result);
375#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
376
377 return result;
378 }

505 void
506 static_runner::run_suites_ (void)
507 {
508#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
509 trace::printf ("%s\n", __PRETTY_FUNCTION__);
510#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
511
512 runner::run_suites_ ();
513
514 if (static_children_suites_ != nullptr)
515 {
516 // Use selection sort with std::swap on raw pointers (returns void)
517 // to avoid std::sort triggering -Waggregate-return via
518 // std::move_backward returning a class-type iterator.
519 const size_t n = static_children_suites_->size ();
520 auto& suites = *static_children_suites_;
521 for (size_t i = 0; i < n; ++i)
522 {
523 size_t min_idx = i;
524 for (size_t j = i + 1; j < n; ++j)
525 {
526 if (std::string_view{ suites[j]->name () }
527 < std::string_view{ suites[min_idx]->name () })
528 min_idx = j;
529 }
530 if (min_idx != i)
531 std::swap (suites[i], suites[min_idx]);
532 }
533
534 for (size_t i = 0; i < n; ++i)
535 {
536 auto* suite_ptr = suites[i];
537
538 suite_ptr->own_index (i + 1 + suites_count ());
539
540 // Run the child suite immediately.
541 suite_ptr->run ();
542
543 // Accumulate the totals from the static suite into the runner
544 // totals.
545 // DO NOT increment executed_subtests here.
546 totals_ += suite_ptr->totals ();
547 }
548 }
