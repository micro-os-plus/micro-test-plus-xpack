137 runner::initialise (int argc, char* argv[], const char* top_suite_name)
138 {
139#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
140 trace::printf ("%s\n", __PRETTY_FUNCTION__);
141
142#if !(defined(MICRO_OS_PLUS_STARTUP_ENABLED) \

144#if defined(MICRO_OS_PLUS_DEBUG_ENABLED)

191 suite&
192 runner::initialise (const runner_options& options,
193 const char* top_suite_name)
194 {
195#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
196 trace::printf ("%s\n", __PRETTY_FUNCTION__);
197#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
198
199 if (aborted_)
200 {
201 fprintf (stderr, "error: test execution aborted\n");
202 return top_suite_;
203 }
204
205 if (session_state_ != session_state::created)
206 {
207 fprintf (stderr, "error: test runner already initialised\n");
208 return top_suite_;
209 }
210
211 if (top_suite_name != nullptr && strlen (top_suite_name) > 0)
212 {
213 // If provided by this call, use it, possibly override the
214 // deprecated constructor.
215 top_suite_name_ = top_suite_name;
216 top_suite_.name_ = top_suite_name_.c_str ();
217 }
218 else if (strlen (top_suite_.name ()) == 0)
219 {
220 // If not provided by the constructor or by this call, try to extract a
221 // name from argv[0], which is commonly the executable name. If that
222 // fails, use a default name.
223 if (!options.arguments.empty () && options.arguments[0] != nullptr)
224 {
225 std::string_view top_suite_name_view{ utility::extract_file_name (
226 options.arguments[0]) };
227
228 const auto dot_pos = top_suite_name_view.rfind ('.');
229 if (dot_pos != std::string_view::npos)
230 {
231 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
232 }
233
234 top_suite_name_ = top_suite_name_view;
235 }
236 else
237 {
238 top_suite_name_ = "default suite";
239 }
240 top_suite_.name_ = top_suite_name_.c_str ();
241 }
242
243 verbosity_ = options.verbosity_level;
244
245 // Initialise and configure the reporter; it keeps its own copy of the
246 // verbosity, taken from the same options.
247 if (options.selected_reporter == reporter_kind::human)
248 {
249 reporter_ = std::make_unique<detail::reporter_human> (options);
250 }
251 else
252 {
253 reporter_ = std::make_unique<detail::reporter_tap> (options);
254 }
255
256 if (reporter_->has_output_file_error ())
257 {
258 has_option_errors_ = true;
259 }
260
261 // Copy the strings, not only the views, since the patterns are used
262 // by run(), after the caller's strings may have been destroyed. Copy
263 // them one by one, since the vector assignment uses std::copy, which
264 // triggers -Waggregate-return in the standard library headers on some
265 // targets.
266 filter_patterns_.reserve (options.filter_patterns.size ());
267 for (const std::string_view pattern : options.filter_patterns)
268 {
269 filter_patterns_.emplace_back (pattern);
270 }
271
272 // ------------------------------------------------------------------------
273
274 // When listing, the session is not reported; only the header of the
275 // list is written, and the test cases are listed as they are created.
276 list_tests_ = options.list_tests;
277 if (list_tests_)
278 {
279 reporter_->write_line (verbosity_ != verbosity::silent, "%s\n",
280 filter_patterns_.empty ()
281 ? "All available test cases:"
282 : "Matching test cases:");
283 session_state_ = session_state::initialised;
284 return top_suite_;
285 }
286
287 stopwatch_.start ();
288 reporter_->begin_session (*this);
289
290 top_suite_.stopwatch_.start ();
291 reporter_->begin_suite (top_suite_);
292
293 session_state_ = session_state::initialised;
294
295 return top_suite_;
296 }

330 runner_options
331 runner::parse_options (int argc, char* argv[])
332 {
333 runner_options opts{};
334
335 if (argc <= 0 || argv == nullptr)
336 {
337 return opts;
338 }
339
340 opts.arguments = { argv, static_cast<size_t> (argc) };
341
342 static constexpr std::string_view reporter_option{ "--reporter" };
343 static constexpr std::string_view output_file_option{ "--output-file" };
344 static constexpr std::string_view color_option{ "--color" };
345 static constexpr std::string_view use_colour_option{ "--use-colour" };
346 static constexpr std::string_view filter_option{ "--filter" };
347
348 // Get the value of an option given as `--option=<value>` or as
349 // `--option <value>`; the latter advances the index past the value.
350 // Return false if the argument is not this option, or if its value is
351 // missing, which is reported as an error.
352 auto get_value = [&] (std::string_view option, size_t& i,
353 std::string_view& value) -> bool
354 {
355 const std::string_view argument{ opts.arguments[i] };
356 if (argument == option)
357 {
358 if (i + 1 >= opts.arguments.size ())
359 {
360 fprintf (stderr, "error: %.*s option requires an argument\n",
361 static_cast<int> (option.size ()), option.data ());
362 has_option_errors_ = true;
363 return false;
364 }
365 value = opts.arguments[++i];
366 return true;
367 }
368 if (argument.starts_with (option)
369 && argument.substr (option.size ()).starts_with ('='))
370 {
371 value = argument.substr (option.size () + 1);
372 return true;
373 }
374 return false;
375 };
376
377 // Set the colour mode from the value of `--color` or of
378 // `--use-colour`; both accept the values and the synonyms of the GNU
379 // tools, which include those of Catch2 and Boost UT (`yes`, `no`, and
380 // `auto`). An unknown mode is reported as an error.
381 auto set_colour_mode = [&] (std::string_view value)
382 {
383 if (value == "always" || value == "yes" || value == "force")
384 {
385 opts.colour = colour_mode::always;
386 }
387 else if (value == "never" || value == "no" || value == "none")
388 {
389 opts.colour = colour_mode::never;
390 }
391 else if (value == "auto" || value == "tty" || value == "if-tty")
392 {
393 opts.colour = colour_mode::automatic;
394 }
395 else
396 {
397 fprintf (stderr, "error: unknown colour mode '%.*s'\n",
398 static_cast<int> (value.size ()), value.data ());
399 has_option_errors_ = true;
400 }
401 };
402
403 for (size_t i = 1; i < opts.arguments.size (); ++i)
404 {
405 const std::string_view argument{ opts.arguments[i] };
406 std::string_view value{};
407
408 if (argument == "--verbose")
409 {
410 opts.verbosity_level = verbosity::verbose;
411 }
412 else if (argument == "--quiet")
413 {
414 opts.verbosity_level = verbosity::quiet;
415 }
416 else if (argument == "--silent")
417 {
418 opts.verbosity_level = verbosity::silent;
419 }
420 else if (argument == "--list-tests" || argument == "--list")
421 {
422 opts.list_tests = true;
423 }
424 else if (get_value (reporter_option, i, value))
425 {
426 if (value == "human")
427 {
428 opts.selected_reporter = reporter_kind::human;
429 }
430 else if (value == "tap")
431 {
432 opts.selected_reporter = reporter_kind::tap;
433 }
434 else
435 {
436 fprintf (stderr, "error: unknown reporter '%.*s'\n",
437 static_cast<int> (value.size ()), value.data ());
438 has_option_errors_ = true;
439 }
440 }
441 else if (get_value (output_file_option, i, value))
442 {
443 // The value is a suffix of a null-terminated argv[] string,
444 // so it is null-terminated too. An empty path disables the
445 // output file.
446 opts.output_file_path = value.empty () ? nullptr : value.data ();
447 }
448 else if (get_value (use_colour_option, i, value))
449 {
450 set_colour_mode (value);
451 }
452 else if (argument == color_option)
453 {
454 // As in the GNU tools, the option alone means `always`, and
455 // the next argument is never taken as its value.
456 opts.colour = colour_mode::always;
457 }
458 else if (argument.starts_with (color_option)
459 && argument.substr (color_option.size ()).starts_with ('='))
460 {
461 set_colour_mode (argument.substr (color_option.size () + 1));
462 }
463 else if (get_value (filter_option, i, value))
464 {
465 // The view refers to the argv[] string, which remains valid
466 // until the program ends.
467 opts.filter_patterns.push_back (value);
468 }
469 }
470
471 return opts;
472 }

484 bool
485 runner::is_selected (const char* name, size_t level) const
486 {
487 if (level >= filter_patterns_.size ())
488 {
489 return true;
490 }
491
492 std::string_view alternatives{ filter_patterns_[level] };
493 while (true)
494 {
495 const size_t separator = alternatives.find ('|');
496 if (utility::is_match (name, alternatives.substr (0, separator)))
497 {
498 return true;
499 }
500 if (separator == std::string_view::npos)
501 {
502 return false;
503 }
504 alternatives.remove_prefix (separator + 1);
505 }
506 }

603 void
604 runner::run_suites (void)
605 {
606 // The top suite has index 1.
607 size_t own_index = 1;
608 for (class suite* suite_ptr = suite::first_; suite_ptr != nullptr;
609 suite_ptr = suite_ptr->next_)
610 {
611 // Skip the suites run by other runners.
612 if (&suite_ptr->test_runner () != this)
613 {
614 continue;
615 }
616
617 suite_ptr->own_index_ = ++own_index;
618
619 if (!is_selected (suite_ptr->name (), 0))
620 {
621 ++skipped_suites_count_;
622 if (!list_tests_)
623 {
624 reporter_->skip_suite (*suite_ptr);
625 }
626 continue;
627 }
628
629 suite_ptr->run ();
630
631 // Accumulate the totals from the child suite into the runner
632 // totals.
633 // DO NOT increment executed_test_cases here.
634 totals_ += suite_ptr->totals ();
635 }
636 }

663 int
664 runner::run (void)
665 {
666#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
667 trace::printf ("%s\n", __PRETTY_FUNCTION__);
668#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
669
670 if (aborted_)
671 {
672 fprintf (stderr, "error: test execution aborted\n");
673 return 1;
674 }
675
676 if (session_state_ == session_state::finished)
677 {
678 // The session has ended; do not run the suites again.
679 return exit_code_;
680 }
681
682 if (session_state_ == session_state::running)
683 {
684 fprintf (stderr, "error: test runner already running\n");
685 return 1;
686 }
687
688 if (session_state_ != session_state::initialised)
689 {
690 fprintf (stderr, "error: test runner not initialised\n");
691 return 1;
692 }
693
694 session_state_ = session_state::running;
695
696 // When listing, the suites run only to create their test cases, which
697 // are listed, and the number of listed test cases ends the list.
698 if (list_tests_)
699 {
700 run_suites ();
701
702 const unsigned long test_cases = listed_test_cases_count_;
703 const unsigned long suites = listed_suites_count_;
704 reporter_->write_line (verbosity_ != verbosity::silent,
705 "%lu test case%s in %lu test suite%s\n",
706 test_cases, (test_cases == 1) ? "" : "s",
707 suites, (suites == 1) ? "" : "s");
708 reporter_->flush_streams ();
709
710 exit_code_ = has_option_errors_ ? 2 : 0;
711 session_state_ = session_state::finished;
712 return exit_code_;
713 }
714
715 top_suite_.stopwatch_.stop ();
716 reporter_->end_suite (top_suite_);
717 totals_ += top_suite_.totals ();
718
719 run_suites ();
720
721 stopwatch_.stop ();
722 reporter_->end_session (*this);
723
724 if (has_option_errors_)
725 {
726 exit_code_ = 2;
727 }
728 else
729 {
730 exit_code_ = (totals_.passed () && !has_rejected_suites_) ? 0 : 1;
731 }
732 session_state_ = session_state::finished;
733
734#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
735 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, exit_code_);
736#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
737
738 return exit_code_;
739 }

795 void
796 runner::abort (const reflection::source_location& sl, bool is_requirement)
797 {
798#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
799 trace::printf ("%s\n", __PRETTY_FUNCTION__);
800#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
801
802 if (!aborted_)
803 {
804 aborted_ = true;
805
806 if (session_state_ == session_state::initialised
807 || session_state_ == session_state::running)
808 {
809 reporter_->abort_session (sl, is_requirement);
810 }
811 else
812 {
813 // The type of `line()` depends on the toolchain
814 // (`uint_least32_t` for `std::source_location`); widen it
815 // implicitly, without a cast that would be useless on some
816 // platforms.
817 const unsigned long line = sl.line ();
818 fprintf (stderr, "error: test execution aborted at %s:%lu\n",
819 utility::extract_file_name (sl.file_name ()), line);
820 }
821 }
822
823 fflush (stdout);
824 fflush (stderr);
825
826 if (abort_handler_ != nullptr)
827 {
828 abort_handler_ (sl);
829 }
830
831 ::abort ();
832 }
