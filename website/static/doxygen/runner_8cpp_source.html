228 void
229 runner::initialise (runner_options options)
230 {
231#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
232 trace::printf ("%s\n", __PRETTY_FUNCTION__);
233#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
234
235 if (aborted_)
236 {
237 fprintf (stderr, "error: test execution aborted\n");
238 return;
239 }
240
241 if (session_state_ != session_state::created)
242 {
243 fprintf (stderr, "error: test runner already initialised\n");
244 return;
245 }
246
247 // Move the vectors, since their copy assignment uses std::copy, which
248 // triggers -Waggregate-return in the standard library headers on some
249 // targets.
250 options_ = std::move (options);
251
252 // With an unknown reporter, do not proceed: no output file is created,
253 // nothing is reported, no test is run, and `run()` returns 2. A silent
254 // reporter without output file is still created, since the suites and
255 // the test cases refer to it.
256 if (options_.has_invalid_reporter ())
257 {
258 runner_options silent_options{};
259 silent_options.verbosity_level = verbosity::silent;
260 options_.verbosity_level = verbosity::silent;
261 reporter_ = std::make_unique<detail::reporter_tap> (silent_options);
262 exit_code_ = 2;
263 session_state_ = session_state::finished;
264 return;
265 }
266
267 // Initialise and configure the reporter; it keeps its own copy of the
268 // verbosity, taken from the same options.
269 if (options_.selected_reporter == reporter_kind::human)
270 {
271 reporter_ = std::make_unique<detail::reporter_human> (options_);
272 }
273 else
274 {
275 reporter_ = std::make_unique<detail::reporter_tap> (options_);
276 }
277
278 // ------------------------------------------------------------------------
279
280 // When listing, the session is not reported; only the header of the
281 // list is written, and the test cases are listed as they are created.
282 if (options_.list_tests)
283 {
284 reporter_->write_line (
285 options_.verbosity_level != verbosity::silent, "%s\n",
286 options_.filter_patterns.empty () ? "All available test cases:"
287 : "Matching test cases:");
288 session_state_ = session_state::initialised;
289 return;
290 }
291
292 stopwatch_.start ();
293 reporter_->begin_session (*this);
294
295 session_state_ = session_state::initialised;
296 }

317 suite&
318 runner::add_top_suite (std::string_view top_suite_name)
319 {
320#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
321 trace::printf ("%s\n", __PRETTY_FUNCTION__);
322#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
323
324 if (aborted_)
325 {
326 fprintf (stderr, "error: test execution aborted\n");
327 return top_suite_;
328 }
329
330 if (session_state_ == session_state::created)
331 {
332 initialise (runner_options{});
333 }
334
335 if (options_.has_invalid_reporter ())
336 {
337 return top_suite_;
338 }
339
340 if (has_top_suite_)
341 {
342 fprintf (stderr, "error: top suite already added\n");
343 return top_suite_;
344 }
345
346 if (!top_suite_name.empty ())
347 {
348 // If provided by this call, use it, possibly override the
349 // deprecated constructor.
350 top_suite_name_ = top_suite_name;
351 top_suite_.name_ = top_suite_name_.c_str ();
352 }
353 else if (strlen (top_suite_.name ()) == 0)
354 {
355 // If not provided by the constructor or by this call, try to extract a
356 // name from argv[0], which is commonly the executable name. If that
357 // fails, use a default name.
358 if (!options_.arguments.empty ())
359 {
360 std::string_view top_suite_name_view{ utility::extract_file_name (
361 options_.arguments[0].c_str ()) };
362
363 const auto dot_pos = top_suite_name_view.rfind ('.');
364 if (dot_pos != std::string_view::npos)
365 {
366 top_suite_name_view = top_suite_name_view.substr (0, dot_pos);
367 }
368
369 top_suite_name_ = top_suite_name_view;
370 }
371 else
372 {
373 top_suite_name_ = "default suite";
374 }
375 top_suite_.name_ = top_suite_name_.c_str ();
376 }
377
378 // Named first, so that the rejection names it.
379 if (session_state_ != session_state::initialised)
380 {
381 reject_suite (top_suite_.name ());
382 return top_suite_;
383 }
384
385 has_top_suite_ = true;
386
387 // When listing, the top suite is not reported; its test cases are
388 // listed as they are created.
389 if (!options_.list_tests)
390 {
391 top_suite_.stopwatch_.start ();
392 reporter_->begin_suite (top_suite_);
393 }
394
395 return top_suite_;
396 }

408 bool
409 runner::is_selected (const char* name, size_t level) const
410 {
411 if (level >= options_.filter_patterns.size ())
412 {
413 return true;
414 }
415
416 std::string_view alternatives{ options_.filter_patterns[level] };
417 while (true)
418 {
419 const size_t separator = alternatives.find ('|');
420 if (utility::is_match (name, alternatives.substr (0, separator)))
421 {
422 return true;
423 }
424 if (separator == std::string_view::npos)
425 {
426 return false;
427 }
428 alternatives.remove_prefix (separator + 1);
429 }
430 }

569 void
570 runner::run_suites (void)
571 {
572 // The top suite, if added, has index 1.
573 size_t own_index = has_top_suite_ ? 1 : 0;
574 for (micro_test_plus::suite* suite_ptr = suite::first_;
575 suite_ptr != nullptr; suite_ptr = suite_ptr->next_)
576 {
577 // Skip the suites run by other runners.
578 if (&suite_ptr->test_runner () != this)
579 {
580 continue;
581 }
582
583 suite_ptr->own_index_ = ++own_index;
584
585 if (!is_selected (suite_ptr->name (), 0))
586 {
587 ++skipped_suites_count_;
588 if (!options_.list_tests)
589 {
590 reporter_->skip_suite (*suite_ptr);
591 }
592 continue;
593 }
594
595 suite_ptr->run ();
596
597 // Accumulate the totals from the child suite into the runner
598 // totals.
599 // DO NOT increment executed_test_cases here.
600 totals_ += suite_ptr->totals ();
601 }
602 }

631 int
632 runner::run (void)
633 {
634#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
635 trace::printf ("%s\n", __PRETTY_FUNCTION__);
636#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
637
638 if (aborted_)
639 {
640 fprintf (stderr, "error: test execution aborted\n");
641 return 1;
642 }
643
644 if (session_state_ == session_state::finished)
645 {
646 // The session has ended; do not run the suites again.
647 return exit_code_;
648 }
649
650 if (session_state_ == session_state::running)
651 {
652 fprintf (stderr, "error: test runner already running\n");
653 return 1;
654 }
655
656 if (session_state_ == session_state::created)
657 {
658 initialise (runner_options{});
659 }
660
661 session_state_ = session_state::running;
662 started_next_ = started_first_;
663 started_first_ = this;
664
665 // When listing, the suites run only to create their test cases, which
666 // are listed, and the number of listed test cases ends the list.
667 if (options_.list_tests)
668 {
669 run_suites ();
670
671 const unsigned long test_cases = listed_test_cases_count_;
672 const unsigned long suites = listed_suites_count_;
673 reporter_->write_line (options_.verbosity_level != verbosity::silent,
674 "%lu test case%s in %lu test suite%s\n",
675 test_cases, (test_cases == 1) ? "" : "s",
676 suites, (suites == 1) ? "" : "s");
677 reporter_->flush_streams ();
678
679 exit_code_ = has_option_errors () ? 2 : 0;
680 session_state_ = session_state::finished;
681 return exit_code_;
682 }
683
684 if (has_top_suite_)
685 {
686 top_suite_.stopwatch_.stop ();
687 reporter_->end_suite (top_suite_);
688 totals_ += top_suite_.totals ();
689 }
690
691 run_suites ();
692
693 // A session without suites runs nothing; report it.
694 const bool has_no_suites = (total_suites_count () == 0);
695 if (has_no_suites)
696 {
697 fprintf (stderr, "error: no test suites\n");
698 reporter_->write_line (options_.verbosity_level != verbosity::silent,
699 "%serror: no test suites\n",
700 reporter_->comment_prefix ());
701 }
702
703 // A filter that selects no test case disables the tests; report it.
704 const bool selected_nothing = !has_no_suites
705 && !options_.filter_patterns.empty ()
706 && totals_.executed_test_cases () == 0;
707 if (selected_nothing)
708 {
709 fprintf (stderr, "error: the filters selected no test case\n");
710 reporter_->write_line (options_.verbosity_level != verbosity::silent,
711 "%serror: the filters selected no test case\n",
712 reporter_->comment_prefix ());
713 }
714
715 stopwatch_.stop ();
716 reporter_->end_session (*this);
717
718 if (selected_nothing || has_option_errors ())
719 {
720 exit_code_ = 2;
721 }
722 else
723 {
724 exit_code_
725 = (totals_.passed () && !has_rejected_suites_ && !has_no_suites)
726 ? 0
727 : 1;
728 }
729 session_state_ = session_state::finished;
730
731#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
732 trace::printf ("%s -> %d\n", __PRETTY_FUNCTION__, exit_code_);
733#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
734
735 return exit_code_;
736 }

795 void
796 runner::abort (const reflection::source_location& sl, bool is_requirement)
797 {
798#if defined(MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED)
799 trace::printf ("%s\n", __PRETTY_FUNCTION__);
800#endif // MICRO_OS_PLUS_MICRO_TEST_PLUS_TRACE_ENABLED
801
802 if (!aborted_)
803 {
804 aborted_ = true;
805
806 if (session_state_ == session_state::initialised
807 || session_state_ == session_state::running)
808 {
809 reporter_->abort_session (sl, is_requirement);
810 }
811 else
812 {
813 // The type of `line()` depends on the toolchain
814 // (`uint_least32_t` for `std::source_location`); widen it
815 // implicitly, without a cast that would be useless on some
816 // platforms.
817 const unsigned long line = sl.line ();
818 fprintf (stderr, "error: test execution aborted at %s:%lu\n",
819 utility::extract_file_name (sl.file_name ()), line);
820 }
821 }
822
823 fflush (stdout);
824 fflush (stderr);
825
826 if (abort_handler_ != nullptr)
827 {
828 abort_handler_ (sl);
829 }
830
831 ::abort ();
832 }
