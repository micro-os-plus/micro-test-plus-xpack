126 template <auto N>
127 constexpr auto
128 integral_constant<N>::operator- () const noexcept
129 {
130 using value_type = decltype (N);
131
132 if constexpr (!std::is_signed_v<value_type>)
133 {
134 return integral_constant<static_cast<value_type> (value_type{}
135 - N)>{};
136 }
137 else if constexpr (N == std::numeric_limits<value_type>::min ())
138 {
139 static_assert (N != std::numeric_limits<value_type>::min (),
140 "µTest++: the negation of the minimum value does "
141 "not fit in the type of the literal");
142 return *this;
143 }
144 else
145 {
146 return integral_constant<static_cast<value_type> (-N)>{};
147 }
148 }

173 template <class T, char... Cs>
174 constexpr auto
175 integral_literal (void)
176 {
177 constexpr unsigned long long magnitude
178 = math::integer_magnitude<T, Cs...> ();
179
180 if constexpr (magnitude > static_cast<unsigned long long> (
181 std::numeric_limits<T>::max ()))
182 {
183 return integral_min_magnitude<T>{};
184 }
185 else
186 {
187 return integral_constant<static_cast<T> (magnitude)>{};
188 }
