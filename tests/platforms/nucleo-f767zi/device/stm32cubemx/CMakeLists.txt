message (VERBOSE
         "Processing 'tests/platforms/${PLATFORM_NAME}/stm32cubemx/'..."
)

# -----------------------------------------------------------------------------
# Define the stm32cubemx library.

# Note: It cannot be a library, since it makes extensive use of `weak` symbols.
add_library (stm32cubemx-interface INTERFACE EXCLUDE_FROM_ALL)

# -----------------------------------------------------------------------------
target_include_directories (
  stm32cubemx-interface
  INTERFACE "Core/Inc" 
            "Drivers/STM32F7xx_HAL_Driver/Inc"
            "Drivers/STM32F7xx_HAL_Driver/Inc/Legacy"
            "Drivers/CMSIS/Device/ST/STM32F7xx/Include"
            "Drivers/CMSIS/Include"
)

# find * -type f -name "*.c" -o -name "*.S" -o -name "*.s" | sed -e 's|\(.*\)|
# "\1"|'
target_sources (
  stm32cubemx-interface
  INTERFACE 
            "Core/Src/main.c"
            "Core/Src/gpio.c"
            "Core/Src/stm32f7xx_it.c"
            "Core/Src/stm32f7xx_hal_msp.c"
            #
            # Do not include "Core/Src/sysmem.c", use the µOS++ _sbrk().
            #
            # Do not include the "Core/Src/syscalls.c", use the µOS++
            # definitions.
            #
            # DO NOT include the "Core/Src/system_stm32f7xx.c" assembly startup,
            # use the .c one.
            #
            "Core/Src/system_stm32f7xx.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_cortex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_rcc.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_rcc_ex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_flash.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_flash_ex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_gpio.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_dma.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_dma_ex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_pwr.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_pwr_ex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_i2c.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_i2c_ex.c"
            "Drivers/STM32F7xx_HAL_Driver/Src/stm32f7xx_hal_exti.c"
)

target_compile_definitions (
  stm32cubemx-interface INTERFACE STM32F767xx USE_HAL_DRIVER
)

target_compile_options (
  stm32cubemx-interface
  INTERFACE # Silence the warnings in the code generated by CubeMX.
            $<$<COMPILE_LANGUAGE:C>:-Wno-bad-function-cast>
            $<$<COMPILE_LANGUAGE:C>:-Wno-nested-externs>
            -Wno-padded
            -Wno-redundant-decls
            -Wno-conversion
            -Wno-duplicated-branches
            -Wno-switch-enum
            -Wno-unused-parameter
            -Wno-switch-default
            -Wno-cast-align
)

target_link_options (stm32cubemx-interface INTERFACE 
)

target_link_libraries (stm32cubemx-interface INTERFACE
# Possibly m.
)

if (COMMAND xpack_display_target_lists)
  xpack_display_target_lists (stm32cubemx-interface)
endif ()

# -----------------------------------------------------------------------------
# Aliases.
add_library (micro-os-plus::stm32cubemx ALIAS stm32cubemx-interface)
message (VERBOSE "> micro-os-plus::stm32cubemx -> stm32cubemx-interface")
