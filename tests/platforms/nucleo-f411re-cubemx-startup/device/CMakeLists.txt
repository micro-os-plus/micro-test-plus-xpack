# add_subdirectory ("stm32cubemx")

set (SAVED_CMAKE_PROJECT_NAME "${CMAKE_PROJECT_NAME}")
set (CMAKE_PROJECT_NAME "dummy")
add_executable (${CMAKE_PROJECT_NAME} EXCLUDE_FROM_ALL)
add_subdirectory ("stm32cubemx/cmake/stm32cubemx")
set_target_properties (STM32_Drivers PROPERTIES EXCLUDE_FROM_ALL TRUE)
set (CMAKE_PROJECT_NAME "${SAVED_CMAKE_PROJECT_NAME}")

get_directory_property (
  MX_Include_Dirs DIRECTORY "stm32cubemx/cmake/stm32cubemx" DEFINITION
                            MX_Include_Dirs
)
get_directory_property (
  MX_Defines_Syms DIRECTORY "stm32cubemx/cmake/stm32cubemx" DEFINITION
                            MX_Defines_Syms
)
get_directory_property (
  MX_Application_Src DIRECTORY "stm32cubemx/cmake/stm32cubemx" DEFINITION
                               MX_Application_Src
)
get_directory_property (
  STM32_Drivers_Src DIRECTORY "stm32cubemx/cmake/stm32cubemx" DEFINITION
                              STM32_Drivers_Src
)

# -----------------------------------------------------------------------------
# Define the stm32cubemx interface library.

# Note: It cannot be a library, since it makes extensive use of `weak` symbols.
add_library (stm32cubemx-interface INTERFACE EXCLUDE_FROM_ALL)

# -----------------------------------------------------------------------------
target_include_directories (stm32cubemx-interface INTERFACE ${MX_Include_Dirs})

target_sources (stm32cubemx-interface INTERFACE ${MX_Application_Src})

target_compile_definitions (stm32cubemx-interface INTERFACE ${MX_Defines_Syms})

target_compile_options (
  stm32cubemx-interface
  INTERFACE # Silence the warnings in the code generated by CubeMX.
            $<$<COMPILE_LANGUAGE:C>:-Wno-nested-externs> -Wno-padded
            -Wno-redundant-decls -Wno-unused-parameter
)

target_link_options (stm32cubemx-interface INTERFACE)

target_link_libraries (stm32cubemx-interface INTERFACE # Possibly m.
)

xpack_display_target_lists (stm32cubemx-interface)

# -----------------------------------------------------------------------------
# Aliases.
add_library (micro-os-plus::stm32cubemx ALIAS stm32cubemx-interface)
message (VERBOSE "> micro-os-plus::stm32cubemx -> stm32cubemx-interface")

# -----------------------------------------------------------------------------
# Define the stm32cubemx drivers objects library.

add_library (stm32cubemx-drivers-objects-library OBJECT EXCLUDE_FROM_ALL)

# -----------------------------------------------------------------------------
target_include_directories (
  stm32cubemx-drivers-objects-library PUBLIC ${MX_Include_Dirs}
)

target_sources (
  stm32cubemx-drivers-objects-library PRIVATE ${STM32_Drivers_Src}
)

target_compile_definitions (
  stm32cubemx-drivers-objects-library PUBLIC ${MX_Defines_Syms}
)

target_compile_options (
  stm32cubemx-drivers-objects-library
  PRIVATE # Silence the warnings in the code generated by CubeMX.
          $<$<COMPILE_LANGUAGE:C>:-Wno-bad-function-cast>
          $<$<COMPILE_LANGUAGE:C>:-Wno-nested-externs>
          -Wno-padded
          -Wno-redundant-decls
          -Wno-conversion
          -Wno-duplicated-branches
          -Wno-switch-enum
          -Wno-unused-parameter
          -Wno-switch-default
          -Wno-cast-align
)

target_link_options (stm32cubemx-drivers-objects-library PRIVATE)

target_link_libraries (stm32cubemx-drivers-objects-library PRIVATE)

xpack_display_target_lists (stm32cubemx-drivers-objects-library)

# -----------------------------------------------------------------------------
# Aliases.
add_library (
  micro-os-plus::stm32cubemx-drivers-objects-library ALIAS
  stm32cubemx-drivers-objects-library
)
message (
  VERBOSE
  "> micro-os-plus::stm32cubemx-drivers-objects-library -> stm32cubemx-drivers-objects-library"
)

# Define the device library.
add_library (device-stm32f411xe-interface INTERFACE EXCLUDE_FROM_ALL)

# -----------------------------------------------------------------------------
target_include_directories (device-stm32f411xe-interface INTERFACE "include")

target_sources (device-stm32f411xe-interface INTERFACE)

target_compile_options (
  device-stm32f411xe-interface
  INTERFACE $<$<COMPILE_LANGUAGE:C>:-Wno-missing-prototypes>
            -Wno-missing-declarations
            $<$<COMPILE_LANGUAGE:C>:-Wno-nested-externs>
            -Wno-padded
            $<$<COMPILE_LANGUAGE:C>:-Wno-strict-prototypes>
            -Wno-conversion
            -Wno-sign-conversion
            $<$<COMPILE_LANGUAGE:C>:-Wno-c++-compat>
            $<$<COMPILE_LANGUAGE:C>:-Wno-old-style-definition>
)

target_link_options (
  device-stm32f411xe-interface
  INTERFACE
  #
  -Wl,--wrap=main
  -Wl,--wrap=__libc_init_array
  #
  # Wrap all functions defined in syscalls.c.
  -Wl,--wrap=initialise_monitor_handles
  -Wl,--wrap=_getpid
  -Wl,--wrap=_kill
  -Wl,--wrap=_exit
  -Wl,--wrap=_read
  -Wl,--wrap=_write
  -Wl,--wrap=_close
  -Wl,--wrap=_fstat
  -Wl,--wrap=_lseek
  -Wl,--wrap=_isatty
  -Wl,--wrap=_lseek
  -Wl,--wrap=_open
  -Wl,--wrap=_close
  -Wl,--wrap=_fstat
  -Wl,--wrap=_lseek
  -Wl,--wrap=_isatty
  -Wl,--wrap=_wait
  -Wl,--wrap=_unlink
  -Wl,--wrap=_times
  -Wl,--wrap=_stat
  -Wl,--wrap=_link
  -Wl,--wrap=_fork
  -Wl,--wrap=_execve
  #
  # Prefer absolute paths over paths relative to the build folder.
  -T${CMAKE_CURRENT_SOURCE_DIR}/stm32cubemx/STM32F411xx_FLASH.ld
  # -T${CMAKE_CURRENT_SOURCE_DIR}/linker-scripts/mem.ld
  # -T${CMAKE_BINARY_DIR}/xpacks/@micro-os-plus/startup/linker-scripts/sections.ld
)

target_link_libraries (
  device-stm32f411xe-interface INTERFACE micro-os-plus::stm32cubemx
)

xpack_display_target_lists (device-stm32f411xe-interface)
